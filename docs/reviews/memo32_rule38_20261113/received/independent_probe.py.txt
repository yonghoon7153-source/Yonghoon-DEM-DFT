def dense_gate(reg,observed,violations):
 # Only a registered/observed criterion can support a verified violation.
 # Both known violation and missing evidence block entry; their reporting
 # priority when simultaneous is not used to authorize anything.
 if any(v and r and o for v,r,o in zip(violations,reg,observed)):return 'FAIL'
 if not all(reg) or not all(observed):return 'HOLD'
 return 'PASS'

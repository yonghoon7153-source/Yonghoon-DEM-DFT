+function S1ExactDecimal($Value) {
+    # Decimal strings cross Python->JSON->PS without a binary-double conversion.
+    # A floating/decimal CLR value has already lost the original lexical evidence.
+    if ($Value -is [string]) { $text=$Value }
+    elseif ($Value -is [byte] -or $Value -is [sbyte] -or $Value -is [int16] -or
+        $Value -is [uint16] -or $Value -is [int32] -or $Value -is [uint32] -or
+        $Value -is [int64] -or $Value -is [uint64]) {
+        $text=$Value.ToString([Globalization.CultureInfo]::InvariantCulture)
+    } else { throw 'DECIMAL_TRANSPORT_PRECISION' }
+    $m=[regex]::Match($text,'^([+-]?)(?:([0-9]+)(?:\.([0-9]*))?|\.([0-9]+))(?:[eE]([+-]?[0-9]+))?$')
+    if (-not $m.Success) { throw 'DECIMAL_TRANSPORT_FORMAT' }
+    $whole=$m.Groups[2].Value
+    $fraction=$(if ($m.Groups[4].Success) {$m.Groups[4].Value} else {$m.Groups[3].Value})
+    $exponent=0
+    if ($m.Groups[5].Success -and -not [int]::TryParse($m.Groups[5].Value,
+            [Globalization.NumberStyles]::AllowLeadingSign,[Globalization.CultureInfo]::InvariantCulture,[ref]$exponent)) {
+        throw 'DECIMAL_TRANSPORT_EXPONENT'
+    }
+    $digits=$whole+$fraction; $trimmed=$digits.TrimStart([char]'0')
+    if ($trimmed.Length -eq 0) { return @{sign=0;digits='0';position=[long]0} }
+    $leading=$digits.Length-$trimmed.Length
+    return @{sign=$(if ($m.Groups[1].Value -ceq '-') {-1} else {1});
+        digits=$trimmed.TrimEnd([char]'0');position=([long]$whole.Length+[long]$exponent-[long]$leading)}
+}
+function S1ExactCompare($Left,$Right) {

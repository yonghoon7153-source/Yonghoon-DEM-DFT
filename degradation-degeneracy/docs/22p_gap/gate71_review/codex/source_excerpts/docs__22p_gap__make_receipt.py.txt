330: def _outputs_agree(outputs: list[dict]) -> bool:
331:     """같은 schema·canonicalizer 를 쓰는 산출끼리 semantic digest 가 같은가.
332: 
333:     다르면 `build()` 가 여기서 멈춘다 — 영수증에 `false` 를 적어 두고 통과시키면
334:     "대조했다" 는 말이 다시 거짓이 된다.
335:     """
336:     by: dict[tuple, list] = {}
337:     for o in outputs:
338:         by.setdefault((o["semantic_schema"], o["canonicalizer"]),
339:                       []).append(o["semantic_sha256"])
340:     # ★ 27차 P1-6 — 초판은 산출이 **하나뿐이어도** `True` 였다. 비교 대상이
341:     #   없는데 "일치" 라고 적으면 그 말이 다시 거짓이 된다. 짝이 없으면 실패.
342:     lonely = [k for k, v in by.items() if len(v) < 2]
343:     if lonely:
344:         raise SystemExit(
345:             f"✗ 대조할 짝이 없다: {lonely} — 산출이 하나뿐이면 "
346:             "'일치' 가 아니라 **비교 불가**다 (27차 P1-6)")
347:     bad = [k for k, v in by.items() if len(set(v)) > 1]
348:     if bad:
349:         raise SystemExit(
350:             f"✗ 같은 schema·canonicalizer 인데 semantic digest 가 갈렸다: {bad}\n"
351:             "  같은 object 가 아니면 schema/role 을 갈라라 (26차 P1-6)")
352:     return True

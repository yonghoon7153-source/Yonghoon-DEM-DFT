def rr3_kind(ar, rec):
    """`arms_read_kind` 의 꼴 (고치기 전 판 = 함수 없음 → 'absent' · 예외 = 'exc <이름>')."""
    f = getattr(RR, 'arms_read_kind', None)
    if f is None:
        return 'absent'
    try:
        return f(ar, rec)[0]
    except Exception as e:                                # noqa: BLE001 — 분류 함수 예외 = 그 행 실패
        return f'exc {type(e).__name__}'


def rr3_ids(i=0, kit=SBE_KIT):
    """판독 실패 때 생산자가 남기는 파일 신원 (`arm_file_shas` 꼴 · 칸 셋)."""

def rr3_mock_group(outdir, files):

    원인 None · rc 0 · 확인 True) · 로그 한 줄 → (0, None, 끝 기록)."""
    def fake(argv, cwd, env, logf, left, gdir, name):
        Path(outdir).mkdir(parents=True)
        for nm, raw in files.items():
            (Path(outdir) / nm).write_bytes(raw)
        g = Path(gdir)
        g.mkdir(parents=True, exist_ok=True)
        tok = 'a' * 32
        end = {'schema': RR.GROUP_SCHEMA, 'name': name, 'token': tok, 'by': 'chain', 'cause': None, 'rc': 0, 'confirmed': True}

def rr3_summ(code, run, runs):
    return {'status': 'STOPPED', 'stop': {'code': code, 'run': run, 'reason': None}, 'runs': copy.deepcopy(list(runs))}


def rr3_row(folder, s):
    """요약 s · 시도 폴더 → [원인, 재실행 자격, stop_record_problem 사유 (None | 글)]."""
    fl = RR.result_fields(folder, s)
    return [fl['cause'], fl['resume_ok'], RR.stop_record_problem(s, folder)]


def rr3_ledger(folder, s):
    """실제 결과 줄 → 원장 읽기 → 다음 --resume 1 항목 차례 (판정문 탐침과 같은 합성 머리 · 항목 · 시작 기록 · git · 잠금 · recover 없음) → [원장 결과 줄 resume_ok ·
    attempt_order_problem 멈춤 | None · result_line_problem (폴더 증거로 다시 셈 대조)]."""
    folder = Path(folder)
    (folder / 'ladder_summary.json').write_bytes(RR.canon_bytes(s))
    head = {'kind': 'header', 'schema': RR.LEDGER_SCHEMA, 'plan_sha256': 'b' * 64, 'utc': 'synthetic'}
    ent = {'kind': 'entry', 'n': 1, 'target': 'ladder:P0', 'machine': 'synthetic', 'resume_of': None, 'plan_sha256': 'b' * 64, 'code_key_sha256': 'c' * 64,
           'env': {}, 'bed': {}, 'utc': 'synthetic'}
    start = {'kind': 'start', 'n': 1, 'base': str(folder.parent.resolve()), 'folder': folder.name,
             'host': {'machine_id': None, 'boot_id': None, 'pidns': None}, 'utc': 'synthetic', 'chain_pid': 1, 'chain_start': None}
    res = RR.attempt_result_line(1, folder, s, 'recover')
    st = RR.ledger_parse(''.join(RR.ledger_line(x) for x in (head, ent, start, res)).encode('utf-8'))
    order = RR.attempt_order_problem(st, dict(ent, n=2, resume_of=1), {'exec_policy': {'max_attempts': 2}})
    return [st['results'][1]['resume_ok'], order[0] if order else None, RR.result_line_problem(1, folder, res)]


def rr3_check_rows(rows, bad, items):
    """items (이름, 폴더, 요약, 기대 [원인, 자격], 사유에 있어야 할 낱말 (None = 사유 없음)) → rows · bad."""
    for nm, f, s, want, word in items:
        cause, ok, prob = rr3_row(f, s)
        rows[nm] = [cause, ok, None if prob is None else prob[:160]]
        code = s['stop']['code']
        prob_ok = (prob is None) if word is None else (isinstance(prob, str) and code in prob and word in prob)
        if [cause, ok] != want or not prob_ok:
            bad[nm] = {'got': rows[nm], 'want': want, 'word': word}

def rr3_calls(src):
    """함수 (클래스.메서드) 이름 → (그 몸 (안쪽 함수 포함) 에서 부른 이름 목록 · ARM_PUBLIC 위 내포 수)."""
    calls, proj = {}, {}

    def walk(node, prefix):
        for ch in ast.iter_child_nodes(node):
            if isinstance(ch, ast.ClassDef):
                walk(ch, prefix + ch.name + '.')
            elif isinstance(ch, (ast.FunctionDef, ast.AsyncFunctionDef)):
                q = prefix + ch.name
                calls[q] = [x.func.id if isinstance(x.func, ast.Name) else x.func.attr for x in ast.walk(ch)
                            if isinstance(x, ast.Call) and isinstance(x.func, (ast.Name, ast.Attribute))]
                proj[q] = sum(1 for x in ast.walk(ch) if isinstance(x, ast.comprehension) and isinstance(x.iter, ast.Name) and x.iter.id == 'ARM_PUBLIC')
                walk(ch, q + '.')
            else:
                walk(ch, prefix)
    walk(ast.parse(src), '')
    return calls, proj

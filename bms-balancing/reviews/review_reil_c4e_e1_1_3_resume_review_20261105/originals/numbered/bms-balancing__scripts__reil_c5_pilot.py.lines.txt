418: def orchestrate(runs, ctx, *, n_workers, limits, t0, clock=time.monotonic, rss_fn=None, disk_fn=None, out_dir=None,

513:                     reason = f"디스크 표본: 새 산출 {used} B (상한 {limits['disk_new']}) · 여유 {free} B (하한 {limits['disk_free']})"
514:                     break
515:             elif now - last_disk > limits["disk_gap"]:
516:                 reason = f"디스크 표본 누락 {now - last_disk:.1f} s > {limits['disk_gap']} s"

540:     states = assemble_states(runs, ev, prog, reason, t_stop)

812:     t_meas = clock()

816:     end_hashes = source_hashes(watch_files)
817:     changed = source_changed(start_hashes, end_hashes)
818:     rel_out = os.path.relpath(out_dir, REPO)
819:     end = {"git_start": git0, "git_end": git1, "dirty_end": split_dirty(git1["porcelain"], [rel_out]),
820:            "source_hashes_start": start_hashes, "source_hashes_end": end_hashes, "source_changed": changed,
821:            "model_digest_start": model_h0, "model_digest_end": model_digest(ctx["model"]),
822:            "halfcell_digest_start": half_h0, "halfcell_digest_end": model_digest(ctx["halfcell"]),

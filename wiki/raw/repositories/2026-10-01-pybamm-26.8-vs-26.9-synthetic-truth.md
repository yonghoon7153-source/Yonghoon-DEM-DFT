# PyBaMM 26.8.0.0 ↔ 26.9.0.0 — 우리 합성 truth 두 조건의 비교 (2026-10-01 실측)

## 조건 · 격리

- 코드: `git archive 9ca6df54e34db5023ce2bfc2d1e60434affd4eaf` 사본의 `degradation-degeneracy/` 에서 돌렸다. 운영 작업 트리 · 등록부 · 산출물에는 쓰지 않았다 (쓴 것은 스크래치패드 JSON 과 stdout). grid 파이프라인이 한 조건을 만드는 **같은 함수** `src.grid._solve_condition` 을 두 인터프리터에서 불렀다.
- 26.8 = 컨테이너 시스템 python: pybamm 26.8.0.0 · pybammsolvers 0.9.1 · casadi 3.7.2. 26.9 = 버리는 별도 venv: `pybamm[all]==26.9.0.0` · pybammsolvers 0.10.0 · casadi 3.8.1. 둘 다 `effective_class = IDAKLUSolver` · rtol = atol = 1e-6.
- config `configs/grid_22p.yaml` · protocol `charge_first` (0.05 C) · 조건 2: pristine (0, 0, 0) · 열화 (lli 0.10 · lam_pe 0.13 · lam_ne 0.13 · type de) · noise 0.
- 기본 격자 (우리 코드는 `var_pts` · `submesh_types` 를 설정하지 않는다 → PyBaMM 기본): x_n · x_s · x_p 각 20 점 → 간격 4.26 µm · 0.60 µm · 3.78 µm — **subdomain 안은 균일, 접합부에서는 비균일**. submesh 는 전부 `Uniform1DSubMesh`.
- 두 번째 실행 `UNIFORM_X=1`: `pybamm.Simulation` 에 `var_pts` x_n 71 · x_s 10 · x_p 63 을 강제 → 세 영역 간격이 모두 ≈1.2 µm (접합부도 균일). 격자 효과와 solver 스택 효과를 가르기 위한 대조.

## 드라이버 (cmp_pybamm.py — 실행한 그대로)

```python
"""PyBaMM 버전 간 합성 truth 비교 드라이버 (2026-10-01 · 브리핑 후속 A).

우리 grid 파이프라인이 한 조건을 만드는 **같은 함수**(`src.grid._solve_condition`)를 그대로 부른다.
쓰는 것은 stdout 과 argv[1] 의 JSON 하나뿐 — 저장소 산출물 · 등록부에 아무 것도 쓰지 않는다.
`git archive` 로 뽑은 사본의 degradation-degeneracy/ 에서 `PYTHONPATH=. python cmp_pybamm.py out.json` 로 돈다.
"""
import json
import sys
import time
from dataclasses import asdict

import numpy as np
import pybamm

from src.baseline import compute_discharged_state
from src.config import load_config
from src.grid import Condition, _solve_condition

CFG = "configs/grid_22p.yaml"
CONDS = [
    Condition(lli=0.0, lam_pe=0.0, lam_ne=0.0, lam_pe_type="de", lam_ne_type="de", noise=0.0, seed=0),
    Condition(lli=0.10, lam_pe=0.13, lam_ne=0.13, lam_pe_type="de", lam_ne_type="de", noise=0.0, seed=0),
]


def mesh_spacing(cfg):
    """기본 geometry · var_pts 로 만든 격자에서 subdomain 별 x 간격 (접합부가 비균일인가를 보인다)."""
    from src.model import build_model
    from src.runner import build_param
    model = build_model(cfg)
    param = build_param(cfg)
    geo = model.default_geometry
    param.process_geometry(geo)
    mesh = pybamm.Mesh(geo, model.default_submesh_types, model.default_var_pts)
    out = {}
    for dom in ("negative electrode", "separator", "positive electrode"):
        e = np.asarray(mesh[dom].edges)
        dx = np.diff(e)
        out[dom] = {"n": int(len(dx)), "dx_min": float(dx.min()), "dx_max": float(dx.max())}
    out["submesh_types"] = {k: type(v).__name__ if not isinstance(v, type) else v.__name__
                            for k, v in model.default_submesh_types.items()}
    return out


def _force_uniform_x():
    """UNIFORM_X=1 — x 격자를 세 subdomain 에서 같은 간격(1.2 µm)으로: x_n 71 · x_s 10 · x_p 63.
    #5755 는 uniform 격자에서 bit-for-bit 불변이라 했으므로, 이 조건에서 남는 버전 차는 solver 쪽이다."""
    import os
    if os.environ.get("UNIFORM_X") != "1":
        return None
    orig = pybamm.Simulation.__init__
    def init(self, model, *a, var_pts=None, **kw):
        if var_pts is None:
            var_pts = dict(model.default_var_pts)
            var_pts.update({"x_n": 71, "x_s": 10, "x_p": 63})
        return orig(self, model, *a, var_pts=var_pts, **kw)
    pybamm.Simulation.__init__ = init
    return {"x_n": 71, "x_s": 10, "x_p": 63}


def main(path):
    forced = _force_uniform_x()
    cfg = load_config(CFG)
    t0 = time.time()
    d = compute_discharged_state(cfg)
    res = {"pybamm": pybamm.__version__, "cfg": CFG, "forced_var_pts": forced, "discharged_state": asdict(d), "conditions": []}
    try:
        res["mesh"] = mesh_spacing(cfg)
    except Exception as e:  # noqa: BLE001 — 격자 진단은 부가 정보
        res["mesh"] = {"error": f"{type(e).__name__}: {e}"}
    proto = cfg.get("grid_protocol_override", "charge_first")
    for c in CONDS:
        r = _solve_condition(cfg, c, asdict(d), proto)
        res["conditions"].append({k: r.get(k) for k in ("cond", "cond_id", "error", "elapsed_s", "solver_identity",
                                                         "q_mah", "x_norm", "v_pe", "v_ne", "v_full")})
    res["wall_s"] = time.time() - t0
    with open(path, "w", encoding="utf-8") as f:
        json.dump(res, f)
    print(json.dumps({"pybamm": res["pybamm"], "wall_s": round(res["wall_s"], 1), "mesh": res["mesh"],
                      "errors": [c["error"] for c in res["conditions"]],
                      "q_mah": [c["q_mah"] for c in res["conditions"]]}, ensure_ascii=False))


if __name__ == "__main__":
    main(sys.argv[1])
```

def utc() -> str:
    return datetime.datetime.now(datetime.timezone.utc).strftime("%Y-%m-%dT%H:%M:%S.%f")[:-3] + "Z"


def sha256(b: bytes) -> str:
    return hashlib.sha256(b).hexdigest()


def is_sha256(v) -> bool:
    return isinstance(v, str) and len(v) == 64 and all(c in "0123456789abcdef" for c in v)

def read_regular(path) -> tuple:
    """일반 파일만 읽는다 → (바이트, None) · 아니면 (None, 사유). 절대 경로 · resolve 가 자기 자신 (링크된 상위 폴더 거부) · 읽기 오류도 사유로."""
    p = Path(path)
    if not p.is_absolute():
        return None, "절대 경로가 아니다"
    try:
        st = os.lstat(p)
    except OSError as e:
        return None, f"없음 ({e.__class__.__name__})"
    if not stat.S_ISREG(st.st_mode):
        return None, "일반 파일이 아니다 (심볼릭 링크 · 디렉터리 등)"
    try:
        if p.resolve() != p:
            return None, f"resolve 경로가 다르다: {p.resolve()}"
        return p.read_bytes(), None
    except OSError as e:
        return None, f"읽기 실패 ({e.__class__.__name__}: {e})"


def load_sources(path, want_sha256, *, relpaths=MODULE_RELPATHS, schema=SOURCES_SCHEMA) -> tuple:
    """신뢰 목록 → ({경로: {sha256, bytes}}, 문제). 목록 바이트 sha256 = 고정값 · schema · 경로 정확히 relpaths · 항목 형식."""
    b, why = read_regular(path)
    if b is None:
        return None, [f"목록 {path}: {why}"]
    got = sha256(b)
    if got != want_sha256:
        return None, [f"목록 sha256 {got} ≠ 고정값 {want_sha256}"]
    try:
        doc = json.loads(b)
    except ValueError as e:
        return None, [f"목록이 JSON 이 아니다: {e}"]
    files = doc.get("files") if isinstance(doc, dict) else None
    if not isinstance(files, dict) or doc.get("schema") != schema:
        return None, [f"목록 형식이 다르다 (schema {schema} · files)"]
    missing, extra = [r for r in relpaths if r not in files], sorted(set(files) - set(relpaths))
    if missing or extra:
        return None, [f"목록 경로 집합이 다르다 — 누락 {missing} · 목록 밖 {extra}"]
    out = {}
    for rel in relpaths:
        e = files[rel]
        s, n = (e.get("sha256"), e.get("bytes")) if isinstance(e, dict) else (None, None)
        if not is_sha256(s) or not isinstance(n, int) or isinstance(n, bool) or n < 0:
            return None, [f"목록 항목 {rel} 형식 (sha256 소문자 16 진 64 · bytes 정수 ≥ 0): {e!r}"]
        out[rel] = {"sha256": s, "bytes": n}
    return out, []


def check_files(base, expected: dict) -> tuple:
    """base/<이름> 의 실제 바이트 (일반 파일 · resolve 일치) 를 기대 sha256 · bytes 와 대조 → (행, 문제)."""
    rows, probs = {}, []
    for rel, want in expected.items():
        p = Path(base) / rel
        b, why = read_regular(p)
        row = {"path": str(p), "expected": dict(want)}
        if b is None:
            row.update(ok=False, why=why)
            probs.append(f"{rel}: {why}")
        else:
            row.update(sha256=sha256(b), bytes=len(b))
            row["ok"] = row["sha256"] == want["sha256"] and row["bytes"] == want["bytes"]
            if not row["ok"]:
                probs.append(f"{rel}: sha256 {row['sha256']} · {row['bytes']} B ≠ 기대 {want}")
        rows[rel] = row
    return rows, probs

def leaf_shadow_problems(bms=BMS, leaves=LEAVES) -> list:
    """N1 — 승인 `<이름>.py` 보다 먼저 잡히는 것: 같은 이름 패키지 (`<이름>/__init__*` — FileFinder 는 패키지를 모듈 파일보다 먼저) · 같은 이름 확장 모듈
    (확장 접미사는 `.py` 보다 먼저)."""
    d, probs = Path(bms) / "scripts", []
    for leaf in leaves:
        probs += [f"{d / leaf / ('__init__' + s)} 가 있다 — 승인 {leaf}.py 보다 먼저 잡히는 패키지" for s in importlib.machinery.all_suffixes()
                  if (d / leaf / f"__init__{s}").exists()]
        probs += [f"{d / (leaf + s)} 가 있다 — 승인 {leaf}.py 보다 먼저 잡히는 확장 모듈" for s in importlib.machinery.EXTENSION_SUFFIXES
                  if (d / f"{leaf}{s}").exists()]
    return probs


def shadow_problems(entries, bms=BMS) -> list:
    """`scripts` 최상위 이름이 우리 `bms-balancing/scripts/` 로만 가는가 — 우리 쪽은 `__init__` · 같은 이름 모듈 파일 없는 네임스페이스여야 하고, 자식
    sys.path 의 어느 항목에도 `scripts` 정규 패키지 · 모듈 (소스 · 바이트코드 · 확장 · zip 안) 이 없어야 한다."""
    suffixes = tuple(importlib.machinery.all_suffixes())
    names = [f"scripts/__init__{s}" for s in suffixes] + [f"scripts{s}" for s in suffixes]
    probs = [f"{Path(bms) / n} 가 있다 — 우리 `scripts` 는 `__init__` · 같은 이름 모듈 없는 네임스페이스여야 한다" for n in names
             if (Path(bms) / n).exists()]
    if not isinstance(entries, list):
        return probs + [f"탐침 sys.path 가 목록이 아니다: {entries!r}"]
    for e in entries:
        if not isinstance(e, str) or not e:
            probs.append(f"sys.path 항목 {e!r} (빈 항목 = 현재 폴더)")
            continue
        p = Path(e)
        if p.is_dir():
            probs += [f"{p / n} — `scripts` 를 가린다" for n in names if (p / n).exists()]
        elif p.is_file() and zipfile.is_zipfile(p):
            with zipfile.ZipFile(p) as z:
                inside = set(z.namelist())
            probs += [f"{p}!{n} — zip 안의 `scripts` 가 가린다" for n in names if n in inside]
    return probs


def spec_problems(specs, bms=BMS, leaves=LEAVES) -> list:
    """N1 — 탐침이 finder 에 물은 적재 선택: `scripts` = PathFinder 의 네임스페이스 (loader 없음 · 첫 위치 = 우리 scripts) · 셋 = PathFinder ·
    SourceFileLoader · origin = 승인 `.py` 절대 경로 · 패키지 아님."""
    probs, want_loc = [], str(Path(bms) / "scripts")
    top = specs.get("scripts") if isinstance(specs, dict) else None
    if (not isinstance(top, dict) or top.get("finder") != PATHFINDER or top.get("loader") not in NAMESPACE_LOADERS
            or top.get("origin") not in (None, "namespace") or not top.get("locations") or top["locations"][0] != want_loc):
        probs.append(f"scripts 가 {want_loc} 를 첫 위치로 하는 PathFinder 네임스페이스가 아니다: {top}")
    for leaf in leaves:
        s = specs.get(f"scripts.{leaf}") if isinstance(specs, dict) else None
        want = str(Path(bms) / "scripts" / f"{leaf}.py")
        if (not isinstance(s, dict) or s.get("finder") != PATHFINDER or s.get("loader") != SOURCE_LOADER or s.get("origin") != want
                or s.get("locations") is not None):
            probs.append(f"scripts.{leaf} 의 적재 선택이 승인 소스가 아니다 (기대 {want} · SourceFileLoader · PathFinder): {s}")
    return probs


def interp(py, prefix) -> list:
    """자식 · 탐침 공통의 해석기 · 플래그 (고정)."""
    return [str(py), "-I", "-B", "-X", f"pycache_prefix={prefix}"]


def flag_problems(probe_doc: dict, prefix, py) -> list:
    want = {"executable": str(py), "isolated": 1, "ignore_environment": 1, "no_user_site": 1, "safe_path": True,
            "dont_write_bytecode": 1, "pycache_prefix": str(prefix)}
    return [f"탐침 {k}: {probe_doc.get(k)!r} ≠ {v!r}" for k, v in want.items() if probe_doc.get(k) != v]


def probe(py, prefix, *, env, cwd, bms=BMS, timeout=PROBE_TIMEOUT_S) -> tuple:
    """자식과 같은 해석기 · 플래그 · env · cwd 로 플래그 · sys.path · 적재 선택만 읽는다 (우리 모듈 import · 실행 0) → (문서, None) · (None, 사유)."""
    code = f"BMS = {json.dumps(str(bms))}\nLEAVES = {json.dumps(list(LEAVES))}\n{PROBE}"
    try:
        r = subprocess.run(interp(py, prefix) + ["-c", code], cwd=str(cwd), env=env, capture_output=True, timeout=timeout)
    except (OSError, subprocess.TimeoutExpired) as e:
        return None, f"탐침 실패 ({e.__class__.__name__})"
    if r.returncode != 0:
        return None, f"탐침 rc {r.returncode}: {r.stderr.decode('utf-8', 'replace')[-300:]}"
    try:
        doc = json.loads(r.stdout)
    except ValueError as e:
        return None, f"탐침 출력이 JSON 이 아니다: {e}"
    return (doc, None) if isinstance(doc, dict) else (None, "탐침 출력이 객체가 아니다")


def build_argv(step, evidence, section, prefix, *, py=PY, bms=BMS, venv=VENV, seal=SEAL, data=DATA, sources=SOURCES,
               sources_sha256=SOURCES_SHA256, seal_manifest_sha256=SEAL_MANIFEST_SHA256) -> list:
    """세 단계의 고정 argv (승인 요청 v2 §6). 바뀌는 것은 증거 폴더 · 승인 절 · pycache 폴더뿐이다."""
    head, ev, run_script = interp(py, prefix), Path(evidence), str(Path(bms) / RUN_SCRIPT)
    if step == "identity":
        return head + [str(ev / IDENTITY_HELPER), str(venv), str(seal), seal_manifest_sha256, str(ev / "03_identity.json")]
    if step == "fetch":
        return head + [run_script, "fetch", "--dest", str(data)]
    if step == "run":
        return head + [run_script, "run", "--venv", str(venv), "--seal", str(seal), "--seal-manifest-sha256", seal_manifest_sha256,
                       "--xlsx", str(Path(data) / "LFP_Data.xlsx"), "--util", str(Path(data) / "util_LFP.py"),
                       "--sources", str(sources), "--sources-sha256", sources_sha256, "--out", str(ev / "run"),
                       "--approval", f"상태 문서 {section} (사용자 E1-1–3 승인 · {REQUEST})"]
    raise ValueError(f"단계는 {sorted(STEPS)} 뿐이다: {step!r}")


def child_env(base: dict, work=WORK) -> tuple:
    """자식 env → (env, 뺀 이름). 비밀 형태 · `GIT_SSL_CAINFO` 밖의 `GIT_*` · `SSH_ASKPASS` 를 뺀다 (N4). 값은 기록하지 않는다."""
    removed = sorted(k for k in base if SECRET.search(k) or k in DROP or (k.startswith("GIT_") and k not in GIT_KEEP))
    env = {k: v for k, v in base.items() if k not in removed}
    env.update({"MPLBACKEND": "Agg", "MPLCONFIGDIR": str(Path(work) / "mpl"), "TMPDIR": str(Path(work) / "tmp"), "OMP_NUM_THREADS": "1",
                "OPENBLAS_NUM_THREADS": "1", "MKL_NUM_THREADS": "1", "GIT_TERMINAL_PROMPT": "0"})
    return env, removed


# ---------------------------------------------------------------- Git (기록 · 출처)
def git_run(args, *, env=None, cwd=None, timeout=GIT_TIMEOUT_S) -> dict:
    """git 하나 → {argv, rc, out, err[, error]} — 실패를 빈 정상 출력으로 바꾸지 않는다 (rc · 오류를 그대로 둔다)."""
    argv = ["git", *[str(a) for a in args]]
    try:
        r = subprocess.run(argv, cwd=None if cwd is None else str(cwd), env=env, capture_output=True, text=True, timeout=timeout)
        return {"argv": argv, "rc": r.returncode, "out": r.stdout, "err": r.stderr[-500:]}
    except (OSError, subprocess.TimeoutExpired) as e:
        return {"argv": argv, "rc": None, "out": "", "err": "", "error": f"{e.__class__.__name__}: {e}"}


def git_config_problems(listing: str) -> list:
    """`git config --list --show-origin` 출력에서 유효 출처를 바꾸거나 우회하는 설정 (URL 바꿈 · 프록시 · 자격 도우미 · TLS 검증 끔)."""
    probs = []
    for line in listing.splitlines():
        origin, _, kv = line.partition("\t")
        key, _, val = kv.partition("=")
        k = key.strip().lower()
        if GIT_CONFIG_DENY.match(k):
            probs.append(f"git 설정 {k} ({origin}) — URL 바꿈 · 프록시 · 자격 도우미는 허용하지 않는다")
        elif GIT_SSLVERIFY.match(k) and val.strip().lower() in ("false", "no", "off", "0", ""):
            probs.append(f"git 설정 {k}={val.strip()} ({origin}) — TLS 검증을 끄지 않는다")
    return probs


def fetch_source_precheck(env, cwd, url=SOURCE_URL) -> tuple:
    """N4 (fetch 전 · 원격 접속 0) — 자식 env · 작업 폴더에서 Git 설정 금지 키 없음 · `ls-remote --get-url` (rewrite 적용 URL) = 고정 URL."""
    c = git_run(["config", "--list", "--show-origin"], env=env, cwd=cwd)
    g = git_run(["ls-remote", "--get-url", url], env=env, cwd=cwd)
    probs = []
    if c["rc"] != 0:
        probs.append(f"git config --list 실패 (rc {c['rc']} {c.get('error', '')})")
    else:
        probs += git_config_problems(c["out"])
    if g["rc"] != 0 or g["out"].strip() != url:
        probs.append(f"유효 출처 {g['out'].strip()!r} (rc {g['rc']} {g.get('error', '')}) ≠ 고정 {url}")
    return {"config": c, "get_url": g}, probs


def fetch_source_postcheck(env, data, url=SOURCE_URL, commit=SOURCE_COMMIT) -> tuple:
    """N4 (fetch 뒤 · 원격 접속 0) — 받은 저장소의 `remote.origin.url` · 유효 출처 = 고정 URL · `.git/shallow` = 고정 커밋 하나 · alternates 없음.
    FETCH_HEAD 는 기록만 한다."""
    gd = Path(data) / "git"
    u = git_run(["-C", gd, "config", "--get", "remote.origin.url"], env=env, cwd=data)
    g = git_run(["-C", gd, "ls-remote", "--get-url", "origin"], env=env, cwd=data)
    probs = []
    if u["rc"] != 0 or u["out"].strip() != url:
        probs.append(f"remote.origin.url {u['out'].strip()!r} (rc {u['rc']}) ≠ 고정 {url}")
    if g["rc"] != 0 or g["out"].strip() != url:
        probs.append(f"받은 저장소의 유효 출처 {g['out'].strip()!r} (rc {g['rc']}) ≠ 고정 {url}")
    sh, why = read_regular(gd / ".git" / "shallow")
    shallow = sh.decode("utf-8", "replace").split() if sh is not None else None
    if shallow != [commit]:
        probs.append(f".git/shallow {shallow if sh is not None else why} ≠ [{commit}]")
    alt = gd / ".git" / "objects" / "info" / "alternates"
    if alt.exists() or alt.is_symlink():
        probs.append(f"{alt} 가 있다 — 객체를 다른 곳에서 빌려 오지 않는다")
    fh, _ = read_regular(gd / ".git" / "FETCH_HEAD")
    return {"remote_url": u, "get_url": g, "shallow": shallow, "fetch_head": None if fh is None else fh.decode("utf-8", "replace")[:500]}, probs


def record_problems(records: dict) -> list:
    """N3 — Git 기록 (HEAD 시작 · 끝 · porcelain 시작 · 끝) 이 성공했고 단계 중 HEAD 가 그대로인가."""
    probs, heads = [], {}
    for k in ("head_start", "head_end"):
        r = records.get(k) or {}
        h = (r.get("out") or "").strip()
        if r.get("rc") != 0 or not re.fullmatch(r"[0-9a-f]{40}", h):
            probs.append(f"Git 기록 {k} 실패 (rc {r.get('rc')} {r.get('error', '')})")
        else:
            heads[k] = h
    for k in ("porcelain_start", "porcelain_end"):
        r = records.get(k) or {}
        if r.get("rc") != 0:
            probs.append(f"Git 기록 {k} 실패 (rc {r.get('rc')} {r.get('error', '')})")
    if len(heads) == 2 and heads["head_start"] != heads["head_end"]:
        probs.append(f"단계 중 HEAD 가 바뀌었다: {heads['head_start']} → {heads['head_end']}")
    return probs


# ---------------------------------------------------------------- 사전 · 사후 대조 · 판정
def precheck(step, evidence, argv, prefix, *, env, repo=REPO, py=PY, seal=SEAL, work=WORK, data=DATA, data_files=DATA_FILES,
             sources=SOURCES, sources_sha256=SOURCES_SHA256, seal_manifest_sha256=SEAL_MANIFEST_SHA256, lock_sha256=C6_LOCK_SHA256,
             extra=EXTRA_EXECUTED, helper_sha256=IDENTITY_HELPER_SHA256) -> dict:
    """자식 시작 전 대조 (모듈 docstring 의 1). 문제가 하나라도 있으면 ok false — 자식을 띄우지 않는다."""
    bms, probs = Path(repo) / "bms-balancing", []
    entries, p = load_sources(sources, sources_sha256)
    probs += p
    modules, p = check_files(bms, entries) if entries else ({}, [])
    probs += p
    extra_rows, p = check_files(repo, extra)
    probs += p
    seal_rec, p = seal_lock_problems(seal, seal_manifest_sha256, lock_sha256, (extra.get(ENV_PROFILE) or {}).get("sha256"))
    probs += p
    probs += leaf_shadow_problems(bms)
    if list(argv[:5]) != interp(py, prefix):
        probs.append(f"해석기 · 플래그가 고정값과 다르다: {list(argv[:5])}")
    script = argv[5] if len(argv) > 5 else None
    helper = None
    if step in ("fetch", "run"):
        if script != str(bms / RUN_SCRIPT) or script != (modules.get(RUN_SCRIPT) or {}).get("path"):
            probs.append(f"실행 스크립트 {script} ≠ 대조한 run 모듈 경로")
    elif step == "identity":
        b, why = read_regular(script) if script else (None, "argv 에 없음")
        helper = {"path": script, "sha256": sha256(b) if b is not None else None, "expected": helper_sha256, "why": why}
        if b is None or helper["sha256"] != helper_sha256 or Path(script) != Path(evidence) / IDENTITY_HELPER:
            probs.append(f"identity helper {script}: {why or helper['sha256']} ≠ 고정 {helper_sha256}")
    else:
        probs.append(f"단계 {step!r}")
    doc, why = probe(py, prefix, env=env, cwd=bms, bms=bms)
    if doc is None:
        probs.append(why)
    else:
        probs += flag_problems(doc, prefix, py) + shadow_problems(doc.get("path"), bms) + spec_problems(doc.get("specs"), bms)
    data_rows = source = None
    if step == "run":
        data_rows, p = check_files(data, data_files)
        probs += p
    if step == "fetch":
        if Path(data).exists() or Path(data).is_symlink():
            probs.append(f"자료 폴더 {data} 가 이미 있다 — fetch 는 새 폴더에만")
        source, p = fetch_source_precheck(env, work)
        probs += p
    return {"ok": not probs, "problems": probs, "utc": utc(), "step": step,
            "sources": {"path": str(sources), "sha256_fixed": sources_sha256, "entries": entries}, "modules": modules,
            "extra_executed": extra_rows, "seal_lock": seal_rec, "helper": helper, "probe": doc, "data": data_rows, "source": source,
            "git": shutil.which("git", path=env.get("PATH"))}


def identity_link(path, modules: dict, *, relpaths=MODULE_RELPATHS) -> list:
    """run 의 IDENTITY.json (시작 · 끝 표) 모듈 행의 경로 · sha256 · bytes = 사전 대조의 경로 · 목록 (실제 적재 경로 ↔ 대조 경로)."""
    if set(modules) != set(relpaths):
        return [f"사전 대조 모듈 집합이 다섯이 아니다: {sorted(modules)}"]
    try:
        rec = json.loads(Path(path).read_text(encoding="utf-8"))
    except (OSError, ValueError) as e:
        return [f"{path} 를 읽지 못했다 ({e.__class__.__name__})"]
    probs = []
    for side in ("start", "end"):
        table = rec.get(side) if isinstance(rec, dict) else None
        if not isinstance(table, dict):
            probs.append(f"IDENTITY.json {side} 표가 없다")
            continue
        for rel, row in modules.items():
            r = table.get(rel)
            if not isinstance(r, dict):
                probs.append(f"{side} · {rel}: 행이 없다")
            elif r.get("path") != row["path"]:
                probs.append(f"{side} · {rel}: 적재 경로 {r.get('path')} ≠ 대조 경로 {row['path']}")
            elif r.get("sha256") != row["expected"]["sha256"] or r.get("bytes") != row["expected"]["bytes"]:
                probs.append(f"{side} · {rel}: sha256 · bytes 가 목록과 다르다")
    return probs


def postcheck(step, evidence, pre, prefix, child, *, env, repo=REPO, data=DATA, data_files=DATA_FILES, extra=EXTRA_EXECUTED,
              records=None) -> dict:
    """자식이 끝난 뒤 대조 (모듈 docstring 의 3)."""
    entries = (pre.get("sources") or {}).get("entries")
    if not entries:
        return {"ok": False, "problems": ["사전 대조의 목록 항목이 없다"], "utc": utc()}
    child = child or {}
    modules, probs = check_files(Path(repo) / "bms-balancing", entries)
    extra_rows, p = check_files(repo, extra)
    probs += p
    data_rows = link = source = None
    if step in ("fetch", "run"):
        data_rows, p = check_files(data, data_files)
        probs += p
    if step == "fetch":
        source, p = fetch_source_postcheck(env, data)
        probs += p
    if step == "run":
        link = identity_link(Path(evidence) / "run" / "IDENTITY.json", pre.get("modules") or {})
        probs += link
    left = sorted(str(x) for x in Path(prefix).rglob("*")) if Path(prefix).is_dir() else ["(pycache 폴더가 없다)"]
    if left:
        probs.append(f"pycache 폴더에 남은 것 {left[:5]}")
    group = child.get("group") or {}
    if group.get("status") != "empty":
        probs.append(f"자식 프로세스 그룹 상태 {group.get('status')!r} — 비어 있음이 확인되지 않았다")
    if child.get("timed_out") and not child.get("exited_after_kill"):
        probs.append("timeout 뒤 자식이 제한 시간 안에 끝나지 않았다 (미종료)")
    if child.get("errors"):
        probs.append(f"자식 단계 오류 {child['errors']}")
    probs += record_problems(records or {})
    return {"ok": not probs, "problems": probs, "utc": utc(), "modules": modules, "extra_executed": extra_rows, "data": data_rows,
            "source": source, "identity_link": link}


def child_result(step, stdout_text: str) -> tuple:
    """자식 stdout 의 마지막 비어 있지 않은 줄 (JSON 객체) → (ok, 요약). run 은 ok AND final.complete."""
    lines = [x for x in stdout_text.splitlines() if x.strip()]
    try:
        doc = json.loads(lines[-1]) if lines else None
    except ValueError:
        doc = None
    if not isinstance(doc, dict):
        return False, {"why": "stdout 마지막 줄이 JSON 객체가 아니다"}
    if step == "run":
        fin = doc.get("final") if isinstance(doc.get("final"), dict) else {}
        return (doc.get("ok") is True and fin.get("complete") is True,
                {"ok": doc.get("ok"), "final_complete": fin.get("complete"), "E3a": fin.get("E3a"), "stop": doc.get("stop")})
    return doc.get("ok") is True, {"ok": doc.get("ok"), "stop": doc.get("stop"), "problems": doc.get("problems")}


def stage_verdict(*, precheck_ok, child_ok, rc, timed_out, postcheck_ok) -> dict:
    """단계 판정 = 다섯 항의 AND (§72 조건 3 의 넷 + 사후 대조)."""
    terms = {"precheck_ok": precheck_ok is True, "child_ok": child_ok is True,
             "rc_0": isinstance(rc, int) and not isinstance(rc, bool) and rc == 0, "timeout_false": timed_out is False,
             "postcheck_ok": postcheck_ok is True}
    return {"rule": "precheck_ok AND child_ok AND rc_0 AND timeout_false AND postcheck_ok", "terms": terms, "complete": all(terms.values())}


# ---------------------------------------------------------------- 자식 · 프로세스 그룹 (N2 · N3)
def proc_members(pgid, *, proc=PROC, read=None) -> dict:
    """/proc 에서 프로세스 그룹 pgid 의 구성원 [(pid, 상태)] · 사라진 PID 수 · 오류. PID 소멸 (FileNotFoundError · ProcessLookupError) 만 "사라짐" 이고
    권한 · I/O · 파싱 · 목록 오류는 오류로 남긴다 (관측 불능을 "구성원 없음" 으로 바꾸지 않는다)."""
    read = read or (lambda p: p.read_text())
    out = {"members": [], "vanished": 0, "errors": []}
    try:
        entries = [d for d in Path(proc).iterdir() if d.name.isdigit()]
    except OSError as e:
        out["errors"].append({"where": "list", "error": f"{e.__class__.__name__}: {e}"})
        return out
    for d in entries:
        try:
            s = read(d / "stat")
        except (FileNotFoundError, ProcessLookupError):
            out["vanished"] += 1
            continue
        except OSError as e:
            out["errors"].append({"pid": d.name, "error": f"{e.__class__.__name__}: {e}"})
            continue
        try:
            rest = s[s.rindex(")") + 2:].split()
            state, pgrp = rest[0], int(rest[2])
        except (ValueError, IndexError) as e:
            out["errors"].append({"pid": d.name, "error": f"파싱 실패: {e}"})
            continue
        if pgrp == pgid:
            out["members"].append((int(d.name), state))
    return out


def group_status(pgid, *, grace=GROUP_GRACE_S, proc=PROC, read=None) -> dict:
    """그룹 상태 = empty (좀비 아닌 구성원 0 · 오류 0) / alive (grace 초 뒤에도 살아 있음) / unknown (관측 오류 — 뒤의 관측으로 지우지 않는다)."""
    t, n = time.monotonic(), 0
    while True:
        o = proc_members(pgid, proc=proc, read=read)
        n += 1
        alive = [m for m in o["members"] if m[1] != "Z"]
        if o["errors"]:
            return {"status": "unknown", "alive": alive, "errors": o["errors"], "observations": n}
        if not alive:
            return {"status": "empty", "alive": [], "zombies": [m for m in o["members"] if m[1] == "Z"], "errors": [], "observations": n}
        if time.monotonic() - t > grace:
            return {"status": "alive", "alive": alive, "errors": [], "observations": n}
        time.sleep(0.1)


def kill_group(pgid, killpg=os.killpg) -> dict:
    """그룹 SIGKILL 요청 — 결과를 버리지 않고 돌려준다 (보냄 · 이미 없음 · 권한 · 그 밖)."""
    try:
        killpg(pgid, signal.SIGKILL)
        return {"utc": utc(), "result": "sent"}
    except ProcessLookupError:
        return {"utc": utc(), "result": "no_process"}
    except PermissionError as e:
        return {"utc": utc(), "result": "permission", "error": str(e)}
    except OSError as e:
        return {"utc": utc(), "result": "error", "error": f"{e.__class__.__name__}: {e}"}


def run_child(argv, *, cwd, env, stdout, stderr, timeout, popen=subprocess.Popen, killpg=os.killpg, kill_wait=KILL_WAIT_S,
              grace=GROUP_GRACE_S, proc=PROC, read=None) -> dict:
    """새 세션에서 자식을 띄우고 기다린다 (N3). 시작 실패 · timeout · kill 결과 · 제한된 종료 확인 · 그룹 상태를 모두 기록하고, 어느 대기도 무기한이 아니다."""
    rec = {"started": False, "rc": None, "timed_out": None, "errors": []}
    t = time.monotonic()
    try:
        p = popen(argv, cwd=str(cwd), env=env, stdout=stdout, stderr=stderr, start_new_session=True)
    except Exception as e:  # noqa: BLE001 — 시작 실패는 기록하고 돌아간다
        rec["start_error"] = f"{e.__class__.__name__}: {e}"
        return rec
    rec.update(started=True, pgid=p.pid, start_utc=utc())
    try:
        try:
            rec["rc"], rec["timed_out"] = p.wait(timeout=timeout), False
        except subprocess.TimeoutExpired:
            rec["timed_out"] = True
            rec["kill"] = [kill_group(p.pid, killpg)]
            try:
                rec["rc"], rec["exited_after_kill"] = p.wait(timeout=kill_wait), True
            except subprocess.TimeoutExpired:
                rec["exited_after_kill"] = False
        rec["elapsed_s"] = round(time.monotonic() - t, 3)
        rec["group"] = group_status(p.pid, grace=grace, proc=proc, read=read)
        if rec["group"]["status"] == "alive":
            rec.setdefault("kill", []).append(kill_group(p.pid, killpg))
            rec["group_first"] = rec["group"]
            rec["group"] = group_status(p.pid, grace=grace, proc=proc, read=read)
            if rec["group"]["status"] == "empty":
                rec["group"] = dict(rec["group"], status="alive_then_killed")
    except Exception as e:  # noqa: BLE001 — 대기 · 정리 중 오류도 기록 (판정은 false)
        rec["errors"].append(f"{e.__class__.__name__}: {e}")
    return rec


def git_out(repo, *args) -> dict:
    """기록용 git (HEAD · porcelain) — rc · 오류를 그대로 남긴다."""
    return git_run(["-C", repo, *args])


# ---------------------------------------------------------------- 단계 하나
def launch(step, evidence, section, *, repo=REPO, py=PY, venv=VENV, seal=SEAL, work=WORK, data=DATA, data_files=DATA_FILES,
           sources=SOURCES, sources_sha256=SOURCES_SHA256, seal_manifest_sha256=SEAL_MANIFEST_SHA256, lock_sha256=C6_LOCK_SHA256,
           extra=EXTRA_EXECUTED, helper_sha256=IDENTITY_HELPER_SHA256, timeouts=TIMEOUTS, env_base=None, popen=subprocess.Popen,
           killpg=os.killpg, kill_wait=KILL_WAIT_S, grace=GROUP_GRACE_S, proc=PROC, read=None) -> int:
    """사전 대조 → (통과할 때만) 자식 → 사후 대조 → 단계 판정 → 기록. 반환 0 = 단계 판정 complete · 1 = 아님 · 4 = meta 를 쓰지 못함."""
    if step not in STEPS:
        raise SystemExit(f"단계는 {sorted(STEPS)} 뿐이다: {step!r}")
    ev, name, bms = Path(evidence), STEPS[step], Path(repo) / "bms-balancing"
    files = {k: ev / f"{name}.{k}" for k in ("precheck.json", "meta.json", "stdout.log", "stderr.log", "porcelain_start.txt",
                                             "porcelain_end.txt")}
    taken = [str(p) for p in files.values() if p.exists()]
    if taken:
        raise SystemExit(f"이미 있다 — 덮어쓰지 않는다: {taken}")
    prefix = Path(work) / "pycache" / name
    if prefix.exists():
        raise SystemExit(f"pycache 폴더 {prefix} 가 이미 있다 — 새 빈 폴더만")
    prefix.mkdir(parents=True)
    for k in ("mpl", "tmp"):
        (Path(work) / k).mkdir(parents=True, exist_ok=True)
    env, removed = child_env(dict(os.environ if env_base is None else env_base), work)
    argv = build_argv(step, ev, section, prefix, py=py, bms=bms, venv=venv, seal=seal, data=data, sources=sources,
                      sources_sha256=sources_sha256, seal_manifest_sha256=seal_manifest_sha256)
    me = Path(__file__).resolve()
    meta = {"step": step, "name": name, "argv": argv, "cwd": str(bms), "timeout_s": timeouts[step], "kill_wait_s": kill_wait,
            "group_grace_s": grace, "pycache_prefix": str(prefix), "env_removed_names": removed,
            "env_set": {k: env[k] for k in ENV_SET_KEYS}, "request": REQUEST, "approval_section": section,
            "launcher": {"path": str(me), "sha256": sha256(me.read_bytes())}, "start_utc": utc(),
            "disk_free_start_bytes": shutil.disk_usage(bms).free}
    errors, records, phase = [], {}, "precheck"
    pre = child = post = detail = None
    ok = False
    try:
        try:
            pre = precheck(step, ev, argv, prefix, env=env, repo=repo, py=py, seal=seal, work=work, data=data, data_files=data_files,
                           sources=sources, sources_sha256=sources_sha256, seal_manifest_sha256=seal_manifest_sha256,
                           lock_sha256=lock_sha256, extra=extra, helper_sha256=helper_sha256)
        except Exception as e:  # noqa: BLE001
            errors.append({"phase": "precheck", "type": e.__class__.__name__, "message": str(e)[:500]})
            pre = {"ok": False, "problems": [f"사전 대조 예외 {e.__class__.__name__}: {e}"]}
        phase = "record"
        write_new(files["precheck.json"], dump(pre))
        if pre.get("ok") is True and not errors:
            records["head_start"] = git_out(repo, "rev-parse", "HEAD")
            records["porcelain_start"] = git_out(repo, "status", "--porcelain=v1", "--untracked-files=all")
            write_new(files["porcelain_start.txt"], records["porcelain_start"]["out"])
            phase = "start"
            with open(files["stdout.log"], "xb") as fo, open(files["stderr.log"], "xb") as fe:
                child = run_child(argv, cwd=bms, env=env, stdout=fo, stderr=fe, timeout=timeouts[step], popen=popen, killpg=killpg,
                                  kill_wait=kill_wait, grace=grace, proc=proc, read=read)
            if not child.get("started"):
                errors.append({"phase": "start", "type": "StartError", "message": str(child.get("start_error"))[:500]})
            errors += [{"phase": "wait", "type": "ChildError", "message": str(msg)[:500]} for msg in child.get("errors") or []]
            phase = "record"
            records["porcelain_end"] = git_out(repo, "status", "--porcelain=v1", "--untracked-files=all")
            write_new(files["porcelain_end.txt"], records["porcelain_end"]["out"])
            records["head_end"] = git_out(repo, "rev-parse", "HEAD")
            if child.get("started"):
                ok, detail = child_result(step, files["stdout.log"].read_text(encoding="utf-8", errors="replace"))
            phase = "postcheck"
            post = postcheck(step, ev, pre, prefix, child, env=env, repo=repo, data=data, data_files=data_files, extra=extra,
                             records=records)
    except Exception as e:  # noqa: BLE001 — 어느 단계의 예외든 기록하고 판정을 false 로 둔다
        errors.append({"phase": phase, "type": e.__class__.__name__, "message": str(e)[:500]})
    launched = bool(child and child.get("started"))
    stage = stage_verdict(precheck_ok=bool(pre and pre.get("ok") is True), child_ok=ok, rc=(child or {}).get("rc"),
                          timed_out=(child or {}).get("timed_out"), postcheck_ok=bool(post and post.get("ok") is True) and not errors)
    meta.update(launched=launched, child=child, child_result=detail, postcheck=post, stage=stage, errors=errors,
                first_error=errors[0] if errors else None, precheck_ok=bool(pre and pre.get("ok") is True),
                precheck_problems=(pre or {}).get("problems"),
                records={k: {kk: vv for kk, vv in v.items() if kk != "out"} | ({"out": v.get("out", "").strip()} if k.startswith("head") else {})
                         for k, v in records.items()},
                end_utc=utc(), child_maxrss_kib=resource.getrusage(resource.RUSAGE_CHILDREN).ru_maxrss)
    try:
        meta["disk_free_end_bytes"] = shutil.disk_usage(bms).free
        meta["evidence_bytes_before_meta"] = sum(p.stat().st_size for p in ev.rglob("*") if p.is_file())
        write_new(files["meta.json"], dump(meta))
    except Exception as e:  # noqa: BLE001 — 기록 자체가 불가능하면 외부 stderr · rc 4 로 한계를 분리한다
        sys.stderr.write(json.dumps({"step": step, "meta_write_failed": f"{e.__class__.__name__}: {e}", "launched": launched,
                                     "first_error": errors[0] if errors else None, "complete": False}, ensure_ascii=False) + "\n")
        return 4
    print(json.dumps({"step": step, "launched": launched, "complete": stage["complete"], "rc": (child or {}).get("rc"),
                      "timed_out": (child or {}).get("timed_out"), "errors": len(errors)}, ensure_ascii=False))
    return 0 if stage["complete"] else 1


def main(argv=None) -> int:
    ap = argparse.ArgumentParser(prog="reil_c4e1_launch", description=(__doc__ or "").splitlines()[0])
    ap.add_argument("step", choices=sorted(STEPS))
    ap.add_argument("--evidence", required=True, type=Path)
    ap.add_argument("--approval-section", required=True)
    a = ap.parse_args(argv)
    if not SECTION.match(a.approval_section):
        raise SystemExit(f"승인 절은 §NN 형식이어야 한다: {a.approval_section!r}")
    ev = a.evidence
    if ev.parent != BMS / "evidence" or not EVIDENCE_NAME.match(ev.name) or not ev.is_dir():
        raise SystemExit(f"증거 폴더는 {BMS / 'evidence'}/reil_c4e1_YYYYMMDD (미리 만든 폴더) 여야 한다: {ev}")
    return launch(a.step, ev, a.approval_section)


if __name__ == "__main__":
    sys.exit(main())

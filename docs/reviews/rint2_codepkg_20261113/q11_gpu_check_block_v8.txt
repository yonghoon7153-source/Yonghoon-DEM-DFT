"""Q11 GPU check (no kit, no sigma) - harness v8 2026-11-13 (Rint 2, preconditioner representation only).

Runs the PRODUCTION function scripts/step3_sigma.py _solve_cg from two commits on the same synthetic SPD system:
    old = commit 11a794674  (A form: M = cupyx.scipy.sparse.diags(1/diag), a dia_matrix)

    every pair below is bit-identical (solution bytes, info, iteration count, LAST_BACKEND), and the new form is faster per
    iteration (synthetic bench 2026-11-12: A / C = 3.40 gabia, 3.79 kgy).  Anything else = report, do not "fix" the check.

    control  old vs old and new vs new, fixed 200 iterations (run-to-run determinism of this machine)
    short    old vs new, fixed 200 iterations
    main     old vs new, from zero to convergence at the production stop rule (rtol 1e-12, callback on = CONV_CERT)
    resolve  old vs new, from the SAME start (the old main solution) at the certificate re-solve rtol (1e-14) - the x0 path
Consumption rule (v8, fixed before any result - Codex rebuttal 1/2 Q11): result.json "consume" separates
    identical  (the five pairs bit-identical)      from      converged  (main and resolve runs, old and new: info 0, finite)
    and adds backend_ok (every run used the expected backend + jacobi, no fallback) and short_iters_ok (the six fixed-iteration
    runs ran exactly --short-iters iterations, info = --short-iters).  adopt = backend cupy AND all four.  Equality is not
    convergence: all_identical true alone is never adoption.  The process rc only says the harness ran to the end.
Module source (v8, SELF-181): se_material is imported only from --scripts (the block writes it there from the pin) - its sha256
    is checked, its loaded path is checked, and any project module loaded from --forbid-dir (the live repository) stops the run.

Python 3.8 compatible.  --backend numpy = small CPU flow check (scipy path of the same function), not the GPU check.

import hashlib
import importlib.util
import json
import os
import sys

import numpy as np
from scipy import sparse

SCHEMA = 'q11-gpu-check-v2'
CONSUME_RULE = ('adopt only if: backend cupy; the five pairs bit-identical (solution bytes, info, iterations, LAST_BACKEND); '
                'main and resolve runs (old and new) info 0 with finite solutions; every run used gpu + jacobi with no fallback; '
                'the six fixed-iteration runs ran exactly short_iters iterations (info = short_iters). '
                'Equality is not convergence: all_identical true alone is not adoption. The process rc is not the verdict.')
PAIR_KEYS = ('control_old_old', 'control_new_new', 'short_old_new', 'main_old_new', 'resolve_old_new')


def log(msg, fh):
    line = '[q11] ' + msg
    print(line, flush=True)
    fh.write(line + '\n')
    fh.flush()


def sha_file(p):
    h = hashlib.sha256()
    with open(p, 'rb') as f:
        h.update(f.read())
    return h.hexdigest()


def build_system(n_side, seed, logc):
    """Synthetic SPD graph Laplacian on an n_side^3 grid + plate couplings (the 2026-11-11 bench construction; edge
    conductances log-uniform 10^-logc .. 10^logc).  -> scipy CSR L (float64, canonical), rhs b."""
    nx = ny = nz = int(n_side)
    n = nx * ny * nz
    rng = np.random.default_rng(seed)

    def cond(k):
        return np.power(10.0, rng.uniform(-logc, logc, size=k))

    gz = cond(n - 1)
    gz[(np.arange(n - 1, dtype=np.int64) % nz) == nz - 1] = 0.0
    gy = cond(n - nz)
    gy[((np.arange(n - nz, dtype=np.int64) // nz) % ny) == ny - 1] = 0.0
    gx = cond(n - ny * nz)
    diag = np.zeros(n, np.float64)
    diag[:-1] += gz
    diag[1:] += gz
    diag[:n - nz] += gy
    diag[nz:] += gy
    diag[:n - ny * nz] += gx
    diag[ny * nz:] += gx
    bot = np.arange(0, n, nz, dtype=np.int64)
    top = bot + (nz - 1)
    gb, gt = cond(nx * ny), cond(nx * ny)
    diag[bot] += gb
    diag[top] += gt
    b = np.zeros(n, np.float64)
    b[bot] = gb
    L = sparse.diags([diag, -gz, -gz, -gy, -gy, -gx, -gx], [0, 1, -1, nz, -nz, ny * nz, -ny * nz],
                     shape=(n, n), format='csr', dtype=np.float64)
    L.sum_duplicates()
    return L, b


def load(path, name):
    spec = importlib.util.spec_from_file_location(name, path)
    m = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(m)
    return m


def configure(m, backend, maxiter):
    m.GPU_SOLVE = (backend == 'cupy')
    m.REQUIRE_GPU = (backend == 'cupy')
    m.CONV_CERT = True
    m.AMG_SOLVE = False
    m.CG_MAXITER = int(maxiter)


def make_sync(backend):
    if backend == 'cupy':
        import cupy as cp

        def sync():
            cp.cuda.Device(0).synchronize()
        return sync
    return lambda: None


def run(m, L, b, x0, rtol, maxiter, sync):
    m.CG_MAXITER = int(maxiter)
    sync()
    t0 = time.perf_counter()
    x, info = m._solve_cg(L, b, x0=x0, rtol=rtol)
    sync()
    dt = time.perf_counter() - t0
    it = m.LAST_CG.get('iters')
    x = np.asarray(x)
    return {'x': x, 'info': int(info), 'iters': (None if it is None else int(it)), 'seconds': round(dt, 3),
            'per_iter_s': (round(dt / it, 6) if it else None), 'backend': dict(m.LAST_BACKEND),
            'finite': bool(np.isfinite(x).all()), 'x_sha256': hashlib.sha256(np.ascontiguousarray(x).tobytes()).hexdigest()}


def compare(a, b):
    xa, xb = a['x'], b['x']
    same_bytes = xa.dtype == xb.dtype and xa.shape == xb.shape and xa.tobytes() == xb.tobytes()
    diff = np.abs(xa - xb) if xa.shape == xb.shape else None
    n_val = None if diff is None else int(np.count_nonzero(np.logical_not(xa == xb)))
    return {'bytes_identical': bool(same_bytes), 'value_differ_count': n_val,
            'max_abs_diff': (None if diff is None or diff.size == 0 else float(diff.max())),
            'info_same': a['info'] == b['info'], 'iters_same': a['iters'] == b['iters'], 'backend_same': a['backend'] == b['backend']}


def pub(r):
    return {k: v for k, v in r.items() if not k == 'x'}


def pair_ok(p):
    return bool(p.get('bytes_identical') and p.get('info_same') and p.get('iters_same') and p.get('backend_same'))


def consume(backend, short_iters, pairs, short_runs, solve_runs, source_ok=True):
    """The adoption verdict (rule fixed before any result).  pairs = the five compare() dicts; short_runs = the six fixed-iteration
    run records; solve_runs = main old, main new, resolve old, resolve new; source_ok = no project module from the live repository
    by the end of the run.  -> dict (identical, converged, backend_ok, short_iters_ok, source_ok, adopt, why, rule)."""
    why = []
    identical = len(pairs) == 5 and all(pair_ok(p) for p in pairs)
    if not identical:
        why.append('identical false: the five pairs are not all bit-identical (solution bytes, info, iterations, LAST_BACKEND)')
    converged = len(solve_runs) == 4 and all(r.get('info') == 0 and r.get('finite') is True for r in solve_runs)
    if not converged:
        why.append('converged false: main/resolve runs are not all info 0 with finite solutions (equality is not convergence)')
    want = 'gpu' if backend == 'cupy' else 'cpu'
    runs = list(short_runs) + list(solve_runs)
    backend_ok = len(runs) == 10 and all((r.get('backend') or {}).get('used') == want and (r.get('backend') or {}).get('precond') == 'jacobi'
                                         and (r.get('backend') or {}).get('fallback_reason') is None for r in runs)
    if not backend_ok:
        why.append('backend_ok false: a run did not use ' + want + ' + jacobi without fallback')
    short_iters_ok = len(short_runs) == 6 and all(r.get('iters') == short_iters and r.get('info') == short_iters for r in short_runs)
    if not short_iters_ok:
        why.append('short_iters_ok false: a fixed-iteration run did not run exactly ' + str(short_iters) + ' iterations')
    if not source_ok:
        why.append('source_ok false: a project module was loaded from the live repository (--forbid-dir) during the run')
    if not backend == 'cupy':
        why.append('backend ' + str(backend) + ': small CPU flow check, never the GPU check')
    adopt = bool(backend == 'cupy' and identical and converged and backend_ok and short_iters_ok and source_ok)
    return {'identical': bool(identical), 'converged': bool(converged), 'backend_ok': bool(backend_ok),
            'short_iters_ok': bool(short_iters_ok), 'source_ok': bool(source_ok), 'adopt': adopt, 'why': why, 'rule': CONSUME_RULE}


def project_modules_under(root):
    """Loaded modules whose file lies under root (the live repository) - names sorted."""
    if not root:
        return []
    base = os.path.realpath(root) + os.sep
    out = []
    for name, m in list(sys.modules.items()):
        f = getattr(m, '__file__', None)
        if f and os.path.realpath(f).startswith(base):
            out.append(name)
    return sorted(out)


def main(argv=None):
    ap = argparse.ArgumentParser()
    ap.add_argument('--old', required=True)
    ap.add_argument('--new', required=True)
    ap.add_argument('--old-sha', default='')
    ap.add_argument('--new-sha', default='')
    ap.add_argument('--se-sha', default='', help='sha256 of se_material.py in --scripts (the pin copy)')
    ap.add_argument('--scripts', required=True, help='folder the two modules import se_material from (the block writes the pin copy)')
    ap.add_argument('--forbid-dir', default='', help='live repository: a project module loaded from here stops the run (SELF-181)')
    ap.add_argument('--out', required=True)
    ap.add_argument('--label', default='x')
    ap.add_argument('--backend', choices=('cupy', 'numpy'), default='cupy')
    ap.add_argument('--n-side', type=int, default=298)
    ap.add_argument('--logc', type=float, default=1.0)
    ap.add_argument('--seed', type=int, default=20261112)
    ap.add_argument('--rtol', type=float, default=1e-12)
    ap.add_argument('--rtol-resolve', type=float, default=1e-14)
    ap.add_argument('--maxiter', type=int, default=60000)
    ap.add_argument('--short-iters', type=int, default=200)
    a = ap.parse_args(argv)
    os.makedirs(a.out, exist_ok=True)
    fh = open(os.path.join(a.out, 'check.log'), 'w', encoding='utf-8')
    res = {'schema': SCHEMA, 'label': a.label, 'backend': a.backend,
           'args': {k: v for k, v in vars(a).items() if k not in ('out', 'scripts', 'forbid_dir')},
           'expect': 'all pairs bit-identical; new faster per iteration', 'consume_rule': CONSUME_RULE, 'consume': None, 'error': None}

    def save():
        with open(os.path.join(a.out, 'result.json'), 'w', encoding='utf-8') as f:
            json.dump(res, f, indent=1, sort_keys=True)
    try:
        so, sn = sha_file(a.old), sha_file(a.new)
        sep = os.path.join(a.scripts, 'se_material.py')
        ss = sha_file(sep) if os.path.isfile(sep) else None
        res['files'] = {'old_sha256': so, 'new_sha256': sn, 'se_material_sha256': ss}
        if (a.old_sha and not so == a.old_sha) or (a.new_sha and not sn == a.new_sha):
            raise SystemExit('module file sha256 differs from the block (wrong commit copied) - nothing run')
        if a.se_sha and not ss == a.se_sha:
            raise SystemExit('se_material.py in --scripts missing or sha256 differs from the block - nothing run')
        if a.scripts not in sys.path:
            sys.path.insert(0, a.scripts)
        old, new = load(a.old, 's3_old_q11'), load(a.new, 's3_new_q11')
        semf = getattr(sys.modules.get('se_material'), '__file__', None)
        res['files']['se_material_from_scripts'] = bool(semf and os.path.realpath(semf) == os.path.realpath(sep))
        if not res['files']['se_material_from_scripts']:
            raise SystemExit('se_material was not imported from --scripts - nothing run')
        bad = project_modules_under(a.forbid_dir)
        if bad:
            res['files']['loaded_from_forbid_dir'] = bad
            raise SystemExit('project module(s) loaded from the live repository (--forbid-dir): ' + ', '.join(bad) + ' - nothing run')
        for m in (old, new):
            configure(m, a.backend, a.maxiter)
        sync = make_sync(a.backend)
        if a.backend == 'cupy':
            import cupy as cp
            res['env'] = {'python': sys.version.split()[0], 'numpy': np.__version__, 'cupy': cp.__version__,
                          'gpu': cp.cuda.runtime.getDeviceProperties(0)['name'].decode('ascii', 'replace')}
        else:
            import scipy
            res['env'] = {'python': sys.version.split()[0], 'numpy': np.__version__, 'scipy': scipy.__version__}
        t0 = time.perf_counter()
        L, b = build_system(a.n_side, a.seed, a.logc)
        log(f'built: n {L.shape[0]}, nnz {L.nnz} in {time.perf_counter() - t0:.1f} s (logc {a.logc})', fh)
        res['system'] = {'n': int(L.shape[0]), 'nnz': int(L.nnz)}
        tiny = 1e-300                                         # fixed-iteration runs: a stop rule nothing reaches
        oo1 = run(old, L, b, None, tiny, a.short_iters, sync)
        oo2 = run(old, L, b, None, tiny, a.short_iters, sync)
        nn1 = run(new, L, b, None, tiny, a.short_iters, sync)
        nn2 = run(new, L, b, None, tiny, a.short_iters, sync)
        so_ = run(old, L, b, None, tiny, a.short_iters, sync)
        sn_ = run(new, L, b, None, tiny, a.short_iters, sync)
        short_runs = [oo1, oo2, nn1, nn2, so_, sn_]
        res['short_runs'] = {k: pub(r) for k, r in zip(('old_1', 'old_2', 'new_1', 'new_2', 'old_3', 'new_3'), short_runs)}
        c_oo, c_nn, c_on = compare(oo1, oo2), compare(nn1, nn2), compare(so_, sn_)
        res['control_old_old'], res['control_new_new'], res['short_old_new'] = c_oo, c_nn, c_on
        log(f'control old vs old ({a.short_iters} it): {c_oo}', fh)
        log(f'control new vs new ({a.short_iters} it): {c_nn}', fh)
        log(f'short old vs new ({a.short_iters} it): {c_on} - per iteration old {so_["per_iter_s"]} s, new {sn_["per_iter_s"]} s', fh)
        save()
        mo = run(old, L, b, None, a.rtol, a.maxiter, sync)
        log(f'main old: info {mo["info"]}, iterations {mo["iters"]}, {mo["seconds"]} s ({mo["per_iter_s"]} s per iteration)', fh)
        res['main_old'] = pub(mo)
        save()
        mn = run(new, L, b, None, a.rtol, a.maxiter, sync)
        log(f'main new: info {mn["info"]}, iterations {mn["iters"]}, {mn["seconds"]} s ({mn["per_iter_s"]} s per iteration)', fh)
        res['main_new'] = pub(mn)
        res['main_old_new'] = compare(mo, mn)
        log(f'main old vs new: {res["main_old_new"]}', fh)
        save()
        x0 = mo['x']
        ro = run(old, L, b, x0, a.rtol_resolve, a.maxiter, sync)
        rn = run(new, L, b, x0, a.rtol_resolve, a.maxiter, sync)
        res['resolve_old'], res['resolve_new'] = pub(ro), pub(rn)
        res['resolve_old_new'] = compare(ro, rn)
        log(f'resolve (x0 = old main solution, rtol {a.rtol_resolve}): old info {ro["info"]} it {ro["iters"]}, new info {rn["info"]} '
            f'it {rn["iters"]} - {res["resolve_old_new"]}', fh)
        pairs = [res[k] for k in PAIR_KEYS]
        res['all_identical'] = all(pair_ok(p) for p in pairs)
        res['speed_ratio_main'] = (round(mo['per_iter_s'] / mn['per_iter_s'], 3) if mo['per_iter_s'] and mn['per_iter_s'] else None)
        bad_end = project_modules_under(a.forbid_dir)
        res['files']['loaded_from_forbid_dir_end'] = bad_end
        res['consume'] = consume(a.backend, a.short_iters, pairs, short_runs, [mo, mn, ro, rn], source_ok=not bad_end)
        log(f'DONE label {a.label}: all pairs identical = {res["all_identical"]}, main per-iteration ratio old / new = '
            f'{res["speed_ratio_main"]}', fh)
        cs = res['consume']
        log(f'CONSUME adopt = {cs["adopt"]} (identical {cs["identical"]}, converged {cs["converged"]}, backend_ok {cs["backend_ok"]}, '
            f'short_iters_ok {cs["short_iters_ok"]}, source_ok {cs["source_ok"]}) - why: {cs["why"]}', fh)
        save()
        return 0
    except SystemExit as e:
        res['error'] = str(e)
        log('STOP: ' + str(e), fh)
        save()
        return 2
    except Exception as e:                                   # noqa: BLE001 - record and stop
        res['error'] = f'{type(e).__name__}: {e}'[:400]
        log('ERROR ' + res['error'], fh)
        save()
        return 1


if __name__ == '__main__':
    sys.exit(main())

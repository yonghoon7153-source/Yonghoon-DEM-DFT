192: def cost_projection(snap, plan, *, min_samples=COST_MIN_SAMPLES, measure=LIMITS["measure"]) -> dict:
193:     """비용 투영 (v3 §4-3) — 완료 returned 의 c = wall / n_eval · 위쪽 중앙 순서통계 m (오름차순 0-기반 ⌊n/2⌋) · 최소 표본 · 누적 창 ·
194:     작업자 w 마다 R_w = (미착수 배정 호출의 기대 평가 합 + 진행 중 호출의 max(기대 평가 − progress, 0)) × m · now_rel + max R_w > 경계 → 중지."""
195:     by_id = plan if isinstance(plan, dict) else {t["task_id"]: t for t in plan}
196:     done, started, prog = snap["done"], snap["started"], snap["progress"]
197:     ret = sorted((tid for tid in done if done[tid]["termination"] == "returned"), key=_tid_key)
198:     samples = [{"task_id": tid, "wall": done[tid]["wall"], "n_eval": done[tid]["n_eval"],
199:                 "c": done[tid]["wall"] / done[tid]["n_eval"]} for tid in ret]
200:     rec = {"now_rel": snap["now_rel"], "returned_ids": ret, "n_samples": len(samples), "samples": samples,
201:            "min_samples": int(min_samples), "boundary": float(measure), "m": None, "judged": False,
202:            "remaining_expected_evals": None, "R": None, "value": None, "stop": False, "reason": None}
203:     if len(samples) < min_samples:
204:         return rec
205:     cs = sorted(s["c"] for s in samples)
206:     m = cs[len(cs) // 2]
207:     rem = {}
208:     for tid in snap["planned"]:
209:         t = by_id[tid]
210:         w, exp = t["worker"], t["algorithm"]["expected_evals"]
211:         rem.setdefault(w, 0)
212:         if tid not in started:
213:             rem[w] += exp
214:         elif tid not in done:
215:             rem[w] += max(exp - prog[tid], 0)
216:     r_w = {w: v * m for w, v in rem.items()}
217:     value = snap["now_rel"] + (max(r_w.values()) if r_w else 0.0)
218:     stop = value > measure
219:     rec.update(m=m, judged=True, remaining_expected_evals={str(w): rem[w] for w in sorted(rem)},
220:                R={str(w): r_w[w] for w in sorted(r_w)}, value=value, stop=stop,
221:                reason=(f"{COST_REASON} (now_rel {snap['now_rel']:.3f} + max R_w {value - snap['now_rel']:.3f} = {value:.3f} s > "
222:                        f"{measure} s · m {m:.6g} s/평가 · 표본 {len(samples)})") if stop else None)
223:     return rec

226: class StopRule:
227:     """감시가 tick 마다 부르는 규칙 (v3 §4-2 · §4-3): 입력 검증 → 호출당 상한 → 비용 투영. 상태는 바꾸지 않고 사유와 기록만 남긴다.
228:     진행 중 초과는 '중단 요청 대상' 으로 적고, 최종 상태는 감시의 마지막 drain 뒤 사후 집계 (`aggregate`) 가 붙인다."""
229: 
230:     def __init__(self, plan, *, cap=CALL_CAP_S, min_samples=COST_MIN_SAMPLES, measure=LIMITS["measure"]):
231:         self.plan = {t["task_id"]: t for t in plan}
232:         self.cap, self.min_samples, self.measure = float(cap), int(min_samples), float(measure)
233:         self.active_over: list = []
234:         self.late_seen: list = []
235:         self.stop = None
236:         self.last = None
237:         self.n_calls = 0
238: 
239:     def __call__(self, snap):
240:         self.n_calls += 1
241:         validate(snap, self.plan)
242:         cap = call_cap(snap, self.cap)
243:         if cap["reason"] is not None:
244:             self.active_over += cap["active_over"]
245:             self.late_seen += cap["late"]
246:             self.stop = {"kind": "call_cap", "now_rel": snap["now_rel"], **cap}
247:             return cap["reason"]
248:         cost = cost_projection(snap, self.plan, min_samples=self.min_samples, measure=self.measure)
249:         self.last = cost
250:         if cost["stop"]:
251:             self.stop = {"kind": "cost", **cost}
252:             return cost["reason"]
253:         return None
254: 
255:     def params(self) -> dict:
256:         return {"cap_s": self.cap, "compare": "엄격한 > (같으면 넘지 않음)", "min_samples": self.min_samples, "measure_s": self.measure,
257:                 "statistic": "위쪽 중앙 순서통계 — returned 의 c = wall / n_eval 오름차순 0-기반 index ⌊n/2⌋",

676: def fetch(dest, *, repo_url=SOURCE_URL, commit=SOURCE_COMMIT, files=None, allow_protocol="https", run=subprocess.run,
677:           timeout=FETCH_TIMEOUT_S) -> dict:
678:     """저장소 밖 새 폴더에 고정 커밋의 두 경로만 받는다 — sha · bytes 대조 · 기록 (파일 바이트 없음). GIT_ALLOW_PROTOCOL 로 프로토콜을 하나로 묶는다."""
679:     files = dict(FETCH_FILES if files is None else files)
680:     extra = sorted(set(files) - set(FETCH_FILES))
681:     if extra or not files:
682:         raise E3aStop(f"fetch 는 두 경로 ({sorted(FETCH_FILES)}) 만 — 요청 밖 {extra}")
683:     dest = Path(dest).resolve()
684:     if dest.exists():
685:         raise E3aStop(f"fetch 목적지가 이미 있다: {dest} — 새 폴더만")
686:     if mon.owned(dest, [REPO.resolve()]):
687:         raise E3aStop(f"fetch 목적지가 저장소 안이다: {dest}")
688:     dest.mkdir(parents=True)
689:     gd = dest / "git"
690:     env = dict(os.environ, GIT_ALLOW_PROTOCOL=allow_protocol, GIT_TERMINAL_PROMPT="0")
691:     rec = {"repo_url": str(repo_url), "commit": str(commit), "dest": str(dest), "cwd": str(dest), "allow_protocol": allow_protocol,
692:            "start_utc": _utc(), "steps": [], "files": {}}
693:     t = time.monotonic()
694: 
695:     def step(argv, stdout):
696:         s = {"argv": list(argv), "start_utc": _utc()}
697:         t1 = time.monotonic()
698:         r = run(argv, cwd=str(dest), env=env, stdout=stdout, stderr=subprocess.PIPE, timeout=timeout)
699:         s.update(rc=r.returncode, elapsed_s=round(time.monotonic() - t1, 3),
700:                  stderr=(r.stderr or b"").decode("utf-8", "replace")[-2000:])
701:         rec["steps"].append(s)
702:         if r.returncode != 0:
703:             raise _stop(f"git 단계 실패 (rc {r.returncode}): {' '.join(argv)} — {s['stderr'][-300:]}", rec)
704: 
705:     for argv in fetch_steps(gd, repo_url, commit):
706:         step(argv, subprocess.PIPE)
707:     for name, want in files.items():
708:         target = dest / name
709:         with open(target, "xb") as fo:
710:             step(show_argv(gd, commit, name), fo)
711:         b = target.read_bytes()
712:         rec["files"][name] = {"path": str(target), "sha256": _sha(b), "bytes": len(b), "expected": dict(want)}
713:         if _sha(b) != want["sha256"] or len(b) != want["bytes"]:
714:             raise _stop(f"fetch 대조 불일치 — {name}: sha256 {_sha(b)} · {len(b)} B ≠ 기대 {want}", rec)
715:     rec.update(end_utc=_utc(), elapsed_s=round(time.monotonic() - t, 3), ok=True)
716:     return rec

721:     ext = res.get("extraction") or []

731:     if not _finite(res.get("q_NE_hc_max_record_only")):
732:         probs.append("NE q_NE_hc_max_record_only 가 없다")
733:     if (res.get("notebook_check") or {}).get("ok") is not True:
734:         probs.append("P0 의 노트북 대조가 ok 가 아니다")
735:     return probs

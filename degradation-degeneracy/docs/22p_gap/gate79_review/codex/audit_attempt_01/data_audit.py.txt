class RemoveLogging(ast.NodeTransformer):
    def visit_Expr(self,n):return None if isinstance(n.value,ast.Constant) and isinstance(n.value.value,str) else self.generic_visit(n)
    def visit_Assign(self,n):
        names={x.id for t in n.targets for x in ast.walk(t) if isinstance(x,ast.Name)}
        return None if names and names<=logging_names else self.generic_visit(n)
    def visit_AugAssign(self,n):return None if isinstance(n.target,ast.Name) and n.target.id in logging_names else self.generic_visit(n)
    def visit_Return(self,n):
        n=self.generic_visit(n)
        if isinstance(n.value,ast.Tuple) and len(n.value.elts)==5:n.value.elts=n.value.elts[:4]
        return n

import hashlib

# Reviewed job-start bytes, fixed at this draft revision; never discover identity from staged code.
REVIEWED_JOB_START_SHA256='5e140635336be247ee3a776be58b2b3eb8679562c02bfc9cb593e27c24276b91'

def canonical_runner(c, approval_path):
    spec=c.get("runner_bindings",{})
    for k in ("python", "job_start", "stdout", "stderr"):
        # Directive values forbid Unicode whitespace and C0/DEL control bytes.
        require(type(spec.get(k)) is str and spec[k].startswith("/") and not any(ch.isspace() or ord(ch)<32 or ord(ch)==127 for ch in spec[k]), "runner binding "+k)
    require(re.fullmatch(r"[0-9]+-[0-9]{2}:[0-9]{2}:[0-9]{2}",spec.get("walltime","")) is not None,"walltime binding")
    return "\n".join(["#!/usr/bin/env bash","#SBATCH --partition=cpu","#SBATCH --qos=cpu-60","#SBATCH --nodes=1","#SBATCH --ntasks=20","#SBATCH --cpus-per-task=1","#SBATCH --no-requeue","#SBATCH --time="+spec["walltime"],"#SBATCH --output="+spec["stdout"],"#SBATCH --error="+spec["stderr"],"set -euo pipefail","exec "+" ".join(shlex.quote(x) for x in (spec["python"],spec["job_start"],str(Path(approval_path).resolve())))])+"\n"

CASES = {'PRECAL_E0','PRECAL_N0'} | {'PRECAL_R1_B%d_%s'%(b,r) for b in (1,2,3) for r in ('U','C')}
def require(ok, why):
    if not ok: raise ValueError(why)
def digest(p): return hashlib.sha256(p.read_bytes()).hexdigest()
def absent(p): return not os.path.lexists(str(p))
def exclusive_json(p, value):
    fd=os.open(str(p),os.O_WRONLY|os.O_CREAT|os.O_EXCL,0o600)
    with os.fdopen(fd,'w') as out:
        json.dump(value,out);out.write('\n');out.flush();os.fsync(out.fileno())
def scheduler_environment():
    # Reject all current/future option aliases in these namespaces, including
    # inherited allocation variables. --export alone does not control sbatch input.
    require(not any(k.startswith(('SBATCH_','SLURM_','SALLOC_','SRUN_')) for k in os.environ),
            'scheduler option/allocation environment present')
    return {k:os.environ[k] for k in ('PATH','HOME','USER','LOGNAME','LANG','LC_ALL','TMPDIR') if k in os.environ}
def submit_reserved(c, approval_sha, live, env):
    # One reviewed installation is one campaign. All cases share this fixed
    # namespace; per-case approval paths cannot select independent ledgers.
    here=Path(__file__).resolve().parent;root=Path(c['case_dir'])
    require(root==here/'cases'/c['case'],'case outside canonical campaign root')
    policy_ref=c.get('campaign_policy',{});policy_path=here/'campaign_policy.json'
    require(policy_ref.get('path')==str(policy_path) and policy_path.is_file() and not policy_path.is_symlink(),'common campaign policy missing')
    policy_raw=policy_path.read_bytes();policy_sha=hashlib.sha256(policy_raw).hexdigest()
    require(policy_ref.get('sha256')==policy_sha,'campaign policy drift')
    policy=json.loads(policy_raw)
    require(policy.get('schema')=='mixer_first8_campaign/1' and policy.get('campaign_id')=='mixer_precal_first8_20261106'
            and policy.get('execution_authorized') is True,'campaign approval absent')
    require(policy.get('install_root')==str(here),'campaign policy belongs to another installation')
    require(type(policy.get('max_concurrency')) is int and 1<=policy['max_concurrency']<=2
            and c['approved_concurrency']==policy['max_concurrency'],'campaign concurrency differs')
    require(policy.get('np')==20 and policy.get('cpu_cap')==60 and policy.get('author_go_sha256')==c['author_go']['sha256'],'campaign authority/resource mismatch')
    admission=here/'admission'
    if absent(admission):
        try:admission.mkdir(mode=0o700)
        except FileExistsError:pass
    require(admission.is_dir() and not admission.is_symlink(),'admission directory')
    lock=admission/'admission.lock'
    fd=os.open(str(lock),os.O_WRONLY|os.O_CREAT|os.O_EXCL,0o600)
    try:
        with os.fdopen(fd,'w') as f:f.write(approval_sha+'\n');f.flush();os.fsync(f.fileno())
        reserved=0
        allowed={'admission.lock'}|{k+s for k in CASES for s in ('.reservation.json','.submission.json')}
        require(all(p.name in allowed for p in admission.iterdir()),'unknown campaign state')
        for case in sorted(CASES):
            reservation=admission/(case+'.reservation.json');result=admission/(case+'.submission.json')
            if absent(reservation):
                require(absent(result),'orphan campaign result');continue
            require(reservation.is_file() and not reservation.is_symlink(),'reservation type')
            item=json.loads(reservation.read_bytes())
            require(item.get('case')==case and item.get('campaign_policy_sha256')==policy_sha and item.get('np')==20,'reservation/policy mismatch')
            # Missing/pending/uncertain/failed state holds every later case.
            require(result.is_file() and not result.is_symlink(),'unresolved campaign submission')
            outcome=json.loads(result.read_bytes())
            require(outcome.get('state')=='SUBMITTED' and outcome.get('approval_sha256')==item.get('approval_sha256')
                    and outcome.get('case')==case and str(outcome.get('job_id','')).isdigit(),'campaign uncertainty HOLD')
            reserved+=1
        # Deliberate conservative sum: live counts may already include reservations.
        # No automatic completion/release or deduction from a stale snapshot.
        require(reserved+live['running_first8']<policy['max_concurrency'],'campaign slots reserved')
        require(live['reserved_cpus']+20*(reserved+1)<=60,'campaign CPU reservations')
        entry={'case':c['case'],'np':20,'approval_sha256':approval_sha,'campaign_policy_sha256':policy_sha,'state':'ATTEMPT_RESERVED'}
        exclusive_json(admission/(c['case']+'.reservation.json'),entry)
        exclusive_json(root/'submission.claim.json',entry)
        spec=c['runner_bindings']
        argv=['sbatch','--parsable','--no-requeue','--partition=cpu','--qos=cpu-60','--nodes=1',
              '--ntasks=20','--cpus-per-task=1','--time='+spec['walltime'],
              '--output='+spec['stdout'],'--error='+spec['stderr'],'--chdir='+str(root),'--export=ALL',c['runner']['path']]
        try:
            r=subprocess.run(argv,capture_output=True,text=True,timeout=60,env=env)
            ok=r.returncode==0 and re.fullmatch(r'[1-9][0-9]*(;[A-Za-z0-9_.-]+)?\n?',r.stdout) is not None
            receipt={'case':c['case'],'approval_sha256':approval_sha,'returncode':r.returncode,'stdout':r.stdout,'stderr':r.stderr,
                     'state':'SUBMITTED' if ok else 'AMBIGUOUS_OR_FAILED_HOLD','job_id':r.stdout.strip().split(';')[0] if ok else None}
        except (OSError,subprocess.TimeoutExpired) as exc:
            receipt={'case':c['case'],'approval_sha256':approval_sha,'state':'AMBIGUOUS_OR_FAILED_HOLD','error':str(exc),'job_id':None};ok=False
        exclusive_json(root/'submission.receipt.json',receipt)
        exclusive_json(admission/(c['case']+'.submission.json'),receipt)
        require(ok,'submission failed/ambiguous; campaign HOLD, inspect scheduler manually, never retry')
        print(json.dumps(receipt))
    finally:
        # A process crash can leave this lock; no stale-lock deletion or retry path.
        lock.unlink()
def main(path):
    env=scheduler_environment()
    approval_raw=Path(path).read_bytes()
    approval_sha=hashlib.sha256(approval_raw).hexdigest()
    c=json.loads(approval_raw)
    require(c.get('schema')=='mixer_first8_execution_approval/1', 'wrong approval schema')
    require(c.get('execution_authorized') is True and c.get('gate') in ('QUALIFICATION_PILOT_GO','MAP_CASE_GO'), 'no explicit execution approval')
    for k in ('author_go','qualification','mxd2r_disposition','live_preflight','source_manifest','policy','runner','binary','mpi_launcher'):
        a=c.get(k,{})
        p=Path(a.get('path',''))
        require(p.is_absolute() and p.is_file() and not p.is_symlink(), 'missing physical '+k)
        require(re.fullmatch('[0-9a-f]{64}',a.get('sha256','')) is not None and digest(p)==a['sha256'], 'identity mismatch '+k)
    policy=json.loads(Path(c['policy']['path']).read_text())
    require(policy.get('execution_authorized') is True and policy.get('policy_id') and policy.get('allowed_to_submit')==[c.get('case')], 'policy case authorization')
    require(c.get('case') in CASES, 'outside first8')
    require(c.get('author_choices_confirmed') is True, 'choices missing')
    require(type(c.get('np')) is int and c['np']==20, 'NP20 only; changed NP needs revised review')
    live=json.loads(Path(c['live_preflight']['path']).read_text())
    import time
    age=time.time()-live.get('observed_unix',0)
    require(0 <= age <= 300, 'live preflight older than five minutes or future dated')
    for k in ('qos_verified','disk_budget_verified','inode_budget_verified','d07_finished','paths_verified'):
        require(live.get(k) is True, 'missing preflight '+k)
    require(c['gate']!='MAP_CASE_GO' or live.get('qualification_accepted') is True, 'map release requires accepted qualification')
    require(c['gate']!='QUALIFICATION_PILOT_GO' or c['case']=='PRECAL_E0', 'pilot restricted to E0')
    require(live.get('qos')=='cpu-60' and live.get('partition')=='cpu','scheduler scope')
    used=live.get('reserved_cpus')
    require(type(used) is int and used>=0 and used+c['np']<=60, 'aggregate CPU cap')
    require(type(live.get('running_first8')) is int and 0<=live['running_first8']<c.get('approved_concurrency',0)<=2,'concurrency')
    require(c.get('wave_release')==[c['case']], 'one-case explicit wave release required')
    root=Path(c.get('case_dir',''))
    require(root.is_absolute() and root.is_dir() and not root.is_symlink(), 'case directory')
    require(all(absent(root/n) for n in ('log.liggghts','post','restart','job_start.json','job_end.json',
              'simulator.stdout','simulator.stderr','submission.claim.json','submission.receipt.json')), 'not fresh')
    manifest=json.loads(Path(c['source_manifest']['path']).read_text())
    require(type(manifest.get('files')) is list and bool(manifest['files']), 'empty manifest')
    paths=set()
    for a in manifest['files']:
        p=Path(a['path']); require(p.is_absolute() and p.is_file() and not p.is_symlink(),'manifest path')
        require(str(p) not in paths,'duplicate manifest path'); paths.add(str(p))
        require(digest(p)==a['sha256'],'manifest drift')
    assets=json.loads((Path(__file__).resolve().parent/'static_asset_manifest.json').read_text())
    expected={Path(a['path']).name:a['sha256'] for a in assets['files'] if a['path'].startswith('planned_assets/'+c['case']+'/')}
    for n in ('in.mixer','deck_meta.json','Drum.stl','Front.stl','Back.stl'):
        require(str(root/n) in paths and digest(root/n)==expected.get(n), 'unsealed or changed planned input '+n)
    seal=json.loads((Path(__file__).resolve().parent/'source_seal.json').read_text())
    require(c.get('source_pin')==seal['source_pin'],'source pin differs')
    source_root=Path(c.get('source_root',''))
    require(source_root.is_absolute(),'source root missing')
    by_path={a['path']:a['sha256'] for a in manifest['files']}
    for a in seal['files']:
        required=str(source_root/a['path'])
        require(by_path.get(required)==a['sha256'],'missing or different bounded source dependency '+a['path'])
    require(c.get('runner_reviewed') is True and c.get('walltime_approved') is True, 'runner/walltime review missing')
    require(Path(c['runner']['path']).read_text()==canonical_runner(c,path), 'runner differs from fixed single-case command')
    for k in ('python','job_start'):
        require(c['runner_bindings'][k] in paths,'unsealed execution dependency '+k)
    require(by_path.get(c['runner_bindings']['job_start'])==REVIEWED_JOB_START_SHA256, 'job-start differs from reviewed implementation')
    require(all(absent(Path(c['runner_bindings'][k])) for k in ('stdout','stderr')),'existing scheduler output')
    submit_reserved(c,approval_sha,live,env)
if __name__=='__main__':
    try:
        require(len(sys.argv)==2,'one approval path required');main(sys.argv[1])
    except Exception as exc:
        print('HOLD: '+str(exc),file=sys.stderr);sys.exit(2)

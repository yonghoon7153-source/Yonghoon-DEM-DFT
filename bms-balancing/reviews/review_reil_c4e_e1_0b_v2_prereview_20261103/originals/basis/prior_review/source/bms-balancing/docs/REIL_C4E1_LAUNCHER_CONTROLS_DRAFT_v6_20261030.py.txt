def _sha(b: bytes) -> str:
    return hashlib.sha256(b).hexdigest()


def _git_env() -> dict:
    """fixture 가 직접 부르는 git 의 env (launcher 와 독립 · V2-02 · S3) — 상속 `GIT_*` · 비밀 형태 이름을 모두 빼고, 시스템 · 사용자 설정 (조건부 포함 ·
    서명 · 템플릿) 을 읽지 않는다."""
    env = {k: v for k, v in os.environ.items() if not k.startswith("GIT_") and not SECRETISH.search(k)}
    env.update(GIT_TERMINAL_PROMPT="0", GIT_CONFIG_NOSYSTEM="1", GIT_CONFIG_GLOBAL=os.devnull)
    return env


def _git(*args) -> str:
    r = subprocess.run(["git", *args], env=_git_env(), capture_output=True, text=True, timeout=60)
    assert r.returncode == 0, (args, r.stderr[-500:])
    return r.stdout


def _run_keywords(path, func_name) -> dict:
    """정적 (S6 · 대상을 실행하지 않는다) — 모듈 파일의 최상위 함수 `func_name` 안 `subprocess.run(…)` 호출 하나의 상수 키워드 {이름: 값}."""
    tree = ast.parse(Path(path).read_text(encoding="utf-8"))
    fn = next(n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name == func_name)
    calls = [n for n in ast.walk(fn) if isinstance(n, ast.Call) and isinstance(n.func, ast.Attribute) and n.func.attr == "run"
             and isinstance(n.func.value, ast.Name) and n.func.value.id == "subprocess"]
    assert len(calls) == 1, (func_name, len(calls))
    return {k.arg: k.value.value for k in calls[0].keywords if isinstance(k.value, ast.Constant)}


def _write_list(d: Path, files: dict, schema="reil-c4e-sources-v1", raw: bytes | None = None):
    d.mkdir(parents=True)
    b = raw if raw is not None else json.dumps({"schema": schema, "files": files}, sort_keys=True, indent=1).encode()
    (d / "SOURCES.json").write_bytes(b)
    return d / "SOURCES.json", _sha(b)


def _seal(d: Path, env_sha: str, *, lock_env_sha: str | None = None):
    """가짜 C6 봉인 — lock 머리줄에 env_profile sha256 · MANIFEST 가 lock 을 덮는다 → (lock sha, MANIFEST sha)."""
    d.mkdir(parents=True)
    lock = f"# REIL C6 lock — 대조용 가짜 (env_profile.py · 그 파일 sha256 {lock_env_sha or env_sha})\n# 둘째 줄\n".encode()
    (d / "REIL_C6.lock.txt").write_bytes(lock)
    man = json.dumps({"REIL_C6.lock.txt": _sha(lock)}, sort_keys=True).encode()
    (d / "MANIFEST.json").write_bytes(man)
    return _sha(lock), _sha(man)

def _pre(t, step="run", *, sources=None, sha=None, argv=None, env_extra=None, data=None, extra=None, man_sha=None):
    prefix = t.work / "pycache" / f"pre_{step}"
    prefix.mkdir(parents=True, exist_ok=True)
    env, _ = L.child_env({**os.environ, **(env_extra or {})}, t.work)
    src, s, d = sources or t.list, sha or t.lsha, data or t.data
    argv = argv or L.build_argv(step, t.ev, "§99", prefix, py=PY, bms=t.bms, seal=t.seal, data=d, sources=src, sources_sha256=s,
                                seal_manifest_sha256=man_sha or t.man_sha)
    return L.precheck(step, t.ev, argv, prefix, env=env, repo=t.repo, py=PY, seal=t.seal, work=t.work, data=d, data_files=t.data_files,
                      sources=src, sources_sha256=s, seal_manifest_sha256=man_sha or t.man_sha, lock_sha256=t.lock_sha,
                      extra=extra or t.extra)

def _state(pid):
    """/proc/<pid>/stat 의 상태 글자 — 파일 없음 (소멸) 만 None. 그 밖 읽기 오류는 그대로 올린다 (관측 불능을 종료로 보지 않는다)."""
    try:
        s = Path(f"/proc/{pid}/stat").read_text()
    except FileNotFoundError:
        return None
    return s[s.rindex(")") + 2:].split()[0]

def test_l3_changed_same_size_missing_symlinked_module_or_linked_parent_each_block(tmp_path):
    """[RED 실패 — 골격] 모듈 하나 +1 바이트 · 같은 크기 다른 바이트 · 없음 · 같은 바이트의 심볼릭 링크 · 링크된 상위 폴더 → 각각 실패."""

class _Clock:
    """가짜 단조 시계 — sleep 한 만큼 · 관측 비용만큼 나아간다 (실제로 자지 않는다)."""

    def __init__(self, t=1000.0):
        self.t, self.sleeps = t, []

    def __call__(self):
        return self.t

    def sleep(self, s):
        assert s >= 0, s
        self.sleeps.append((self.t, s))
        self.t += s


class _P:
    """가짜 자식 — pid · poll · returncode. exits_at 이 None 이면 끝나지 않는다."""

    def __init__(self, pid, clock, exits_at=None, rc=0):
        self.pid, self.clock, self.exits_at, self.rc, self.returncode = pid, clock, exits_at, rc, None

    def poll(self):
        if self.exits_at is not None and self.clock() >= self.exits_at:
            self.returncode = self.rc
        return self.returncode


def _fake_group(tmp: Path, pgid: int, members, clock, cost: float, starts: list):
    """가짜 /proc (구성원은 하나 — 관측 한 번 = stat 읽기 한 번) · 읽을 때마다 시작 시각을 남기고 cost 초를 쓰는 read."""
    proc = tmp / "proc"
    for pid, state in members:
        (proc / str(pid)).mkdir(parents=True, exist_ok=True)
        (proc / str(pid) / "stat").write_bytes(b"%d (x) %s 1 %d 0 0\n" % (pid, state.encode(), pgid))

    def read(q):
        starts.append(clock())
        clock.t += cost
        return q.read_bytes()

    return proc, read

def test_l32_an_exception_inside_cleanup_is_caught_and_only_an_emergency_kill_and_observation_follow_under_the_same_deadline(tmp_path):
    """[RED 실패 — v5: AttributeError: module 'scripts.reil_c4e1_launch' has no attribute 'cleanup_group'] (launcher 운영 정리 정책) 관측 0.05 s · 기한
    t0+20 · timeout · 끝나지 않는 자식 · 시작 시각이 t0+10 이후인 첫 sleep 이 RuntimeError (주입) → 정리가 잡는다 · unknown (그룹 기록도) · emergency ·
    errors = ["RuntimeError: sleep failed (시험 주입)"] · 예외 시각 ∈ [t0+10, t0+10.2) · 그룹 SIGKILL 두 번 (처음 · 비상) · 비상 관측도 같은 기한 (모든
    관측 시작 < t0+20) · 새 기한 없음 (끝 ≤ t0+20 + 관측 하나) · deadline_hit."""
    cleanup_group = L.cleanup_group
    clk, starts, kills, fired = _Clock(), [], [], []
    t0 = clk()
    proc, read = _fake_group(tmp_path, 4242, [(4242, "S")], clk, 0.05, starts)

    def sleep(s):
        if clk() >= t0 + 10.0 and not fired:
            fired.append(clk())
            raise RuntimeError("sleep failed (시험 주입)")
        clk.sleep(s)

    r = cleanup_group(_P(4242, clk), True, deadline=t0 + 20.0, clock=clk, sleep=sleep, killpg=lambda pg, sig: kills.append(pg), kill_wait=10.0,
                      grace=5.0, proc=proc, read=read)
    assert r["status"] == "unknown" and r["group"]["status"] == "unknown" and r["emergency"] is True
    assert r["errors"] == ["RuntimeError: sleep failed (시험 주입)"] and t0 + 10.0 <= fired[0] < t0 + 10.2 and kills == [4242, 4242]
    assert starts and all(s < t0 + 20.0 for s in starts) and clk() <= t0 + 20.0 + 0.05 + 1e-9 and r["deadline_hit"] is True

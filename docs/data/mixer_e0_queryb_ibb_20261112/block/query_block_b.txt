AWK_REDLIB='
function isw(c) { return (c ~ /[A-Za-z0-9_]/) }
function rlit(s, a, b,   i, o) { o = ""; while ((i = index(s, a)) > 0) { o = o substr(s, 1, i - 1) b; s = substr(s, i + length(a)) } return o s }
function rword(s, a, b,   i, o, pr, po) {
  o = ""
  while ((i = index(s, a)) > 0) {
    if (i > 1) pr = substr(s, i - 1, 1); else if (length(o) > 0) pr = substr(o, length(o), 1); else pr = ""
    po = substr(s, i + length(a), 1)
    if ((pr == "" || isw(pr) == 0) && (po == "" || isw(po) == 0)) o = o substr(s, 1, i - 1) b
    else o = o substr(s, 1, i - 1 + length(a))
    s = substr(s, i + length(a))
  }
  return o s
}
function buser(s,   i, j, r, k) {
  i = index(s, "compiled "); if (i == 0) return s
  j = index(substr(s, i), " by "); if (j == 0) return s
  j = i + j + 3; r = substr(s, j); k = match(r, /[ ,)]/)
  if (k == 0) return substr(s, 1, j - 1) "<BUILD_USER>"
  return substr(s, 1, j - 1) "<BUILD_USER>" substr(r, k)
}
function loadtok(   k, m, v) {
  NT = 0; m = ENVIRON["MIX_TN"] + 0; if (m > 5000) m = 5000
  for (k = 1; k <= m; k++) { v = ENVIRON["MIX_T" k]; if (length(v) > 0) { NT++; TT[NT] = v; TR[NT] = ENVIRON["MIX_R" k]; TW[NT] = ENVIRON["MIX_W" k] } }
}
function red(s,   k) {
  for (k = 1; k <= NT; k++) { if (TW[k] == "1") s = rword(s, TT[k], TR[k]); else s = rlit(s, TT[k], TR[k]) }
  gsub(/\/home\/[A-Za-z0-9._-]+/, "/home/<OTHER_USER>", s)
  gsub(/[0-9]+\.[0-9]+\.[0-9]+\.[0-9]+/, "<IPV4>", s); gsub(/<IPV4>:[0-9]+/, "<IPV4>:<PORT>", s)
  gsub(/host:[^: ]+:[0-9]+/, "host:<HOST>:<PORT>", s)
  gsub(/uid[ =]?[0-9]+/, "uid <UID>", s); gsub(/gid[ =]?[0-9]+/, "gid <GID>", s)
  gsub(/CANCELLED by [0-9]+/, "CANCELLED by <UID>", s)
  return buser(s)
}
function redall(s,   k) { for (k = 1; k <= NT; k++) if (length(TT[k]) >= 4) s = rlit(s, TT[k], TR[k]); return s }
'

put() { local v=${2-}; v=${v//$'\n'/ }; v=${v//$'\r'/ }; printf '%s=%s\n' "$1" "$v" >> "$FR"; }

over() { [ "$INTR" = 1 ] || [ "$SECONDS" -ge "$BUDGET" ]; }
why() { if [ "$INTR" = 1 ]; then printf SKIPPED_INTERRUPT; else printf SKIPPED_BUDGET; fi; }
want() { case " $QSEL " in *" $1 "*) return 0;; esac; return 1; }
probe() {  # probe <이름> <초> <출력 파일> <명령...> → 출력 파일 (stdout+stderr · 상한 CAP) · raw/probes.txt 한 줄 · 반환 = 명령 rc
  local name=$1 secs=$2 f=$3 t0 rc st by
  shift 3
  if over; then printf '%s|%s|-|0|0|%s\n' "$name" "$(why)" "${1##*/}" >> "$PR"; : > "$f"; return 200; fi
  command -v -- "$1" >/dev/null 2>&1 || { printf '%s|MISSING|-|0|0|%s\n' "$name" "${1##*/}" >> "$PR"; : > "$f"; return 127; }
  t0=$SECONDS
  timeout $TK "$secs" "$@" </dev/null 2>&1 | timeout 900 head -c "$CAP" > "$f"
  rc=${PIPESTATUS[0]}
  by=$(timeout 10 wc -c < "$f" 2>/dev/null); by=${by//[^0-9]/}
  case "$rc" in 0) st=OK;; 124|137) st=TIMEOUT;; 126|127) st=MISSING;; 141) st=TRUNC;; *) st=FAIL;; esac
  [ "$st" = OK ] && [ "${by:-0}" -ge "$CAP" ] && st=TRUNC
  printf '%s|%s|%s|%s|%s|%s\n' "$name" "$st" "$rc" "$((SECONDS - t0))" "${by:-0}" "${1##*/}" >> "$PR"
  return "$rc"
}
stl() {  # stl <이름> <rc> <초> <바이트> <명령> → 상태 줄 하나 (rc → 상태)
  local s
  case "$2" in 0) s=OK;; 124|137) s=TIMEOUT;; 126|127) s=MISSING;; *) s=FAIL;; esac
  printf '%s|%s|%s|%s|%s|%s\n' "$1" "$s" "$2" "$3" "$4" "$5" >> "$PR"
}
ftype() {  # ftype <경로> → "f|크기|mtime" (정규) · e (빈 정규) · l (링크 · 따라가지 않음) · d · o 또는 "ABSENT||" · "DENIED||" · "STATFAIL||"
  local s e c
  s=$(timeout 10 stat -c '%F|%s|%Y' -- "$1" </dev/null 2>"$D/work/stat_err.txt") || {
    e=$(timeout 5 head -c 400 -- "$D/work/stat_err.txt" 2>/dev/null)
    case "$e" in *'ermission denied'*) printf 'DENIED||';; *'o such file'*|*'ot a directory'*) printf 'ABSENT||';; *) printf 'STATFAIL||';; esac
    return 1; }
  case "${s%%|*}" in 'regular file') c=f;; 'regular empty file') c=e;; 'symbolic link') c=l;; directory) c=d;; *) c=o;; esac
  printf '%s|%s' "$c" "${s#*|}"
}
hashline() {  # hashline <경로> <최대 바이트> → "sha256|바이트|mtime" 또는 "사유|바이트|mtime" (해시 전후 stat 이 같아야 sha256)
  local p=$1 max=$2 s t sz mt h a rc
  s=$(timeout 10 stat -c '%F|%s|%Y' -- "$p" </dev/null 2>/dev/null) || { printf 'ABSENT||'; return 1; }
  t=${s%%|*}; s=${s#*|}; sz=${s%%|*}; mt=${s#*|}
  case "$t" in 'regular file'|'regular empty file') ;; *) printf 'NOT_REGULAR|%s|%s' "$sz" "$mt"; return 1;; esac
  [ "$sz" -le "$max" ] 2>/dev/null || { printf 'TOO_BIG|%s|%s' "$sz" "$mt"; return 1; }
  h=$(timeout $TK 120 sha256sum -- "$p" </dev/null 2>/dev/null); rc=$?; h=${h%% *}
  case "$rc" in 124|137) printf 'HASH_TIMEOUT|%s|%s' "$sz" "$mt"; return 1;; esac
  case "$h" in ''|*[^0-9a-f]*) printf 'HASH_FAIL|%s|%s' "$sz" "$mt"; return 1;; esac
  [ "${#h}" -eq 64 ] || { printf 'HASH_FAIL|%s|%s' "$sz" "$mt"; return 1; }
  a=$(timeout 10 stat -c '%F|%s|%Y' -- "$p" </dev/null 2>/dev/null)
  [ "$a" = "$t|$sz|$mt" ] || { printf 'CHANGED_DURING_HASH|%s|%s' "$sz" "$mt"; return 1; }
  printf '%s|%s|%s' "$h" "$sz" "$mt"
}
DONE=0
batch() {  # batch <이름> <초> <출력> <명령…> — 전역 배열 BATCH 를 CH7 개씩 잘라 명령 뒤에 붙인다 · 상태 줄 하나 · DONE = 끝까지 돈 묶음까지의 항목 수
  local name=$1 secs=$2 f=$3 i=0 rc w=0 t0=$SECONDS nb=${#BATCH[@]} s
  shift 3
  : > "$f"; : > "$f.err"; DONE=0
  while [ "$i" -lt "$nb" ]; do
    if over; then printf '%s|%s|-|%s|%s|%s\n' "$name" "$(why)" "$((SECONDS - t0))" "$i" "${1##*/}" >> "$PR"; return 0; fi
    timeout $TK "$secs" "$@" "${BATCH[@]:i:CH7}" </dev/null >> "$f" 2>> "$f.err"; rc=$?
    case "$rc" in 0) ;; 124|137) w=124; break;; 126|127) w=127; break;; *) [ "$w" = 0 ] && w=$rc;; esac
    i=$((i + CH7)); DONE=$i
  done
  [ "$DONE" -gt "$nb" ] && DONE=$nb
  case "$w" in 0) s=OK;; 124) s=TIMEOUT;; 127) s=MISSING;; *) s=FAIL;; esac
  printf '%s|%s|%s|%s|%s|%s\n' "$name" "$s" "$w" "$((SECONDS - t0))" "$nb" "${1##*/}" >> "$PR"
}

# ── q2b DEV 덤프 프레임의 줄 끝 꼴 (처음 · 가운데 · 끝 — 이름만 고름) — 머리는 셸 변수로만 읽는다 (파일에 안 씀)
frame_check() {  # frame_check <자리> <step> <경로>
  local pos=$1 step=$2 f=$3 t ty sz t0 h rc lb rc2 n rc3 k r e1=- nl=- l1=- ts=- e3=- at=- eq=-
  local -a HL=()
  t=$(ftype "$f"); ty=${t%%|*}; sz=${t#*|}; sz=${sz%%|*}
  case "$ty" in
    f|e) ;;
    *) case "$ty" in ABSENT|DENIED) r=$ty;; *) r=FAIL;; esac
       printf 'q2b_%s_stat|%s|-|0|0|stat\n' "$pos" "$r" >> "$PR"
       printf 'frame|pos=%s|step=%s|type=%s|bytes=%s|end_nl=-|newlines=-|l1_timestep=-|tstep=-|l3_natoms=-|atoms=-|nl_eq_9_plus_atoms=-\n' "$pos" "$step" "$ty" "${sz:--}" >> "$D/raw/q2b_frames.txt"
       return 0;;
  esac
  if over; then printf 'q2b_%s_head|%s|-|0|0|head\n' "$pos" "$(why)" >> "$PR"
  else
    t0=$SECONDS
    h=$(timeout $TK 30 head -c 4096 -- "$f" 2>/dev/null; printf 'x%s' "$?")
    rc=${h##*x}; h=${h%x*}
    stl "q2b_${pos}_head" "$rc" "$((SECONDS - t0))" "${#h}" head
    if [ "$rc" = 0 ]; then
      k=0; r=$h
      while [ "$k" -lt 4 ]; do
        case "$r" in *$'\n'*) ;; *) break;; esac
        k=$((k + 1)); HL[k]=${r%%$'\n'*}; r=${r#*$'\n'}
      done
      if [ "${HL[1]-}" = 'ITEM: TIMESTEP' ]; then l1=yes; else l1=no; fi
      if [ "$k" -lt 2 ]; then ts=missing
      else
        case "${HL[2]}" in
          ''|*[^0-9]*) ts=nonint;;
          *) if [ "${HL[2]}" = "$step" ]; then ts=same
             elif [ "${#HL[2]}" -le 18 ] && [ "${#step}" -le 18 ] && [ "$((10#${HL[2]}))" = "$((10#$step))" ]; then ts=num
             else ts=differ; fi;;
        esac
      fi
      if [ "${HL[3]-}" = 'ITEM: NUMBER OF ATOMS' ]; then e3=yes; else e3=no; fi
      if [ "$k" -lt 4 ]; then at=missing
      else case "${HL[4]}" in ''|*[^0-9]*) at=NONINT;; *) if [ "${#HL[4]}" -le 18 ]; then at=$((10#${HL[4]})); else at=TOOLONG; fi;; esac; fi
    fi
    h=''; r=''; HL=()
  fi
  if over; then printf 'q2b_%s_tail|%s|-|0|0|tail\n' "$pos" "$(why)" >> "$PR"
  else
    t0=$SECONDS
    lb=$(timeout $TK 30 tail -c 1 -- "$f" 2>/dev/null; printf 'x%s' "$?")
    rc2=${lb##*x}; lb=${lb%x*}
    stl "q2b_${pos}_tail" "$rc2" "$((SECONDS - t0))" "${#lb}" tail
    if [ "$rc2" = 0 ]; then if [ "$lb" = $'\n' ]; then e1=yes; else e1=no; fi; fi
    lb=''
  fi
  if over; then printf 'q2b_%s_wc|%s|-|0|0|wc\n' "$pos" "$(why)" >> "$PR"
  else
    t0=$SECONDS
    n=$(timeout $TK 120 wc -l -- "$f" 2>/dev/null); rc3=$?
    n=${n%% *}
    stl "q2b_${pos}_wc" "$rc3" "$((SECONDS - t0))" 0 wc
    if [ "$rc3" = 0 ]; then case "$n" in ''|*[^0-9]*) ;; *) nl=$n;; esac; fi
  fi
  case "$nl" in ''|*[^0-9]*) eq=-;; *) case "$at" in ''|*[^0-9]*) eq=-;; *) if [ "$nl" = "$((at + 9))" ]; then eq=yes; else eq=no; fi;; esac;; esac
  printf 'frame|pos=%s|step=%s|type=%s|bytes=%s|end_nl=%s|newlines=%s|l1_timestep=%s|tstep=%s|l3_natoms=%s|atoms=%s|nl_eq_9_plus_atoms=%s\n' \
    "$pos" "$step" "$ty" "$sz" "$e1" "$nl" "$l1" "$ts" "$e3" "$at" "$eq" >> "$D/raw/q2b_frames.txt"
}
q2b() {
  local P="$BASE/$ONE_RUN/post" tg pos step
  if over; then printf 'q2b|%s|-|0|0|-\n' "$(why)" >> "$PR"; return 0; fi
  put q2b.run "$ONE_RUN"
  probe q2b_names 120 "$D/work/q2b_names.txt" find "$P" -mindepth 1 -maxdepth 1 -type f -name 'mix_*.liggghts' -printf '%f\n'
  probe q2b_steps 30 "$D/work/q2b_steps.txt" awk "$AWK_Q2B_STEPS" "$D/work/q2b_names.txt"
  probe q2b_sort 60 "$D/work/q2b_sorted.txt" sort -n -- "$D/work/q2b_steps.txt"
  probe q2b_pick 30 "$D/raw/q2b_pick.txt" awk -v run="$ONE_RUN" -v f1="$D/work/q2b_sorted.txt" "$AWK_Q2B_PICK" "$D/work/q2b_sorted.txt" "$D/work/q2b_names.txt"
  : > "$D/raw/q2b_frames.txt"
  while IFS='|' read -r tg pos step; do
    [ "$tg" = sel ] || continue
    case "$step" in ''|*[^0-9]*) continue;; esac
    case "$pos" in first|mid|last) ;; *) continue;; esac
    frame_check "$pos" "$step" "$P/mix_$step.liggghts"
  done < "$D/raw/q2b_pick.txt"
}
# ── q7b Open MPI 런타임 부품의 설치 신원 (conda-meta JSON 의 파일 목록 → 파일마다 sha256 · 번호로만)
q7b_pkg() {  # q7b_pkg <번호> <꾸러미 이름> <환경 실제 경로>
  local k=$1 p=$2 ER=$3 S="$D/raw/q7b_p$1_sum.txt" FL="$D/raw/q7b_p$1_files.txt" W7="$D/work/q7b_p$1" jn m x J h po DL DT DH DT2 nd sl

  m=$(printf '%s' "$jn" | timeout 10 awk 'END { print NR }' 2>/dev/null); m=${m//[^0-9]/}; m=${m:-0}
  if [ "$m" = 0 ]; then printf 'pkg|%s|name=%s|json=-|matches=0\n' "$k" "$p" > "$S"; printf 'q7b_p%s_json|ABSENT|-|0|0|find\n' "$k" >> "$PR"; return 0; fi
  if [ "$m" -gt 1 ]; then printf 'pkg|%s|name=%s|json=-|matches=%s\n' "$k" "$p" "$m" > "$S"; printf 'q7b_p%s_json|FAIL|-|0|0|find\n' "$k" >> "$PR"; return 0; fi
  J="$ER/conda-meta/$jn"
  printf 'pkg|%s|name=%s|json=%s|matches=1\n' "$k" "$p" "$jn" > "$S"
  printf 'q7b_p%s_json|OK|-|0|0|find\n' "$k" >> "$PR"
  h=$(hashline "$J" 67108864) && x=OK || x=FAIL
  printf 'json|%s|%s\n' "$k" "$h" >> "$S"
  printf 'q7b_p%s_jsonhash|%s|-|0|0|sha256sum\n' "$k" "$x" >> "$PR"
  probe "q7b_p${k}_parse" 60 "${W7}_parse.txt" awk "$AWK_JSONFILES" "$J"
  po=$(timeout 10 awk -F'|' '$1 == "S" { print substr($0, 3) }' "${W7}_parse.txt" 2>/dev/null)
  printf 'parse|%s|%s\n' "$k" "${po:--}" >> "$S"
  case "$po" in elements=*'|files_keys=1|depth_end=0|instr_end=0|bad=0') printf 'q7b_p%s_parse_ok|OK|-|0|0|awk\n' "$k" >> "$PR";;
    *) printf 'q7b_p%s_parse_ok|FAIL|-|0|0|awk\n' "$k" >> "$PR"; return 0;; esac
  probe "q7b_p${k}_list" 30 "${W7}_abs.txt" awk -v pre="$ER/" -v maxf="$MAXF7" "$AWK_Q7B_LIST" "${W7}_parse.txt"
  BATCH=(); mapfile -t BATCH < "${W7}_abs.txt"
  batch "q7b_p${k}_lstat" 60 "${W7}_lstat.txt" stat -c 'L|%F|%s|%Y|%n' --; DL=$DONE
  batch "q7b_p${k}_tstat" 60 "${W7}_tstat.txt" stat -L -c 'T|%F|%s|%Y|%n' --; DT=$DONE
  probe "q7b_p${k}_sel" 30 "${W7}_tohash.txt" awk -F'|' -v maxb="$MAXB7" '$1 == "T" && ($2 == "regular file" || $2 == "regular empty file") && $3 + 0 <= maxb + 0 { p = $0; sub(/^[^|]*[|][^|]*[|][^|]*[|][^|]*[|]/, "", p); print p }' "${W7}_tstat.txt"
  BATCH=(); mapfile -t BATCH < "${W7}_tohash.txt"
  batch "q7b_p${k}_hash" 120 "${W7}_sha.txt" sha256sum --; DH=$DONE

  case "${sl%%|*}" in ''|0) ;; *) printf 'q7b_p%s_cap|TRUNC|-|0|%s|awk\n' "$k" "${sl%%|*}" >> "$PR";; esac
}

class Refusal(Exception):
    """의도된 거부 — 아무것도 쓰지 않고 반환 2."""


def utc() -> str:
    return datetime.now(timezone.utc).strftime("%Y-%m-%dT%H:%M:%S.%f")[:-3] + "Z"


def sha256(b: bytes) -> str:
    return hashlib.sha256(b).hexdigest()


def dump(obj) -> bytes:
    return (json.dumps(obj, ensure_ascii=False, indent=1, sort_keys=True) + "\n").encode("utf-8")


def write_new(path, data: bytes) -> None:
    """배타 생성 (있으면 FileExistsError)."""
    with open(path, "xb") as f:
        f.write(data)


def publish(path: Path, data: bytes) -> None:
    """임시 이름에 쓰고 fsync · 닫은 뒤 최종 이름으로 배타 링크 (이미 있으면 FileExistsError) → 임시 이름을 지운다. 최종 이름이 보이는 순간 내용이
    완전하다."""
    tmp = path.with_name(path.name + ".tmp")
    with open(tmp, "xb") as f:
        f.write(data)
        f.flush()
        os.fsync(f.fileno())
    os.link(tmp, path)
    os.unlink(tmp)


def names(ev: Path, step: str) -> dict:
    name = STEPS[step]
    return {k: ev / f"{name}.{k}" for k in LAUNCHER_FILES + OUTER_FILES}


def stat_fields(pid: int) -> dict:
    """/proc/<pid>/stat → 상태 · ppid · pgid · 세션 · 시작 시각 (괄호 안 comm 의 공백 · 괄호와 무관하게 마지막 ')' 뒤를 읽는다)."""
    raw = (PROC / str(pid) / "stat").read_bytes()
    rest = raw[raw.rindex(b")") + 2:].split()
    return {"state": rest[0].decode("ascii", "replace"), "ppid": int(rest[1]), "pgid": int(rest[2]), "sid": int(rest[3]),
            "starttime": int(rest[19])}


def ancestors() -> set:
    """자기와 조상 pid (셸 도구의 셸 포함) — 관측 · 정리에서 뺀다."""
    seen, pid = set(), os.getpid()
    while pid > 1 and pid not in seen:
        seen.add(pid)
        try:
            pid = stat_fields(pid)["ppid"]
        except (OSError, ValueError, IndexError):
            break
    return seen


def start_record(p: dict) -> dict:
    return json.loads(p["outer_start.json"].read_text(encoding="utf-8"))


# ---------------------------------------------------------------- start
def start(step: str, ev: Path, section: str) -> int:
    p = names(ev, step)
    taken = [k for k in OUTER_FILES if k != "wrapper_output.txt" and os.path.lexists(p[k])]
    if taken:
        raise Refusal(f"외곽 기록이 이미 있다 — 덮어쓰지 않는다: {taken}")
    me = Path(__file__).resolve()
    lb = LAUNCHER.read_bytes()
    argv = [PY, "-I", str(LAUNCHER), step, "--evidence", str(ev), "--approval-section", section]
    t0 = time.time()
    rec = {"kind": "REIL_E113_OUTER_START", "step": step, "name": STEPS[step], "utc": utc(), "epoch": t0,
           "deadline_epoch": t0 + ALLOW_S[step], "allow_s": ALLOW_S[step], "pid": os.getpid(), "ppid": os.getppid(),
           "sid": os.getsid(0), "pgid": os.getpgid(0), "argv": argv, "cwd": str(BMS),
           "outer": {"path": str(me), "sha256": sha256(me.read_bytes())},
           "launcher": {"path": str(LAUNCHER), "sha256": sha256(lb), "expected": LAUNCHER_SHA256}, "python": sys.version}
    publish(p["outer_start.json"], dump(rec))
    if rec["launcher"]["sha256"] != LAUNCHER_SHA256:
        publish(p["outer_error.json"], dump({"phase": "precheck", "utc": utc(),
                                             "message": f"launcher sha256 {rec['launcher']['sha256']} ≠ 고정 {LAUNCHER_SHA256}"}))
        return 3
    try:
        with open(p["outer_stdout.txt"], "xb") as fo, open(p["outer_stderr.txt"], "xb") as fe:
            child = subprocess.Popen(argv, cwd=str(BMS), stdin=subprocess.DEVNULL, stdout=fo, stderr=fe)
            rc = child.wait()
    except Exception as e:  # noqa: BLE001 — 띄우기 · 기다리기의 실패는 외곽 오류로 남기고 반환을 게시하지 않는다
        publish(p["outer_error.json"], dump({"phase": "launch", "utc": utc(), "type": e.__class__.__name__, "message": str(e)[:500]}))
        return 3
    code = rc if rc >= 0 else 128 - rc  # 시그널로 끝나면 셸 관례 128 + 번호
    try:
        publish(p["outer_rc.txt"], f"{code}\n".encode("ascii"))
    except Exception as e:  # noqa: BLE001
        publish(p["outer_error.json"], dump({"phase": "publish", "utc": utc(), "rc": code, "type": e.__class__.__name__,
                                             "message": str(e)[:500]}))
        return 4
    return 0


# ---------------------------------------------------------------- wait (읽기 전용)
def wait(step: str, ev: Path) -> int:
    p = names(ev, step)
    t_begin = time.time()

    def out(state: str, code: int, **kw) -> int:
        print(json.dumps({"step": step, "state": state, "utc": utc(), "waited_s": round(time.time() - t_begin, 1), **kw},
                         ensure_ascii=False))
        return code

    while not os.path.lexists(p["outer_start.json"]):
        if os.path.lexists(p["outer_error.json"]):
            return out("outer_error", 12)
        if time.time() - t_begin >= START_WAIT_S:
            return out("start_missing", 13)
        time.sleep(1.0)
    try:
        rec = start_record(p)
        deadline, wpid = float(rec["deadline_epoch"]), int(rec["pid"])
    except (OSError, ValueError, KeyError, TypeError) as e:
        return out("start_unreadable", 14, error=f"{e.__class__.__name__}: {e}"[:300])
    while True:
        if os.path.lexists(p["outer_rc.txt"]):
            b = p["outer_rc.txt"].read_bytes()
            if RC_FORMAT.fullmatch(b):
                # 게시 직후의 wrapper 가 잔류 관측에 잡히지 않도록, wrapper 가 끝날 때까지 (없음 또는 Z) 최대 WRAPPER_EXIT_S 를 읽기만 하며 기다린다
                t_pub = time.time()
                while time.time() - t_pub < WRAPPER_EXIT_S:
                    try:
                        if stat_fields(wpid)["state"] == "Z":
                            break
                    except (OSError, ValueError, IndexError):
                        break
                    time.sleep(0.5)
                try:
                    wstate = stat_fields(wpid)["state"]
                except (OSError, ValueError, IndexError):
                    wstate = "absent"
                return out("published", 0, rc=int(b), deadline_epoch=deadline, wrapper_state=wstate)
            return out("malformed", 11, raw_hex=b[:20].hex(), deadline_epoch=deadline)
        if os.path.lexists(p["outer_error.json"]):
            return out("outer_error", 12, deadline_epoch=deadline)
        if time.time() >= deadline:
            return out("deadline", 10, deadline_epoch=deadline)
        time.sleep(POLL_S)


# ---------------------------------------------------------------- observe (읽기 · 기록 하나)
def observe(step: str, ev: Path, rnd: str) -> int:
    p, name = names(ev, step), STEPS[step]
    outp = ev / f"{name}.leftover_{rnd}.json"
    if os.path.lexists(outp):
        raise Refusal(f"이미 있다: {outp}")
    sid, sid_error = None, None
    try:
        sid = int(start_record(p)["sid"])
    except (OSError, ValueError, KeyError, TypeError) as e:
        sid_error = f"{e.__class__.__name__}: {e}"[:300]
    markers = [f"pycache_prefix={WORK / 'pycache' / name}".encode(), str(ev).encode()]
    skip = ancestors()
    rows, errors = [], []
    for d in sorted((x for x in PROC.iterdir() if x.name.isdigit()), key=lambda x: int(x.name)):
        pid = int(d.name)
        if pid in skip:
            continue
        try:
            st = stat_fields(pid)
            argv = (d / "cmdline").read_bytes().split(b"\0")
        except (FileNotFoundError, ProcessLookupError):
            continue  # 관측 사이에 끝난 프로세스
        except (OSError, ValueError, IndexError) as e:
            errors.append({"pid": pid, "type": e.__class__.__name__, "message": str(e)[:200]})
            continue
        why = (["sid"] if sid is not None and st["sid"] == sid else []) + \
              [f"marker{i}" for i, m in enumerate(markers) if any(m in a for a in argv)]
        if why:
            rows.append({"pid": pid, **st, "matched": why, "argv": [a.decode("utf-8", "replace")[:300] for a in argv if a][:40]})
    alive = [r for r in rows if r["state"] != "Z"]
    write_new(outp, dump({"kind": "REIL_E113_LEFTOVER", "step": step, "round": rnd, "utc": utc(), "sid": sid, "sid_error": sid_error,
                          "markers": [m.decode() for m in markers], "skipped_self_and_ancestors": sorted(skip), "rows": rows,
                          "alive_count": len(alive), "errors": errors}))
    print(json.dumps({"step": step, "round": rnd, "rows": len(rows), "alive": len(alive), "errors": len(errors)}, ensure_ascii=False))
    return 0 if not alive and not errors else 1


# ---------------------------------------------------------------- kill (R1)
def kill(step: str, ev: Path) -> int:
    name = STEPS[step]
    src, outp = ev / f"{name}.leftover_1.json", ev / f"{name}.kill.json"
    if os.path.lexists(outp):
        raise Refusal(f"이미 있다: {outp}")
    raw = src.read_bytes()
    obs = json.loads(raw)
    skip = ancestors()
    own = {os.getpgid(0)}
    for a in skip:
        try:
            own.add(stat_fields(a)["pgid"])
        except (OSError, ValueError, IndexError):
            pass
    groups = {}
    for r in obs.get("rows") or []:
        if r.get("state") != "Z":
            groups.setdefault(int(r["pgid"]), []).append(r)
    results = []
    for pgid, members in sorted(groups.items()):
        if pgid <= 1 or pgid in own:
            results.append({"pgid": pgid, "action": "skip", "why": "1 이하 · 자기 · 조상의 그룹"})
            continue
        same = None
        for m in members:
            try:
                st = stat_fields(int(m["pid"]))
            except (OSError, ValueError, IndexError):
                continue
            if st["starttime"] == m.get("starttime") and st["pgid"] == pgid:
                same = int(m["pid"])
                break
        if same is None:
            results.append({"pgid": pgid, "action": "skip", "why": "관측한 구성원이 없거나 시작 시각 · 그룹이 바뀌었다"})
            continue
        try:
            os.killpg(pgid, signal.SIGKILL)
            results.append({"pgid": pgid, "action": "SIGKILL", "ok": True, "verified_pid": same})
        except OSError as e:
            results.append({"pgid": pgid, "action": "SIGKILL", "ok": False, "verified_pid": same, "type": e.__class__.__name__,
                            "message": str(e)[:200]})
    write_new(outp, dump({"kind": "REIL_E113_KILL", "step": step, "utc": utc(), "source": {"path": str(src), "sha256": sha256(raw)},
                          "results": results}))
    print(json.dumps({"step": step, "groups": len(groups), "results": [r["action"] for r in results]}, ensure_ascii=False))
    return 0


# ---------------------------------------------------------------- freeze (F3)
def freeze(step: str, ev: Path) -> int:
    name = STEPS[step]
    dst, man = ev / f"frozen_{name}", ev / f"frozen_{name}.json"
    if os.path.lexists(dst) or os.path.lexists(man):
        raise Refusal(f"이미 있다: {dst} 또는 {man}")
    p = names(ev, step)
    files = [p[k] for k in LAUNCHER_FILES + OUTER_FILES]
    if step == "identity":
        files.append(ev / "03_identity.json")
    if step == "run" and (ev / "run").is_dir():
        files += sorted(x for x in (ev / "run").rglob("*") if x.is_file())
    dst.mkdir()
    rows = []
    for f in files:
        rel = f.relative_to(ev)
        if f.is_symlink() or not f.is_file():
            rows.append({"path": str(rel), "state": "absent_or_not_regular"})
            continue
        t = utc()
        b = f.read_bytes()
        target = dst / rel
        target.parent.mkdir(parents=True, exist_ok=True)
        write_new(target, b)
        rows.append({"path": str(rel), "bytes": len(b), "sha256": sha256(b), "read_utc": t})
    write_new(man, dump({"kind": "REIL_E113_FROZEN", "step": step, "utc": utc(), "rows": rows,
                         "note": "읽은 범위의 고정 관측 사본 — 원본은 그대로 · 여러 파일의 원자적 · 동시 시점이 아니다"}))
    print(json.dumps({"step": step, "copied": sum(1 for r in rows if "sha256" in r), "absent": sum(1 for r in rows if "state" in r)},
                     ensure_ascii=False))
    return 0


# ---------------------------------------------------------------- main
def evidence(arg: str) -> Path:
    ev = Path(arg)
    if ev.parent != BMS / "evidence" or not EVIDENCE_NAME.match(ev.name) or ev.is_symlink() or not ev.is_dir():
        raise Refusal(f"증거 폴더 규칙 밖: {arg}")
    me = Path(__file__)
    if me.is_symlink() or me.resolve() != (ev / "00_outer.py").resolve():
        raise Refusal(f"이 파일이 그 증거 폴더의 00_outer.py 가 아니다: {me}")
    return ev


def main(argv=None) -> int:
    a = list(sys.argv[1:] if argv is None else argv)
    try:
        if len(a) < 3 or a[0] not in MODES or a[1] not in STEPS:
            raise Refusal(f"인자 — <{'|'.join(MODES)}> <{'|'.join(STEPS)}> <E> …: {a}")
        mode, step, ev = a[0], a[1], evidence(a[2])
        if mode == "start":
            if len(a) != 4 or not SECTION.match(a[3]):
                raise Refusal(f"start 는 승인 절 §NN 하나를 더 받는다: {a[3:]}")
            return start(step, ev, a[3])
        if mode == "observe":
            if len(a) != 4 or a[3] not in ("1", "2"):
                raise Refusal(f"observe 는 차례 1 또는 2 를 더 받는다: {a[3:]}")
            return observe(step, ev, a[3])
        if len(a) != 3:
            raise Refusal(f"{mode} 는 인자 셋이다: {a}")
        return {"wait": wait, "kill": kill, "freeze": freeze}[mode](step, ev)
    except Refusal as e:
        sys.stderr.write(f"거부: {e}\n")
        return 2


if __name__ == "__main__":
    sys.exit(main())

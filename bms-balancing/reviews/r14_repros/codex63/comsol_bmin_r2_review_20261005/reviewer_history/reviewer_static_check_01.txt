def model(text):
    mentions = [x for x in text.splitlines() if 'Number of degrees of freedom' in x]
    if len(mentions) != 1: return {'reason': 'TRANSIENT_DOF_READBACK_COUNT:'+str(len(mentions))}
    hit = re.fullmatch(r'\s*'+pattern+r'\s*', mentions[0])
    if hit is None: return {'reason': 'TRANSIENT_DOF_READBACK_FORMAT'}
    solved, internal = map(int, hit.groups())
    if solved <= 0: return {'reason': 'TRANSIENT_DOF_READBACK_VALUE'}
    return {'reason': 'ACCEPT_RECORD_ONLY', 'solved': solved, 'internal': internal}

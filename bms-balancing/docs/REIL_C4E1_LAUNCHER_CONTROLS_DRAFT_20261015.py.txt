def _sha(b: bytes) -> str:
    return hashlib.sha256(b).hexdigest()


def _write_list(d: Path, files: dict, schema="reil-c4e-sources-v1", raw: bytes | None = None):
    d.mkdir(parents=True)
    b = raw if raw is not None else json.dumps({"schema": schema, "files": files}, sort_keys=True, indent=1).encode()
    (d / "SOURCES.json").write_bytes(b)
    return d / "SOURCES.json", _sha(b)


def _tree(tmp: Path) -> SimpleNamespace:
    """tmp 안의 가짜 bms (scripts 다섯 · `__init__` 없음) · 그 목록 · 자료 두 파일 · 증거 폴더 · 작업 폴더."""
    tmp.mkdir(parents=True, exist_ok=True)
    tmp = tmp.resolve()
    bms = tmp / "bms"
    (bms / "scripts").mkdir(parents=True)
    content = {r: f"# {r} (대조용 가짜 · 실행하지 않음)\n".encode() for r in RELS}
    content["scripts/reil_c4e_run.py"] = FAKE_RUN.encode()
    for r, b in content.items():
        (bms / r).write_bytes(b)
    entries = {r: {"sha256": _sha(b), "bytes": len(b)} for r, b in content.items()}
    lst, lsha = _write_list(tmp / "list", entries)
    data = tmp / "data"
    data.mkdir()
    (data / "LFP_Data.xlsx").write_bytes(b"xlsx-bytes")
    (data / "util_LFP.py").write_bytes(b"util-bytes")
    data_files = {"LFP_Data.xlsx": {"sha256": _sha(b"xlsx-bytes"), "bytes": 10}, "util_LFP.py": {"sha256": _sha(b"util-bytes"), "bytes": 10}}
    ev = tmp / "ev"
    ev.mkdir()
    return SimpleNamespace(tmp=tmp, bms=bms, list=lst, lsha=lsha, entries=entries, data=data, data_files=data_files, ev=ev,
                           work=tmp / "work")


def _pre(t, step="run", *, sources=None, sha=None, argv=None):
    prefix = t.work / "pycache" / f"pre_{step}"
    prefix.mkdir(parents=True, exist_ok=True)
    env, _ = L.child_env(dict(os.environ), t.work)
    src, s = sources or t.list, sha or t.lsha
    argv = argv or L.build_argv(step, t.ev, "§99", prefix, py=PY, bms=t.bms, data=t.data, sources=src, sources_sha256=s)
    return L.precheck(step, t.ev, argv, prefix, env=env, py=PY, bms=t.bms, data=t.data, data_files=t.data_files, sources=src,
                      sources_sha256=s)


def _launch(t, step="run", mode="ok", extra_env=None, **kw):
    opts = dict(repo=t.tmp, bms=t.bms, py=PY, work=t.work, data=t.data, data_files=t.data_files, sources=t.list, sources_sha256=t.lsha,
                timeouts=FAST, env_base={**os.environ, "FAKE_MODE": mode, **(extra_env or {})})
    opts.update(kw)
    return L.launch(step, t.ev, "§99", **opts)


def _meta(t, name="05_run"):
    return json.loads((t.ev / f"{name}.meta.json").read_text(encoding="utf-8"))

def test_l2_list_sha_schema_path_set_entry_format_and_non_json_each_block(tmp_path):
    """[RED 실패 — 골격] 고정 sha 와 다른 목록 · schema 다름 · 넷 · 목록 밖 · 대문자 sha · 음수 bytes · bool bytes · JSON 아님 → 각각 실패."""
    t = _tree(tmp_path)
    e, p7 = t.entries, t.entries["scripts/reil_p0.py"]
    cases = {"schema": _write_list(t.tmp / "schema", e, schema="reil-c4e-sources-v0"),
             "four": _write_list(t.tmp / "four", {k: v for k, v in e.items() if k != "scripts/reil_p0.py"}),
             "extra": _write_list(t.tmp / "extra", dict(e, **{"scripts/reil_c4_h.py": dict(p7)})),

             "negative": _write_list(t.tmp / "negative", dict(e, **{"scripts/reil_p0.py": {"sha256": p7["sha256"], "bytes": -1}})),
             "boolbytes": _write_list(t.tmp / "boolbytes", dict(e, **{"scripts/reil_p0.py": {"sha256": p7["sha256"], "bytes": True}})),
             "nonjson": _write_list(t.tmp / "nonjson", {}, raw=b"not json\n")}

def test_l3_changed_same_size_missing_symlinked_module_or_linked_parent_each_block(tmp_path):
    """[RED 실패 — 골격] 모듈 하나 +1 바이트 · 같은 크기 다른 바이트 · 없음 · 같은 바이트의 심볼릭 링크 · 링크된 상위 폴더 → 각각 실패."""

            t.bms.symlink_to(moved, target_is_directory=True)

def test_l4_a_shadowing_scripts_package_module_zip_entry_or_our_init_blocks(tmp_path):
    """[RED 실패 — 골격] 자식 sys.path 의 `scripts/__init__.py` · `scripts.py` · zip 안 `scripts/__init__.py` · 빈 항목 · 우리 `scripts/__init__.py` ·
    우리 bms 의 `scripts.py` → 문제. `__init__` 없는 같은 이름 폴더 (네임스페이스 조각) 만 있으면 문제 없음."""
    t = _tree(tmp_path / "t")
    clean = tmp_path / "clean"
    (clean / "scripts").mkdir(parents=True)
    assert L.shadow_problems([str(clean)], t.bms) == []
    pkg = tmp_path / "pkg"
    (pkg / "scripts").mkdir(parents=True)
    (pkg / "scripts" / "__init__.py").write_text("")
    mod = tmp_path / "mod"
    mod.mkdir()
    (mod / "scripts.py").write_text("")
    zp = tmp_path / "z.zip"
    with zipfile.ZipFile(zp, "w") as z:
        z.writestr("scripts/__init__.py", "")
    for entries in ([str(pkg)], [str(mod)], [str(zp)], [""]):
        assert L.shadow_problems(entries, t.bms), entries
    (t.bms / "scripts.py").write_text("")
    assert L.shadow_problems([str(clean)], t.bms)
    (t.bms / "scripts.py").unlink()
    (t.bms / "scripts" / "__init__.py").write_text("")
    assert L.shadow_problems([str(clean)], t.bms) and _pre(t)["ok"] is False

def test_l8_identity_link_binds_the_run_import_paths_to_the_checked_paths(tmp_path):
    """[RED 실패 — 골격] run 의 IDENTITY.json 시작 · 끝 표 모듈 행 = 사전 대조 경로 · 목록 → 문제 없음. 경로 다름 · sha 다름 · 끝 표 없음 ·

    """[RED 실패 — 골격] run 이 손주 프로세스를 띄우고 잠든다 → timeout (3 s) → 그룹 전체 SIGKILL · timed_out · stage false · 손주는 살아 있지

    """[RED 실패 — 골격] run 도중 모듈 하나가 바뀜 → 이 실행기의 직접 재해시로 사후 대조 실패 · stage false. fetch 가 기대와 다른 바이트를 쓰면

87: def _verify_files(dist) -> tuple[int, int, list]:
88:     """RECORD 의 해시 있는 항목을 다시 해시한다 → (일치 수, 해시 없는 수, [(상대경로, 기대, 관측 | None)]).
89: 
90:     없는 파일은 불일치다 (관측 `None`). 그 밖의 읽기 오류 · 모르는 해시 알고리즘은 **측정 실패**로 올린다 —
91:     파일 하나를 못 읽은 것을 "그 파일은 맞다" 로도 "다르다" 로도 적지 않는다.
92:     """
93:     ok, unhashed, bad = 0, 0, []
94:     for f in dist.files or ():
95:         if f.hash is None:
96:             unhashed += 1
97:             continue
98:         h = hashlib.new(f.hash.mode)                     # 모르는 알고리즘 → ValueError (측정 실패)
99:         want = f"{f.hash.mode}={f.hash.value}"
100:         try:
101:             with open(f.locate(), "rb") as fh:
102:                 for chunk in iter(lambda: fh.read(1 << 20), b""):
103:                     h.update(chunk)
104:         except FileNotFoundError:
105:             bad.append((str(f), want, None))
106:             continue
107:         got = base64.urlsafe_b64encode(h.digest()).rstrip(b"=").decode()
108:         if got == f.hash.value:
109:             ok += 1
110:         else:
111:             bad.append((str(f), want, f"{f.hash.mode}={got}"))
112:     return ok, unhashed, bad

115: def _origin(module: str, paths: list, effective: dict, files: dict, no_record: list):
116:     """핵심 module 을 `PathFinder.find_spec(module, paths)` 로 경로 검색한 origin 파일의 RECORD 소속
117:     → ("in_record", 주인) · ("unverifiable", origin) · ("mismatch", 관측).
118: 
119:     이미 로드된 module 객체 (`sys.modules`) · 다른 meta-path finder 는 보지 않는다 (91차 G90-N1 — 결과 `not_measured`).
120:     """
121:     spec = importlib.machinery.PathFinder.find_spec(module, paths)
122:     origin = getattr(spec, "origin", None) if spec is not None else None
123:     if spec is None:
124:         return "mismatch", "<not found>"
125:     if not origin or not os.path.isfile(origin):
126:         return "mismatch", f"<{origin or 'namespace'}>"
127:     real = os.path.realpath(origin)
128:     owners = sorted(
129:         name for name, entries in files.items()
130:         if any(os.path.realpath(f.locate()) == real for f in entries
131:                if f.parts and f.parts[0].split(".")[0] == module))
132:     if len(owners) == 1:
133:         return "in_record", owners[0]
134:     if not owners:
135:         for name in no_record:
136:             base = os.path.realpath(effective[name].locate_file(""))
137:             if os.path.commonpath([base, real]) == base:
138:                 return "unverifiable", real
139:     return "mismatch", real if not owners else f"{real} (주인 {', '.join(owners)})"

142: def measure(paths=None) -> dict:
143:     """그 프로세스의 `sys.path` (또는 주어진 경로 목록) 순서로 환경을 잰다. 실패는 예외로 올린다."""
144:     importlib.invalidate_caches()
145:     paths = [str(p) for p in (sys.path if paths is None else paths)]
146:     effective, meta, shadowed = {}, {}, []
147:     for entry in paths:
148:         for dist in md.distributions(path=[entry]):
149:             raw, version = dist.metadata["Name"], dist.version
150:             if not raw or not version:
151:                 raise ValueError(f"이름 · 버전 없는 배포판 메타데이터 ({entry})")
152:             name = normalize(raw)
153:             if not _NAME.fullmatch(name) or not _VERSION.fullmatch(version):
154:                 raise ValueError(f"lock 에 쓸 수 없는 배포판 이름 · 버전: {raw!r} {version!r} ({entry})")
155:             record = dist.read_text("RECORD")
156:             digest = "none" if record is None else hashlib.sha256(record.encode("utf-8")).hexdigest()
157:             if name in effective:
158:                 shadowed.append((name, version, digest))
159:             else:
160:                 effective[name], meta[name] = dist, (version, digest)
161:     dists, files, no_record = {}, {}, []
162:     verified = unhashed = 0
163:     file_bad = []
164:     for name in sorted(effective):
165:         version, digest = meta[name]
166:         dists[name] = {"version": version, "record": digest}
167:         if digest == "none":
168:             no_record.append(name)
169:             continue
170:         files[name] = list(effective[name].files or ())
171:         ok, un, bad = _verify_files(effective[name])
172:         verified += ok
173:         unhashed += un
174:         file_bad += [(name, rel, want, got) for rel, want, got in bad]
175:     origins, origin_unverifiable, origin_bad = {}, [], []
176:     for module in KEY_MODULES:
177:         kind, value = _origin(module, paths, effective, files, no_record)
178:         if kind == "in_record":
179:             origins[module] = value
180:         elif kind == "unverifiable":
181:             origin_unverifiable.append(module)
182:         else:
183:             origin_bad.append((module, value))
184:     return {
185:         "python": platform.python_version(),
186:         "implementation": platform.python_implementation(),
187:         "system": platform.system(),
188:         "machine": platform.machine(),
189:         "libc": " ".join(platform.libc_ver()).strip(),
190:         "dists": dists,
191:         "shadowed": sorted(shadowed),
192:         "dists_without_record": no_record,
193:         "files": {"verified": verified, "unhashed": unhashed, "mismatches": file_bad},
194:         "path_origins": {"in_record": origins, "unverifiable": origin_unverifiable, "mismatches": origin_bad},
195:     }

200: def emit_lock(profile: dict) -> str:
201:     """정규형 lock 텍스트 — `measure()` 나 `parse_lock()` 의 결과에서."""
202:     lines = list(HEADER)
203:     lines += [f"#@ {k} {json.dumps('C' if k == 'profile' else profile[k], ensure_ascii=False)}" for k in DIRECTIVES]
204:     lines.append("")
205:     lines += [f"{n}=={d['version']}  # record-sha256 {d['record']}" for n, d in sorted(profile["dists"].items())]
206:     if profile["shadowed"]:
207:         lines.append("")

212: def parse_lock(text: str) -> dict:
213:     """닫힌 문법 (고정 표 §3-1) 으로 읽는다. 어긋나면 `LockFormatError` (줄 번호를 담는다)."""
214:     seen, dists, shadowed = {}, {}, []
215:     last, last_shadow = None, None
216: 
217:     def bad(no: int, why: str) -> LockFormatError:
218:         return LockFormatError(f"줄 {no}: {why}")
219: 
220:     lines = text.split("\n")
221:     if lines and lines[-1] == "":
222:         lines.pop()
223:     for no, line in enumerate(lines, 1):
224:         if line == "" or (line.startswith("#") and not line.startswith("#@")):
225:             continue
226:         if line.startswith("#@ shadowed "):
227:             m = _SHADOW.fullmatch(line)
228:             if m is None:
229:                 raise bad(no, "가려진 항목의 꼴이 아니다")
230:             key = (m["name"], m["version"], m["record"])
231:             if not _NAME.fullmatch(key[0]):
232:                 raise bad(no, f"정규형 아닌 이름 {key[0]!r}")
233:             if key[2] != "none" and not _HEX64.fullmatch(key[2]):
234:                 raise bad(no, f"record-sha256 칸 {key[2]!r}")
235:             if last_shadow is not None and key < last_shadow:
236:                 raise bad(no, "가려진 항목의 정렬이 어긋났다")
237:             shadowed.append(key)
238:             last_shadow = key
239:             continue
240:         if line.startswith("#@"):
241:             m = _DIRECTIVE.fullmatch(line)
242:             if m is None or m["key"] not in DIRECTIVES:
243:                 raise bad(no, f"모르는 지시 {line!r}")
244:             if m["key"] in seen:
245:                 raise bad(no, f"지시 {m['key']} 가 두 번 있다")
246:             try:
247:                 value = json.loads(m["value"])
248:             except ValueError as exc:
249:                 raise bad(no, f"지시 {m['key']} 의 값이 JSON 이 아니다 ({exc})") from exc
250:             if not isinstance(value, str):
251:                 raise bad(no, f"지시 {m['key']} 의 값이 문자열이 아니다")
252:             if m["key"] == "profile" and value != "C":
253:                 raise bad(no, f"profile 이 C 가 아니다 ({value!r})")
254:             seen[m["key"]] = value
255:             continue
256:         m = _DIST.fullmatch(line)
257:         if m is None:
258:             raise bad(no, f"모르는 줄 {line[:80]!r}")
259:         name, version, record = m["name"], m["version"], m["record"]
260:         if not _NAME.fullmatch(name):
261:             raise bad(no, f"정규형 아닌 이름 {name!r}")
262:         if record != "none" and not _HEX64.fullmatch(record):
263:             raise bad(no, f"record-sha256 칸 {record!r}")
264:         if last is not None and name <= last:
265:             raise bad(no, f"배포판 정렬 · 중복 ({last!r} 다음 {name!r})")
266:         dists[name] = {"version": version, "record": record}
267:         last = name
268:     missing = [k for k in DIRECTIVES if k not in seen]
269:     if missing:
270:         raise LockFormatError(f"지시가 없다: {', '.join(missing)}")
271:     out = {k: seen[k] for k in DIRECTIVES}
272:     out.update(dists=dists, shadowed=shadowed)
273:     return out

282: def compare(locked: dict, measured: dict) -> list:
283:     """축마다 다른 것을 모은다 → (axis, subject) 순 목록. 빈 목록 = 비교한 모든 축이 같다."""
284:     out = []
285:     for k in DIRECTIVES[1:]:
286:         if locked[k] != measured[k]:
287:             out.append(_mm(k, k, locked[k], measured[k]))
288:     lock_d, got_d = locked["dists"], measured["dists"]
289:     for n in sorted(lock_d.keys() - got_d.keys()):
290:         out.append(_mm("dist_missing", n, lock_d[n]["version"], None))
291:     for n in sorted(got_d.keys() - lock_d.keys()):
292:         out.append(_mm("dist_extra", n, None, got_d[n]["version"]))
293:     for n in sorted(lock_d.keys() & got_d.keys()):
294:         if lock_d[n]["version"] != got_d[n]["version"]:
295:             out.append(_mm("dist_version", n, lock_d[n]["version"], got_d[n]["version"]))
296:         if lock_d[n]["record"] != got_d[n]["record"]:
297:             out.append(_mm("dist_record", n, lock_d[n]["record"], got_d[n]["record"]))
298:     lock_s = Counter(tuple(x) for x in locked["shadowed"])
299:     got_s = Counter(tuple(x) for x in measured["shadowed"])
300:     for n, v, r in sorted((lock_s - got_s).elements()):
301:         out.append(_mm("shadowed", f"{n}=={v}", r, None))
302:     for n, v, r in sorted((got_s - lock_s).elements()):
303:         out.append(_mm("shadowed", f"{n}=={v}", None, r))
304:     for n, rel, want, got in measured["files"]["mismatches"]:
305:         out.append(_mm("file", f"{n}:{rel}", want, got))
306:     for module, seen in measured["path_origins"]["mismatches"]:
307:         out.append(_mm("path_origin", module, "경로 검색 origin 이 RECORD 가 있는 유효 배포판 하나의 파일", seen))
308:     return sorted(out, key=lambda x: (x["axis"], x["subject"]))

319: def compare_lock(lock_path=None, *, paths=None) -> dict:
320:     """lock 과 지금 환경을 대조한 닫힌 결과 dict (고정 표 §4-2 · §13-3). 예외를 내지 않는다 — 못 하면 `UNMEASURED`.
321: 
322:     `not_measured` 는 측정 칸이 아니라 범위 선언이라 세 상태 모두 같은 값이다 (`UNMEASURED` 에서도 `None` 이 아니다).
323:     """
324:     lp = Path(LOCK_DEFAULT if lock_path is None else lock_path)
325:     res = {"profile": "C", "lock_path": _display(lp), "lock_sha256": None, "status": "UNMEASURED",
326:            "reason": "", "mismatches": None, "unverifiable": None, "counts": None,
327:            "not_measured": list(NOT_MEASURED)}
328:     try:
329:         raw = lp.read_bytes()
330:     except OSError as exc:
331:         res["reason"] = f"lock 을 읽지 못했다: {exc}"
332:         return res
333:     res["lock_sha256"] = hashlib.sha256(raw).hexdigest()
334:     try:
335:         locked = parse_lock(raw.decode("utf-8"))
336:     except UnicodeDecodeError as exc:
337:         res["reason"] = f"lock 이 UTF-8 이 아니다: {exc}"
338:         return res
339:     except LockFormatError as exc:
340:         res["reason"] = f"lock 형식 오류 — {exc}"
341:         return res
342:     try:
343:         measured = measure(paths)
344:     except Exception as exc:                              # noqa: BLE001 — 측정 실패는 typed UNMEASURED 다
345:         res["reason"] = f"측정 실패: {exc!r}"
346:         return res
347:     found = compare(locked, measured)
348:     res.update(
349:         status="MISMATCH" if found else "MATCH",
350:         mismatches=found,
351:         unverifiable={"dists_without_record": list(measured["dists_without_record"]),
352:                       "path_origins": list(measured["path_origins"]["unverifiable"])},
353:         counts={"dists_locked": len(locked["dists"]), "dists_measured": len(measured["dists"]),
354:                 "shadowed_locked": len(locked["shadowed"]), "shadowed_measured": len(measured["shadowed"]),
355:                 "files_verified": measured["files"]["verified"], "files_unhashed": measured["files"]["unhashed"],
356:                 "path_origins_in_record": len(measured["path_origins"]["in_record"])})
357:     return res

362: def _summary(r: dict) -> str:
363:     head = f"환경 프로필 C 대조: {r['status']} · lock {r['lock_path']}"
364:     if r["lock_sha256"]:
365:         head += f" (sha256 {r['lock_sha256'][:16]}…)"
366:     if r["status"] == "UNMEASURED":
367:         return head + f"\n  이유: {r['reason']}\n  (기록 전용 — 아무것도 막지 않는다 · 원장 §135)"
368:     c = r["counts"]
369:     lines = [head + f" · 배포판 lock {c['dists_locked']} / 측정 {c['dists_measured']} · 가려진 {c['shadowed_locked']}"
370:              f" / {c['shadowed_measured']} · 설치 파일 일치 {c['files_verified']} (해시 없음 {c['files_unhashed']})"
371:              f" · 경로 검색 origin 의 RECORD 소속 {c['path_origins_in_record']} (로드된 module origin 미측정)"]
372:     u = r["unverifiable"]
373:     if u["dists_without_record"]:
374:         lines.append(f"  · 확인 불가 — RECORD 없는 배포판 {len(u['dists_without_record'])}: "
375:                      + ", ".join(u["dists_without_record"]))
376:     if u["path_origins"]:
377:         lines.append("  · 확인 불가 — 경로 검색 origin: " + ", ".join(u["path_origins"]))
378:     for m in r["mismatches"]:
379:         lines.append(f"  ✗ {m['axis']} {m['subject']} locked={m['locked']} measured={m['measured']}")
380:     lines.append("  (기록 전용 — 불일치여도 아무것도 막지 않는다 · 원장 §135)")
381:     return "\n".join(lines)

387:     ap.add_argument("--lock", default=None, help="대조할 lock (기본: 프로젝트 root 의 requirements-validation-C.lock.txt)")
388:     mode = ap.add_mutually_exclusive_group()
389:     mode.add_argument("--json", action="store_true", help="결과 dict 를 JSON 으로")
390:     mode.add_argument("--emit-lock", action="store_true", help="지금 환경의 정규형 lock 을 낸다")
391:     a = ap.parse_args(argv)

399:             return 1

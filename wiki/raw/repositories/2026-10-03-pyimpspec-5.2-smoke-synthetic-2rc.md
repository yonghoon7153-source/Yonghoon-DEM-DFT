# pyimpspec 5.2.0 — 버리는 venv 설치 · 합성 2-RC 스펙트럼 연기 시험 (2026-10-03 실측)

## 조건 · 격리

- 운영 Python 은 3.11.15 — pyimpspec 5.2 (`requires-python >=3.12`) 는 거기에 깔 수 없다. 운영 환경에는 아무것도 깔지 않았다.
- 버리는 venv = 스크래치패드 (저장소 밖) `…/scratchpad/pyimpspec_try/venv312`. 만든 명령 (그대로): `uv venv -q --python /usr/bin/python3.12 venv312 && timeout 900 /root/.local/bin/uv pip install -q --python venv312/bin/python "pyimpspec==5.2.0"`.
- 첫 실행에서 `tr-rbf` DRT 가 선택 의존성 (볼록 최적화기) 없이 멈췄다 (아래 원문). 같은 venv 에만 `uv pip install --python ./venv312/bin/python cvxopt` 로 더한 뒤 다시 돌렸다. 아래 스크립트 · 출력은 **두 번째 실행** 그대로다 (첫 실행의 `tr-nnls` 두 줄은 피크 높이만 찍었고 값이 같았다).
- 저장소 · 운영 등록부 · 산출물에 쓰지 않았다. 실행 시간: KK 약 9 s · DRT 각 0.2 s 이하.

## 시험 — 합성 2-RC (잡음 없음) · KK 1 회 · DRT 3 회

### `smoke.py` (실행한 그대로)

```python
"""pyimpspec 5.2.0 연기 시험 (버리는 venv · 저장소 밖) — '이 컨테이너에서 설치되고 돈다' 만 확인한다.

합성 스펙트럼: R0=10 Ω + (R1=20 Ω ∥ C1=1e-5 F) + (R2=50 Ω ∥ C2=1e-3 F)
참값: τ1 = R1·C1 = 2e-4 s · τ2 = R2·C2 = 5e-2 s · 피크 넓이 = R1 · R2 (∫γ d ln τ).
잡음 없음. KK(real) 한 번 · DRT tr-nnls λ 두 값 · tr-rbf 자동 λ 한 번.
피크 높이와 넓이를 따로 적는다 — 브리핑의 '피크는 정규화에 민감' 을 높이 · 넓이로 나눠 본다.
"""
import inspect
import time

import numpy as np
from pyimpspec import DataSet, calculate_drt, parse_cdc, perform_kramers_kronig_test

f = np.logspace(5, -2, 71)
circ = parse_cdc("R{R=10}(R{R=20}C{C=1e-5})(R{R=50}C{C=1e-3})")
Z = circ.get_impedances(f)
data = DataSet(frequencies=f, impedances=Z, label="2RC-synthetic")
print("numpy", np.__version__, "| n_f", len(f), "| Z(f_max)", np.round(Z[0], 4), "| Z(f_min)", np.round(Z[-1], 4))

t0 = time.time()
kk = perform_kramers_kronig_test(data, num_procs=1)
print("KK real: num_RC", kk.num_RC, "| pseudo chi2", f"{kk.pseudo_chisqr:.3e}", f"| {time.time()-t0:.1f}s")


def summarize(res):
    tau = np.asarray(res.get_time_constants())
    g = res.get_gammas()
    g = np.asarray(g[0] if isinstance(g, tuple) else g)
    lnt = np.log(tau)
    # 사다리꼴 적분 · 두 피크는 참값 τ 의 기하 평균 (√(2e-4·5e-2) = 3.16e-3 s) 에서 나눈다
    cut = np.sqrt(2e-4 * 5e-2)
    lo, hi = tau <= cut, tau > cut
    a1 = float(abs(np.trapezoid(g[lo], lnt[lo])))
    a2 = float(abs(np.trapezoid(g[hi], lnt[hi])))
    i1 = int(np.argmax(np.where(lo, g, -np.inf)))
    i2 = int(np.argmax(np.where(hi, g, -np.inf)))
    return f"τ_peak {tau[i1]:.2e} / {tau[i2]:.2e} s | height {g[i1]:.2f} / {g[i2]:.2f} Ω | area {a1:.2f} / {a2:.2f} Ω (truth 20 / 50)"


for lam in (1e-3, 1e-1):
    t0 = time.time()
    r = calculate_drt(data, method="tr-nnls", lambda_value=lam)
    print(f"tr-nnls λ={lam:g}: {summarize(r)} | {time.time()-t0:.1f}s")

t0 = time.time()
r = calculate_drt(data, method="tr-rbf", num_procs=1, timeout=120)
print(f"tr-rbf auto λ={r.lambda_value:.2e}: {summarize(r)} | {time.time()-t0:.1f}s")
if hasattr(r, "analyze_peaks"):
    print("analyze_peaks signature:", inspect.signature(r.analyze_peaks))
    try:
        pa = r.analyze_peaks()
        print(pa.to_peaks_dataframe().to_string())
    except Exception as e:  # 연기 시험 — 실패도 그대로 적는다
        print("analyze_peaks error:", type(e).__name__, e)
```

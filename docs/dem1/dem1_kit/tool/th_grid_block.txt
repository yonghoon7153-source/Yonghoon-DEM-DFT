A = (BOX * 1e-4) ** 2                                                    # cm2
VP, VS, VSE = [4 / 3 * math.pi * (r * 1e-4) ** 3 for r in (R_P, R_S, R_SE)]   # cm3
mP, mS, mSE = VP * RHO_AM, VS * RHO_AM, VSE * RHO_SE                    # g
def isprime(n):
    if n < 2: return False
    i = 2
    while i * i <= n:
        if n % i == 0: return False
        i += 1
    return True
def grid2(n):   # processors * * 1 → LIGGGHTS 가 고르는 x·y 격자 (표면적 최소 = 가장 정사각, 같으면 px 작은 쪽 · 로그의 'A by B by 1' 과 같은 순서)
    best = None
    for px in range(1, n + 1):
        if n % px: continue
        py = n // px
        if best is None or px + py < sum(best): best = (px, py)
    return best
def ins_max(N, H, NP):   # 삽입(z 분할 1 1 *)에서 proc 하나가 맡는 최대 개수 — 상자 z -0.01~1.0 을 NP 칸으로
    hz = 1.01 / NP; z0, z1 = 0.005, 0.005 + H; b = 0.0
    for k in range(NP):
        a, c = -0.01 + k * hz, -0.01 + (k + 1) * hz
        b = max(b, min(c, z1) - max(a, z0))
    return N * b / H

  grep -aq '^RESTART_READ_OK' $O
}

ins_ok(){ # C NP → 삽입 덱(z 분할)을 NP 코어로 돌려도 되는지 (proc 당 삽입수 x maxattempt < int32) · ERRM
  local F=$DT0/$1/input_$1_ins.liggghts N Z1 MA M
  [ -s $F ] || { ERRM="삽입 덱 input_$1_ins.liggghts 없음 (gen)"; return 1; }
  N=$(sed -n 's/^N_TOTAL=//p' $DT0/$1/th_case.txt 2>/dev/null); Z1=$(awk '$1=="region" && $2=="reg_mix"{print $9; exit}' $F < /dev/null)
  MA=$(awk '{for(i=1;i<NF;i++) if($i=="maxattempt"){print $(i+1); exit}}' $F < /dev/null)
  M=$(awk -v n="${N:-0}" -v z1="${Z1:-0}" -v np=$2 -v ma="${MA:-15000}" 'BEGIN{h=z1-0.005; hz=1.01/np; b=0; for(k=0;k<np;k++){a=-0.01+k*hz; c=a+hz; lo=(a>0.005)?a:0.005; hi=(c<z1)?c:z1; if(hi-lo>b) b=hi-lo}; m=(h>0)?n*b/h:n; m=m*1.02+10; print int(m), (m*ma<2147483647)?1:0}')
  [ "${M##* }" = 1 ] || { ERRM="${2}코어로 삽입하면 proc 하나에 ~${M%% *}개 x maxattempt ${MA} 가 int32 넘음 (삽입 0개) → 코어 늘려서"; return 1; }
}

arm(){ # C DEP → 다음 링크 (이미 대기 중인 게 있으면 그것) · NXT
  local P
  P=$(jobs_of $1 | awk '$3=="PENDING"{print $1}' | head -1)
  if [ -n "$P" ]; then NXT=$P; say "다음 링크 이미 있음: job $P"; return 0; fi
  NXT=$( (for v in $(env | sed -n 's/^\(SLURM[A-Za-z0-9_]*\)=.*/\1/p'); do [ "$v" = SLURM_CONF ] || unset $v; done; cd $DT0 && sbatch --parsable --dependency=afterany:$2 $DT0/$1/run_$1.sh 2> $ST/sbatch_$1.err) | tail -1 ); NXT=${NXT%%;*}
  case "$NXT" in ""|*[^0-9]*) say "⚠ 다음 링크 예약 실패 ($(tail -1 $ST/sbatch_$1.err)) — 5일 제한에 걸리면 resume $1 필요"; NXT=;; *) say "다음 링크 예약: job $NXT (afterany:$2)";; esac
}

submit_pair(){ # C → 첫 링크(보류로 제출) + 다음 링크(afterany) → 첫 링크 풀기 (잡 안 예약과 엇갈리지 않게)
  local J1 J2
  J1=$(cd $DT0 && sbatch --parsable --hold $1/run_$1.sh 2> $ST/sbatch_$1.err | tail -1); J1=${J1%%;*}
  case "$J1" in ""|*[^0-9]*) echo "⛔ $1: sbatch 실패 — $(tail -1 $ST/sbatch_$1.err)"; return 1;; esac
  J2=$(cd $DT0 && sbatch --parsable --dependency=afterany:$J1 $1/run_$1.sh 2> $ST/sbatch_$1.err | tail -1); J2=${J2%%;*}
  case "$J2" in ""|*[^0-9]*) J2="(실패: $(tail -1 $ST/sbatch_$1.err) — 잡 안에서 다시 예약)";; esac
  scontrol release $J1 > /dev/null 2>&1 || echo "⚠ $1: 보류 풀기 실패 → 직접: scontrol release $J1"
  echo "▶ $1: job $J1 · $(np_of $1)코어 · 다음 링크 $J2"
}

kill_case(){ # C → 대기 링크 취소 + 돌던 잡 KILL 을 큐에서 다 빠질 때까지 반복 (최대 ~90초 · squeue 가 실패하면 비었다고 보지 않음)
  local C=$1 i P R O
  for i in $(seq 1 45); do
    O=$(jobs_of $C) || { sleep 2; continue; }
    [ -z "$O" ] && return 0
    P=$(echo "$O" | awk '$3=="PENDING"{print $1}'); [ -n "$P" ] && scancel $P
    R=$(echo "$O" | awk '$3=="RUNNING" || $3=="CONFIGURING" || $3=="SUSPENDED" {print $1}'); [ -n "$R" ] && scancel -s KILL $R
    if [ $i -ge 30 ]; then scancel $(echo "$O" | awk '{print $1}'); fi
    sleep 2
  done
  return 1
}

_job(){ # 잡 안 (러너가 부름): 끝났으면 그냥 끝 · 아니면 다음 링크 예약 → 체크포인트 없으면 삽입(z 분할) → 최신 체크포인트에서 x·y 로
  local C=$1 W J NP PG RC LG OT IN N i
  [ -n "$SLURM_JOB_ID" ] || { echo "⛔ _job 은 Slurm 잡 안에서만 (러너가 부름)"; exit 1; }
  W=$DT0/$C; J=$SLURM_JOB_ID; NP=${SLURM_NTASKS:-$(np_of $C)}; LG=$DT0/logs/output_${C}_$J.out
  cd $W || exit 1
  say "링크 시작 · $C · job $J · ${NP}코어 · $(hostname)"
  [ -e $ST/$C.stop ] && { say "멈춤 표시 있음 ($(cat $ST/$C.stop)) → 안 돌림 (풀기: resume $C)"; cancel_pending $C; exit 0; }
  finished $C $LG && { say "이미 끝난 케이스 ($FINLOG) → 안 돌림"; cancel_pending $C; exit 0; }
  OT=$(squeue -u $USER -h -o "%i %j %T" | awk -v c=$C -v j=$J '$2==c && $3=="RUNNING" && $1+0<j+0 {print $1}' | head -1)
  if [ -n "$OT" ]; then say "같은 케이스 잡 $OT 이 먼저 돌고 있음 → 이 잡은 그냥 끝냄"; arm $C $OT; exit 0; fi
  for i in $(seq 1 45); do [ -z "$(squeue -u $USER -h -o "%i %j %T" | awk -v c=$C -v j=$J '$2==c && $1+0<j+0 && $3 ~ /COMPLETING|CONFIGURING|RUNNING/')" ] && break; sleep 2; done
  for i in python3 lmp_mpi mpirun sbatch timeout; do command -v $i > /dev/null || { stopmark $C "$i 못 찾음 (러너 환경 줄 확인)"; cancel_pending $C; exit 1; }; done
  PG=$(prog $C); echo "$J ${PG:-0}" >> $ST/$C.prog
  if [ "$(tail -3 $ST/$C.prog | awk -v p="${PG:-0}" '$2==p' | wc -l)" -ge 3 ]; then stopmark $C "3링크 연속 진전 없음 (체크포인트 ${PG:-0})"; cancel_pending $C; exit 1; fi
  for i in 1 2 3; do
    build $C rs$J; RC=$?
    [ $RC = 0 ] || break
    rcheck $CKF && break
    say "⚠ ${CKF##*/} 를 LIGGGHTS 가 못 읽음 ($(grep -a '^ERROR' $W/rcheck_$C.out | head -1 | cut -c1-90)) → 빼고 이전 체크포인트로"
    echo "${CKF##*/} $(mt $CKF)" >> $ST/$C.bad; rm -f "$D" "$PL"; RC=1; ERRM="체크포인트 3개 연속 못 읽음 (잘못 걸렀으면 ~/.th_grid/$C.bad 지우고 resume)"
  done
  [ $RC = 1 ] && { stopmark $C "재개 덱 실패: $ERRM"; cancel_pending $C; exit 1; }
  if [ $RC = 2 ]; then ins_ok $C $NP || { stopmark $C "$ERRM"; cancel_pending $C; exit 1; }; fi
  arm $C $J
  if [ $RC = 2 ]; then
    N=$(sed -n 's/^N_TOTAL=//p' th_case.txt 2>/dev/null)
    say "체크포인트 없음 → 삽입: input_${C}_ins.liggghts (z 분할 1x1x$NP · 넣고 restart_settling_1 저장)"
    mpirun --oversubscribe --bind-to none -np $NP lmp_mpi -in input_${C}_ins.liggghts
    IN=$(grep -a 'Particle insertion ins_mix: inserted' $LG < /dev/null | head -1 | awk '{for(i=1;i<NF;i++) if($i=="inserted") print $(i+1)}')
    if errlog $LG; then stopmark $C "삽입 단계 ERROR (job $J)"; cancel_pending $C; exit 1; fi
    if grep -aq 'Less insertions than requested' $LG < /dev/null; then stopmark $C "삽입 부족 ${IN:-?}/~$N (vf·maxattempt 확인)"; cancel_pending $C; exit 1; fi
    grep -aq '^====== INSERTED' $LG < /dev/null && [ -s restart_$C/restart_settling_1.bin ] || { stopmark $C "삽입 단계가 끝까지 안 감 (job $J)"; cancel_pending $C; exit 1; }
    say "삽입 ${IN:-?}개 (예상 ~$N) → x·y 분할로 PHASE 1 부터"
    build $C rs$J; RC=$?; [ $RC = 0 ] || { stopmark $C "삽입 뒤 재개 덱 실패: $ERRM"; cancel_pending $C; exit 1; }
    rcheck $CKF || { stopmark $C "삽입 restart 를 LIGGGHTS 가 못 읽음 (rcheck_$C.out)"; cancel_pending $C; exit 1; }
  fi
  say "재개 → ${D##*/} · $INFO"
  echo "$J ${NXT:--} ${D##*/} $(date '+%m-%d_%H:%M')" >> $ST/$C.chain
  mpirun --oversubscribe --bind-to none -np $NP lmp_mpi -in ${D##*/}
  RC=$?
  if grep -aq "^====== $C Finished" $LG < /dev/null; then say "✅ 끝 → 남은 링크 취소"; cancel_pending $C
  elif errlog $LG; then stopmark $C "LIGGGHTS 에러 (job $J)"; cancel_pending $C
  else say "mpirun 끝 (rc $RC) · 다음 링크가 이어받음"; fi
}

dry(){ # C → 지금 재개하면 쓸 덱을 만들어 검사만 (만든 파일은 지움) · 0 OK / 2 체크포인트 없음 / 1 실패
  local RC; build $1 chk; RC=$?
  rm -f $DT0/$1/input_$1_chk.liggghts $DT0/$1/plate_$1_chk.stl $DT0/$1/plan_$1_chk.txt
  return $RC
}

setnp(){ sed -i -E "s/^(#SBATCH +-n +)[0-9]+/\1$2/" $DT0/$1/run_$1.sh; }

start(){
  local C RC
  echo "지금 내 코어: 돌고 있는 것 $(squeue -u $USER -h -t R -o %C | awk '{s+=$1} END{print s+0}') / 60"
  for C in $(for C in $CASES; do N=$(np_of $C); echo "${N:-0} $C"; done | sort -rn | awk '{print $2}'); do
    [ -s $DT0/$C/run_$C.sh ] && [ -s $DT0/$C/input_$C.liggghts ] && [ -s $DT0/$C/input_${C}_ins.liggghts ] || { echo "⛔ $C: 덱·러너 없음 → gen 먼저"; continue; }
    [ -z "$(jobs_of $C)" ] || { echo "⏭ $C: 이미 큐에 있음"; continue; }
    finished $C && { echo "⏭ $C: 이미 끝남 ($FINLOG)"; continue; }
    [ -e $ST/$C.stop ] && { echo "⏭ $C: 멈춤 표시 ($(cat $ST/$C.stop)) → 풀려면 resume $C"; continue; }
    dry $C; RC=$?
    [ $RC = 1 ] && { echo "⛔ $C: $ERRM"; continue; }
    [ $RC = 2 ] && { ins_ok $C $(np_of $C) || { echo "⛔ $C: $ERRM"; continue; }; }
    rm -f $ST/$C.prog; submit_pair $C
  done
  sleep 2; squeue -u $USER -o "%.10i %.10j %.8T %.4C %.12M %R"
}

resume(){
  local A C N RC
  for A in "$@"; do C=${A%%:*}; N=; [ "$C" = "$A" ] || N=${A##*:}
    case "$N" in *[^0-9]*) echo "⛔ $A — CASE 또는 CASE:코어"; continue;; esac
    [ -s $DT0/$C/run_$C.sh ] || { echo "⛔ $C: 러너 없음"; continue; }
    [ -z "$(jobs_of $C)" ] || { echo "⛔ $C: 큐에 있음 → 코어 바꾸기는 cores $C:N"; continue; }
    finished $C && { echo "⏭ $C: 이미 끝남 ($FINLOG)"; continue; }
    dry $C; RC=$?
    [ $RC = 1 ] && { echo "⛔ $C: $ERRM"; continue; }
    [ $RC = 2 ] && { ins_ok $C ${N:-$(np_of $C)} || { echo "⛔ $C: $ERRM"; continue; }; }
    [ -n "$N" ] && setnp $C $N
    rm -f $ST/$C.stop $ST/$C.prog
    echo "… $C: $( [ $RC = 0 ] && echo "$INFO" || echo "체크포인트 없음 → 삽입부터")"; submit_pair $C
  done
}

cores(){
  local A C N RC NW AGE
  for A in "$@"; do C=${A%%:*}; N=${A##*:}
    case "$N" in ""|*[^0-9]*) echo "⛔ $A — CASE:코어 형식"; continue;; esac
    [ "$C" = "$A" ] && { echo "⛔ $A — CASE:코어 형식"; continue; }
    [ -n "$(jobs_of $C)" ] || { echo "⛔ $C: 큐에 없음 → resume $C:$N"; continue; }
    NW=$(command ls -t $DT0/$C/restart_$C 2>/dev/null | head -1); AGE=$(( $(date +%s) - $(mt $DT0/$C/restart_$C/$NW) ))
    [ -n "$(jobs_of $C | awk '$3=="RUNNING"')" ] && [ -n "$NW" ] && [ $AGE -le ${TH_GUARD_S:-60} ] && { echo "⛔ $C: 체크포인트 $NW 지금 쓰는 중 — 1분 뒤 다시"; continue; }
    dry $C; RC=$?
    [ $RC = 1 ] && { echo "⛔ $C: 재개 덱 검사 실패 — $ERRM (잡은 그대로)"; continue; }
    [ $RC = 2 ] && { ins_ok $C $N || { echo "⛔ $C: $ERRM (잡은 그대로)"; continue; }; }
    echo "… $C: $(np_of $C) → ${N}코어 · $( [ $RC = 0 ] && echo "$INFO · 체크포인트 $((AGE/60))분 전 (그 뒤 진행분은 다시 돎)" || echo "체크포인트 없음 → 삽입부터")"
    kill_case $C || { echo "⛔ $C: 옛 잡이 큐에서 안 빠짐 → 잠시 뒤 resume $C:$N"; continue; }
    setnp $C $N; rm -f $ST/$C.stop $ST/$C.prog; submit_pair $C
  done
  sleep 2; squeue -u $USER -o "%.10i %.10j %.8T %.4C %.12M %R"
}

stopc(){
  local C
  for C in "$@"; do [ -d $DT0/$C ] || { echo "⛔ $C: 폴더 없음"; continue; }
    echo "사용자 stop $(date '+%m-%d %H:%M')" > $ST/$C.stop
    kill_case $C && echo "■ $C: 멈춤 (다시: resume $C)" || echo "⛔ $C: 큐에서 안 빠짐 — 잠시 뒤 다시"; done
}

watch1(){
  local C R J NP PD PR LG G PH LS LA LP EL LEFT SS MS S1 Z RT CKP ER STT HS ZH TG P1 ETA NOW SQ
  NOW=$(date +%s); SQ=$(squeue -u $USER -h -o "%i %j %T %C %M %L %S %R" 2>/dev/null)
  echo "══ 후막 6케이스 · $(date '+%m-%d %H:%M:%S') · 내 코어 R $(echo "$SQ" | awk '$3=="RUNNING"{s+=$4} END{print s+0}') / 60 (대기 $(echo "$SQ" | awk '$3=="PENDING"{s+=$4} END{print s+0}')) ══"
  printf "%-8s %-8s %-10s %-10s %-3s %-6s %-9s %-8s %-7s %-7s %-9s %-5s %-7s %-6s %-17s %-8s %s\n" case job elapsed left np grid phase step atoms P_MPa plate_z z/hs kstep/h ETA ckpt next 상태
  for C in $CASES; do [ -d $DT0/$C ] || continue
    R=$(echo "$SQ" | awk -v c=$C '$2==c && $3=="RUNNING"' | head -1); J=$(echo "$R" | awk '{print $1}'); NP=$(echo "$R" | awk '{print $4}')
    PD=$(echo "$SQ" | awk -v c=$C '$2==c && $3=="PENDING"{print $1}' | head -1)
    LG=$(lastlog $C); G=$(grid "$LG")
    PH=$(grep -a '^======' "$LG" 2>/dev/null < /dev/null | tail -1)
    case "$PH" in *Finished*) PH=done;; *"PHASE 1"*) PH=P1-settle;; *"PHASE 2"*) PH=P2-stab;; *"PHASE 3"*) PH=P3-comp;; *"PHASE 4"*) PH=P4-relax;; *INSERT*) PH=insert;; *) PH=-;; esac
    read LS LA LP <<< "$( [ -n "$LG" ] && awk '/^ *Step +Atoms/{p=($NF=="pressMPa");c=0;for(i=1;i<=NF;i++)if($i=="CPU")c=i;q=1;next} q&&$1~/^[0-9]+$/{s=$1; a=$2; if(p&&c>0&&$c+0>0)v=$NF} END{print s+0, a+0, v}' "$LG" < /dev/null)"
    EL=-; LEFT=-; S1=; Z=; RT=-; ETA=-
    if [ -n "$J" ]; then read EL LEFT SS <<< "$(echo "$R" | awk '{print $5, $6, $7}')"
      MS=$(find $DT0/$C/post_$C -maxdepth 1 -name 'mesh_*.stl' -newermt "${SS/T/ }" -printf '%T@ %f\n' 2>/dev/null | awk '{f=$2; gsub(/[^0-9]/,"",f); split($1,t,"."); print f, t[1]}' | sort -n)
      [ -n "$MS" ] || MS=$(find $DT0/$C/restart_$C -maxdepth 1 -name 'restart_*_*.bin' -newermt "${SS/T/ }" -printf '%T@ %f\n' 2>/dev/null | awk '{f=$2; gsub(/[^0-9]/,"",f); split($1,t,"."); print f, t[1]}' | sort -n)
      S1=$(echo "$MS" | awk 'NF==2{s=$1} END{print s}'); [ -n "$S1" ] && [ "${LS:-0}" -lt "$S1" ] && LS=$S1
      [ -s $DT0/$C/post_$C/mesh_$S1.stl ] && Z=$(mz $DT0/$C/post_$C/mesh_$S1.stl)
      RT=$(echo "$MS" | awk -v now=$NOW 'NF==2 && $2>=now-3600 {n++; s[n]=$1; t[n]=$2} END{if(n>=2 && t[n]-t[1]>=600) printf "%.0f", (s[n]-s[1])*3.6/(t[n]-t[1]); else print "-"}')
      if [ "$RT" = - ]; then RT=$(echo "$MS" | awk 'NF==2{n++; s[n]=$1; t[n]=$2} END{if(n>=2 && t[n]-t[1]>=600) printf "%.0f", (s[n]-s[1])*3.6/(t[n]-t[1]); else print "-"}'); fi
      TG=$(awk '$1=="variable" && $2=="target_press"{print $4; exit}' $DT0/$C/input_$C.liggghts < /dev/null 2>/dev/null)
      if [ "$PH" = P3-comp ] && [ -n "$LP" ] && [ -n "${RT//-/}" ] && [ -n "$TG" ]; then
        P1=$(awk -v t=$(( ${LS:-0} - RT*1000 )) '/^ *Step +Atoms/{p=($NF=="pressMPa");c=0;for(i=1;i<=NF;i++)if($i=="CPU")c=i;q=1;next} q&&p&&c>0&&$c+0>0&&$1~/^[0-9]+$/{d=$1-t; if(d<0)d=-d; if(b==""||d<b){b=d; v=$NF}} END{print v}' "$LG" < /dev/null)
        ETA=$(awk -v a="$P1" -v b="$LP" -v g="$TG" 'BEGIN{d=b-a; if(length(a)>0 && d>0) printf "%.0fh", (g-b)/d; else print "-"}')
      fi
      if [ "$PH" = P4-relax ] && [ -n "${RT//-/}" ]; then P1=$(phase_start $C "4: RELAXATION"); [ -n "$P1" ] && ETA=$(awk -v a=$P1 -v s=${LS:-0} -v r=$RT 'BEGIN{x=(a+100000-s)/1000/r; printf "%.1fh", (x>0)?x:0}'); fi
    fi
    [ -z "$Z" ] && [ -n "$(lastmesh $C)" ] && Z=$(mz $DT0/$C/post_$C/mesh_$(lastmesh $C).stl)
    HS=$(sed -n 's/^HS_SIM=//p' $DT0/$C/th_case.txt 2>/dev/null); ZH=-; [ -n "$Z" ] && [ -n "$HS" ] && ZH=$(awk -v z=$Z -v h=$HS 'BEGIN{printf "%.2f", z/h}')
    CKP=$(command ls -t $DT0/$C/restart_$C 2>/dev/null | grep -E '^restart_(settling|compress)_[0-9]+\.bin$|^restart_after_settling\.bin$' | head -1 | sed -E 's/^restart_//; s/\.bin$//')
    ER=$(grep -acE '^ERROR( on proc [0-9]+)?:' "$LG" 2>/dev/null < /dev/null)
    PR=; [ -n "$PD" ] && PR=$(echo "$SQ" | awk -v j=$PD '$1==j{print $8}')
    if finished $C; then STT="✅끝"; elif [ -e $ST/$C.stop ]; then STT="⛔$(cut -c1-60 $ST/$C.stop)"; elif [ -n "$J" ]; then STT="돌고있음$( [ -z "$PD" ] && echo ' ⚠다음링크없음')"; elif [ -n "$PD" ]; then STT="대기 $PR"; else STT="멈춤→resume"; fi
    printf "%-8s %-8s %-10s %-10s %-3s %-6s %-9s %-8s %-7s %-7s %-9s %-5s %-7s %-6s %-17s %-8s %s\n" $C "${J:--}" "$EL" "$LEFT" "${NP:-$(np_of $C)}" "${G:--}" "$PH" "${LS:--}" "${LA:--}" "${LP:--}" "${Z:--}" "$ZH" "$RT" "$ETA" "${CKP:--}" "${PD:--}" "$STT${ER:+ ERR$ER}"
  done
  echo "(next = 5일 제한 대비 다음 링크 · step: 이번 잡 mesh 덤프·로그 중 앞선 값 · z/hs: 판 높이 / 고체 두께 · ETA: P3 는 최근 1시간 압력 기울기로 목표까지 선형, P4 는 이완 남은 step)"
}

main(){ case "$1" in
  gen) gen;;
  start) start;;
  watch) watch1;;
  cores) shift; cores "$@";;
  resume) shift; resume "$@";;
  stop) shift; stopc "$@";;
  _job) _job "$2";;
  *) sed -n '1,6p' "$0";;
esac; }
main "$@"; exit $?

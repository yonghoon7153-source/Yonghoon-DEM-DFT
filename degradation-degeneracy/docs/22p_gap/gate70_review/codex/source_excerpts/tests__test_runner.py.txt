61: def _dry_all(tmp_path, *extra):
62:     """`--mode all` 이 만들 하위 명령을 실제 실행 없이 받아온다."""
63:     import os
64:     import subprocess
65:     from pathlib import Path
66: 
67:     root = Path(__file__).resolve().parent.parent
68:     env = {**os.environ, "RUN_SH_DRY": "1"}
69:     r = subprocess.run(["bash", str(root / "run.sh"), "--mode", "all",
70:                         "--out", str(tmp_path / "o"), *extra],
71:                        cwd=root, capture_output=True, text=True, env=env)
72:     assert r.returncode == 0, r.stdout + r.stderr
73:     return [ln for ln in r.stdout.splitlines() if ln.startswith("--mode")]

76: def test_mode_all_propagates_the_fit_protocol_flags(tmp_path):

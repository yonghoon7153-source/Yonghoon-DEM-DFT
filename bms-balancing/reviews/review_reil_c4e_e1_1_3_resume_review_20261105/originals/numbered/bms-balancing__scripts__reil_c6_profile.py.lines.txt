185: def sobol(out: Path) -> dict:
186:     import numpy as np
187:     from scipy.stats import qmc
188:     params = inspect.signature(qmc.Sobol).parameters
189:     kw = "rng" if "rng" in params else "seed"
190:     arrays, warn = {}, []
191:     for seed in SOBOL_SEEDS:
192:         with warnings.catch_warnings(record=True) as caught:
193:             warnings.simplefilter("always")
194:             unit = qmc.Sobol(d=SOBOL_D, scramble=True, **{kw: seed}).random_base2(m=SOBOL_M)
195:         warn += [f"{w.category.__name__}: {w.message}" for w in caught]
196:         arrays[f"unit_seed{seed}"] = unit
197:         for box, (lo, hi) in BOXES.items():
198:             arrays[f"{box}_seed{seed}"] = qmc.scale(unit, lo, hi)
199:     # 부속 A §3-1 "판에 따라 seed 인자 이름과 생성 배열이 다를 수 있으므로" — 옛 이름 seed= 가 같은 정수로 같은 배열을 내는지 (1.17.1: 아니다 · 경고 없음)
200:     legacy = {}
201:     if kw == "rng" and "seed" in params:
202:         for seed in SOBOL_SEEDS:
203:             with warnings.catch_warnings(record=True) as caught:
204:                 warnings.simplefilter("always")
205:                 old = qmc.Sobol(d=SOBOL_D, scramble=True, seed=seed).random_base2(m=SOBOL_M)
206:             gen = qmc.Sobol(d=SOBOL_D, scramble=True, rng=np.random.default_rng(seed)).random_base2(m=SOBOL_M)
207:             legacy[f"seed{seed}"] = {"seed_keyword_same_array_as_rng": bool(np.array_equal(old, arrays[f"unit_seed{seed}"])),
208:                                      "seed_keyword_warnings": sorted({f"{w.category.__name__}: {w.message}" for w in caught}),
209:                                      "rng_default_rng_same_array_as_rng_int": bool(np.array_equal(gen, arrays[f"unit_seed{seed}"]))}
210:     rec = {}
211:     for name, a in sorted(arrays.items()):
212:         a = np.ascontiguousarray(a, dtype="<f8")
213:         np.save(out / f"sobol_{name}.npy", a, allow_pickle=False)
214:         rec[name] = {"shape": list(a.shape), "raw_float64_le_sha256": _sha(a.tobytes()),
215:                      "npy_sha256": _sha((out / f"sobol_{name}.npy").read_bytes())}
216:     return {"call": f"scipy.stats.qmc.Sobol(d={SOBOL_D}, scramble=True, {kw}=<seed>).random_base2(m={SOBOL_M})",
217:             "seed_keyword": kw, "signature": str(inspect.signature(qmc.Sobol)), "boxes": BOXES,
218:             "warnings": sorted(set(warn)), "arrays": rec, "legacy_seed_keyword": legacy,

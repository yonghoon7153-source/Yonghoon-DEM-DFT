662: # ---------------------------------------------------------------- 자식 · 프로세스 그룹 (N2 · N3)
663: def proc_members(pgid, *, proc=PROC, read=None) -> dict:
664:     """/proc 에서 프로세스 그룹 pgid 의 구성원 [(pid, 상태)] · 사라진 PID 수 · 오류. 바이트로 읽고 comm (괄호 안) 은 해석하지 않는다 (V3-03 — UTF-8 이
665:     아닌 이름도 실패하지 않는다 · 주입한 read 가 문자열을 돌려줘도 같은 파싱). PID 소멸 (FileNotFoundError · ProcessLookupError) 만 "사라짐" 이고, 그 밖의
666:     모든 읽기 예외 (권한 · I/O · 디코딩 등) 와 파싱 · 목록 실패는 오류로 남기고 스캔을 계속한다 (관측 불능을 "구성원 없음" 으로 바꾸지 않는다)."""
667:     read = read or (lambda p: p.read_bytes())
668:     out = {"members": [], "vanished": 0, "errors": []}
669:     try:
670:         entries = [d for d in Path(proc).iterdir() if d.name.isdigit()]
671:     except Exception as e:  # noqa: BLE001 — 목록 실패는 관측 불능
672:         out["errors"].append({"where": "list", "error": f"{e.__class__.__name__}: {e}"})
673:         return out
674:     for d in entries:
675:         try:
676:             s = read(d / "stat")
677:         except (FileNotFoundError, ProcessLookupError):
678:             out["vanished"] += 1
679:             continue
680:         except Exception as e:  # noqa: BLE001 — 소멸이 아닌 모든 읽기 예외 (권한 · I/O · 디코딩 등) 는 그 PID 의 오류
681:             out["errors"].append({"pid": d.name, "error": f"{e.__class__.__name__}: {e}"})
682:             continue
683:         try:
684:             b = s if isinstance(s, bytes) else s.encode("utf-8", "surrogateescape")
685:             rest = b[b.rindex(b")") + 2:].split()
686:             state, pgrp = rest[0].decode("ascii"), int(rest[2])
687:         except Exception as e:  # noqa: BLE001 — 파싱 · 디코딩 실패
688:             out["errors"].append({"pid": d.name, "error": f"파싱 실패: {e.__class__.__name__}: {e}"})
689:             continue
690:         if pgrp == pgid:
691:             out["members"].append((int(d.name), state))
692:     return out

695: def group_status(pgid, *, grace=GROUP_GRACE_S, until=None, clock=time.monotonic, sleep=time.sleep, proc=PROC, read=None) -> dict:
696:     """그룹 관측 창 — 관측은 `until` (주지 않으면 지금 + grace) 전에만 시작한다 (V3-01 규칙 · 관측 하나의 길이는 묶지 않는다). 좀비 아닌 구성원 0 ·
697:     오류 0 → empty · 관측 오류 → unknown (바로 돌아간다 · 뒤의 관측으로 지우지 않는다) · 창 끝까지 살아 있음 → alive · 관측을 하나도 시작하지 못함 →
698:     unverified. sleep 은 남은 시간만큼만 잔다 (0 이하이면 자지 않는다)."""
699:     until = clock() + grace if until is None else until
700:     n, alive, zombies = 0, None, []
701:     while clock() < until:
702:         o = proc_members(pgid, proc=proc, read=read)
703:         n += 1
704:         alive = [m for m in o["members"] if m[1] != "Z"]
705:         zombies = [m for m in o["members"] if m[1] == "Z"]
706:         if o["errors"]:
707:             return {"status": "unknown", "alive": alive, "zombies": zombies, "errors": o["errors"], "observations": n}
708:         if not alive:
709:             return {"status": "empty", "alive": [], "zombies": zombies, "errors": [], "observations": n}
710:         left = until - clock()
711:         if left > 0:
712:             sleep(min(POLL_S, left))
713:     if n == 0:
714:         return {"status": "unverified", "alive": None, "zombies": [], "errors": [], "observations": 0}
715:     return {"status": "alive", "alive": alive, "zombies": zombies, "errors": [], "observations": n}

731: def cleanup_group(p, timed_out, *, deadline, clock=time.monotonic, sleep=time.sleep, killpg=os.killpg, kill_wait=KILL_WAIT_S,
732:                   grace=GROUP_GRACE_S, proc=PROC, read=None) -> dict:
733:     """정리 (launcher 운영 정리 정책 · V3-01 규칙) — 기한 (`deadline` · run_child 가 정리 단계에 들어갈 때 한 번 만든 절대 시각) 은 받기만 한다.
734:     timeout 이면 그룹 SIGKILL → 자식 종료를 비차단 확인 (`poll`) 하며 기다림 (kill_wait · 기한 안) → 그룹 관측 창 (grace · 기한 안) → 살아 있으면 그룹
735:     SIGKILL (기한 뒤라도 보낸다 — 기다리지 않는 동작) → 재관측 창 (grace · 기한 안). 관측 · sleep 은 기한 전에만 시작한다. 끝 상태 = empty (재kill 없이
736:     비어 있음) · killed (재kill 뒤 비어 있음) · alive · unverified (관측을 못 함 · 재kill 뒤 재관측을 못 함) · unknown (관측 오류 · 정리 안의 예외 — 우선).
737:     안의 예외는 여기서 잡아 같은 기한으로 비상 동작 (자식이 살아 있으면 그룹 SIGKILL · 관측) 만 한다 — 새 기한은 없다."""
738:     t0 = clock()
739:     rec = {"kill": [], "exited_after_kill": None, "rc_after_kill": None, "group_first": None, "group": None, "errors": [], "emergency": False}
740: 
741:     def window():
742:         return group_status(p.pid, until=min(deadline, clock() + grace), clock=clock, sleep=sleep, proc=proc, read=read)
743: 
744:     try:
745:         if timed_out:
746:             rec["kill"].append(kill_group(p.pid, killpg))
747:             until = min(deadline, t0 + kill_wait)
748:             while p.poll() is None and clock() < until:
749:                 left = until - clock()
750:                 if left > 0:
751:                     sleep(min(POLL_S, left))
752:             rec["exited_after_kill"] = p.poll() is not None
753:             if rec["exited_after_kill"]:
754:                 rec["rc_after_kill"] = p.returncode
755:         g = window()
756:         if g["status"] == "alive":
757:             rec["group_first"] = g
758:             rec["kill"].append(kill_group(p.pid, killpg))
759:             g2 = window()
760:             g = dict(g2, status="killed") if g2["status"] == "empty" else g2
761:         rec["group"] = g
762:     except Exception as e:  # noqa: BLE001 — 정리 안의 예외 — 같은 기한으로 비상 동작만 (새 기한 없음)
763:         rec["errors"].append(f"{e.__class__.__name__}: {e}")
764:         rec["emergency"] = True
765:         try:
766:             if p.poll() is None:
767:                 rec["kill"].append(kill_group(p.pid, killpg))
768:             rec["group"] = group_status(p.pid, until=deadline, clock=clock, sleep=sleep, proc=proc, read=read)
769:         except Exception as e2:  # noqa: BLE001
770:             rec["errors"].append(f"비상 동작 {e2.__class__.__name__}: {e2}")
771:     status = "unknown" if rec["errors"] else (rec["group"] or {}).get("status", "unknown")
772:     rec["group"] = dict(rec["group"] or {}, status=status)
773:     rec.update(status=status, deadline_hit=clock() >= deadline, cleanup_s=round(clock() - t0, 3))
774:     return rec

777: def run_child(argv, *, cwd, env, stdout, stderr, timeout, popen=subprocess.Popen, killpg=os.killpg, kill_wait=KILL_WAIT_S,
778:               grace=GROUP_GRACE_S, cleanup_s=CLEANUP_S, proc=PROC, read=None, clock=time.monotonic, sleep=time.sleep) -> dict:
779:     """새 세션에서 자식을 띄우고 기다린다 (N3). 자식이 끝나거나 timeout 이 되면 정리 단계에 들어가며 공통 절대 기한 (원점 + cleanup_s) 을 한 번 만들어
780:     `cleanup_group` 에 넘긴다 (launcher 운영 정리 정책 · V3-01). 시작 실패 · timeout · kill 결과 · 비차단 종료 확인 · 그룹 상태 · 기한 도달을 모두
781:     기록하고, 어느 대기도 무기한이 아니다. 시간은 대기 (`wait_s` — 시작 → 종료 또는 timeout) 와 정리 (`cleanup_s` — 정리 원점부터 · kill 뒤 대기 ·
782:     관측 · 재kill · 재관측) 로 나눠 남긴다 (S2)."""
783:     rec = {"started": False, "rc": None, "timed_out": None, "errors": []}
784:     t = clock()
785:     try:
786:         p = popen(argv, cwd=str(cwd), env=env, stdout=stdout, stderr=stderr, start_new_session=True)
787:     except Exception as e:  # noqa: BLE001 — 시작 실패는 기록하고 돌아간다
788:         rec["start_error"] = f"{e.__class__.__name__}: {e}"
789:         return rec
790:     rec.update(started=True, pgid=p.pid, start_utc=utc())
791:     try:
792:         rec["rc"], rec["timed_out"] = p.wait(timeout=timeout), False
793:     except subprocess.TimeoutExpired:
794:         rec["timed_out"] = True
795:     except Exception as e:  # noqa: BLE001 — 대기 자체의 예외도 기록하고 정리로 간다 (timed_out 이 None 이라 판정은 false)
796:         rec["errors"].append(f"wait {e.__class__.__name__}: {e}")
797:     t_c0 = clock()
798:     rec["wait_s"] = round(t_c0 - t, 3)
799:     c = cleanup_group(p, rec["timed_out"] is True, deadline=t_c0 + cleanup_s, clock=clock, sleep=sleep, killpg=killpg, kill_wait=kill_wait,
800:                       grace=grace, proc=proc, read=read)
801:     if rec["timed_out"] and c["exited_after_kill"]:
802:         rec["rc"] = c["rc_after_kill"]
803:     rec.update(kill=c["kill"], exited_after_kill=c["exited_after_kill"], group=c["group"], group_first=c["group_first"],
804:                deadline_hit=c["deadline_hit"], emergency=c["emergency"], cleanup_deadline_s=cleanup_s)
805:     rec["errors"] += c["errors"]
806:     rec["cleanup_s"] = round(clock() - t_c0, 3)
807:     return rec

816: def launch(step, evidence, section, *, repo=REPO, py=PY, venv=VENV, seal=SEAL, work=WORK, data=DATA, data_files=DATA_FILES,
817:            sources=SOURCES, sources_sha256=SOURCES_SHA256, seal_manifest_sha256=SEAL_MANIFEST_SHA256, lock_sha256=C6_LOCK_SHA256,
818:            extra=EXTRA_EXECUTED, helper_sha256=IDENTITY_HELPER_SHA256, timeouts=TIMEOUTS, env_base=None, popen=subprocess.Popen,
819:            killpg=os.killpg, kill_wait=KILL_WAIT_S, grace=GROUP_GRACE_S, cleanup_s=CLEANUP_S, proc=PROC, read=None) -> int:
820:     """준비 → 사전 대조 → (통과할 때만) 시작 기록 → (성공할 때만) 자식 → 끝 기록 → 사후 대조 → 끝 관측 → 단계 판정 → meta. 반환 0 = 단계 판정
821:     complete · 1 = 아님 (meta 있음 · 자식 시작 여부는 `launched`) · 4 = meta 를 쓰지 못함 (stderr 한 줄 · 첫 오류 포함). 의도된 거부 (단계 이름 · 기존 기록
822:     파일 · 기존 pycache 폴더 — lstat 기준 · 끊어진 링크 포함) 는 어떤 파일 · 폴더도 만들기 전 SystemExit 이고, 기록용 `except Exception` 은 SystemExit 를
823:     잡지 않는다 (V2-04 · V3-02)."""
824:     if step not in STEPS:
825:         raise SystemExit(f"단계는 {sorted(STEPS)} 뿐이다: {step!r}")
826:     ev, name, bms = Path(evidence), STEPS[step], Path(repo) / "bms-balancing"
827:     files = {k: ev / f"{name}.{k}" for k in ("precheck.json", "meta.json", "stdout.log", "stderr.log", "porcelain_start.txt",
828:                                              "porcelain_end.txt")}
829:     prefix = Path(work) / "pycache" / name
830:     meta = {"step": step, "name": name, "cwd": str(bms), "timeout_s": timeouts[step], "kill_wait_s": kill_wait, "group_grace_s": grace,
831:             "cleanup_deadline_s": cleanup_s, "pycache_prefix": str(prefix), "request": REQUEST, "approval_section": section,
832:             "start_utc": utc()}
833:     errors, records, phase = [], {}, "prepare"
834:     pre = child = post = detail = None
835:     ok = False
836:     try:
837:         taken = [str(p) for p in files.values() if entry_exists(p)]
838:         if taken:
839:             raise SystemExit(f"이미 있다 — 덮어쓰지 않는다: {taken}")
840:         if entry_exists(prefix):
841:             raise SystemExit(f"pycache 폴더 {prefix} 가 이미 있다 — 새 빈 폴더만")
842:         prefix.mkdir(parents=True)
843:         for k in ("mpl", "tmp"):
844:             (Path(work) / k).mkdir(parents=True, exist_ok=True)
845:         env, removed = child_env(dict(os.environ if env_base is None else env_base), work)
846:         if step == "fetch":  # FUTURE-01 — fetch 단계 전체 (사전 · 자식 · 사후 · 기록 git) 가 같은 fetch env
847:             env = fetch_env(env)
848:         meta.update(env_removed_names=removed, env_set={k: env[k] for k in ENV_SET_KEYS},
849:                     env_git_config={k: env[k] for k in sorted(env) if k.startswith("GIT_CONFIG")})
850:         argv = build_argv(step, ev, section, prefix, py=py, bms=bms, venv=venv, seal=seal, data=data, sources=sources,
851:                           sources_sha256=sources_sha256, seal_manifest_sha256=seal_manifest_sha256)
852:         meta["argv"] = argv
853:         me = Path(__file__).resolve()
854:         meta["launcher"] = {"path": str(me), "sha256": sha256(me.read_bytes())}
855:         meta["disk_free_start_bytes"] = shutil.disk_usage(bms).free
856:         phase = "precheck"
857:         try:
858:             pre = precheck(step, ev, argv, prefix, env=env, repo=repo, py=py, seal=seal, work=work, data=data, data_files=data_files,
859:                            sources=sources, sources_sha256=sources_sha256, seal_manifest_sha256=seal_manifest_sha256,
860:                            lock_sha256=lock_sha256, extra=extra, helper_sha256=helper_sha256)
861:         except Exception as e:  # noqa: BLE001
862:             errors.append({"phase": "precheck", "type": e.__class__.__name__, "message": str(e)[:500]})
863:             pre = {"ok": False, "problems": [f"사전 대조 예외 {e.__class__.__name__}: {e}"]}
864:         phase = "record"
865:         write_new(files["precheck.json"], dump(pre))
866:         if pre.get("ok") is True and not errors:
867:             records["head_start"] = git_out(repo, "rev-parse", "HEAD", env=env)
868:             records["porcelain_start"] = git_out(repo, "status", "--porcelain=v1", "--untracked-files=all", env=env)
869:             write_new(files["porcelain_start.txt"], records["porcelain_start"]["out"])
870:             start_bad = record_problems(records, heads=("head_start",), porcelains=("porcelain_start",))
871:             if start_bad:  # S4 — 시작 기록이 실패하면 자식을 띄우지 않는다 (사후 false 를 기다리지 않는다)
872:                 errors.append({"phase": "record", "type": "GitRecordFailed", "message": "; ".join(start_bad)[:500]})
873:             else:
874:                 phase = "start"
875:                 with open(files["stdout.log"], "xb") as fo, open(files["stderr.log"], "xb") as fe:
876:                     child = run_child(argv, cwd=bms, env=env, stdout=fo, stderr=fe, timeout=timeouts[step], popen=popen, killpg=killpg,
877:                                       kill_wait=kill_wait, grace=grace, cleanup_s=cleanup_s, proc=proc, read=read)
878:                 if not child.get("started"):
879:                     errors.append({"phase": "start", "type": "StartError", "message": str(child.get("start_error"))[:500]})
880:                 errors += [{"phase": "wait", "type": "ChildError", "message": str(msg)[:500]} for msg in child.get("errors") or []]
881:                 phase = "record"
882:                 records["porcelain_end"] = git_out(repo, "status", "--porcelain=v1", "--untracked-files=all", env=env)
883:                 write_new(files["porcelain_end.txt"], records["porcelain_end"]["out"])
884:                 records["head_end"] = git_out(repo, "rev-parse", "HEAD", env=env)
885:                 if child.get("started"):
886:                     ok, detail = child_result(step, files["stdout.log"].read_text(encoding="utf-8", errors="replace"))
887:                 phase = "postcheck"
888:                 post = postcheck(step, ev, pre, prefix, child, env=env, repo=repo, data=data, data_files=data_files, extra=extra,
889:                                  records=records)
890:     except Exception as e:  # noqa: BLE001 — 어느 단계의 예외든 기록하고 판정을 false 로 둔다 (SystemExit 는 잡지 않는다)
891:         errors.append({"phase": phase, "type": e.__class__.__name__, "message": str(e)[:500]})
892:     for key, look in (("disk_free_end_bytes", lambda: shutil.disk_usage(bms).free),
893:                       ("evidence_bytes_before_meta", lambda: sum(p.stat().st_size for p in ev.rglob("*") if p.is_file()))):
894:         try:
895:             meta[key] = look()
896:         except Exception as e:  # noqa: BLE001 — 끝 관측의 실패도 오류로 남기고 meta 는 계속 쓴다
897:             errors.append({"phase": "end", "type": e.__class__.__name__, "message": str(e)[:500]})
898:     launched = bool(child and child.get("started"))
899:     stage = stage_verdict(precheck_ok=bool(pre and pre.get("ok") is True), child_ok=ok, rc=(child or {}).get("rc"),
900:                           timed_out=(child or {}).get("timed_out"), postcheck_ok=bool(post and post.get("ok") is True) and not errors)
901:     meta.update(launched=launched, child=child, child_result=detail, postcheck=post, stage=stage, errors=errors,
902:                 first_error=errors[0] if errors else None, precheck_ok=bool(pre and pre.get("ok") is True),
903:                 precheck_problems=(pre or {}).get("problems"),
904:                 records={k: {kk: vv for kk, vv in v.items() if kk != "out"} | ({"out": v.get("out", "").strip()} if k.startswith("head") else {})
905:                          for k, v in records.items()},
906:                 end_utc=utc(), child_maxrss_kib=resource.getrusage(resource.RUSAGE_CHILDREN).ru_maxrss)
907:     try:
908:         write_new(files["meta.json"], dump(meta))
909:     except Exception as e:  # noqa: BLE001 — 기록 자체가 불가능하면 외부 stderr · rc 4 로 한계를 분리한다
910:         sys.stderr.write(json.dumps({"step": step, "meta_write_failed": f"{e.__class__.__name__}: {e}", "launched": launched,
911:                                      "first_error": errors[0] if errors else None, "complete": False}, ensure_ascii=False) + "\n")
912:         return 4
913:     print(json.dumps({"step": step, "launched": launched, "complete": stage["complete"], "rc": (child or {}).get("rc"),
914:                       "timed_out": (child or {}).get("timed_out"), "errors": len(errors)}, ensure_ascii=False))
915:     return 0 if stage["complete"] else 1

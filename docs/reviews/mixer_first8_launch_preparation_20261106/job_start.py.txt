#!/usr/bin/env python3
"""Inactive job-start draft. Run only from sealed approved single-case runner."""
import hashlib,json,os,pathlib,subprocess,sys,time
_EXECUTING_CODE=sys._getframe().f_code
_SOURCE_BYTES=pathlib.Path(__file__).read_bytes()
if compile(_SOURCE_BYTES,__file__,"exec") != _EXECUTING_CODE:
 raise ValueError("job-start source differs from executing code")
_SOURCE_SHA=hashlib.sha256(_SOURCE_BYTES).hexdigest()

def need(ok,why):
 if not ok:raise ValueError(why)
def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
def main():
 need(len(sys.argv)==2,'approval path required')
 ap=pathlib.Path(sys.argv[1]);approval_raw=ap.read_bytes();approval_sha=hashlib.sha256(approval_raw).hexdigest();c=json.loads(approval_raw)
 need(c.get('execution_authorized') is True and c.get('gate') in ('QUALIFICATION_PILOT_GO','MAP_CASE_GO'),'approval absent')
 need(c.get('np')==20 and os.environ.get('SLURM_NTASKS')=='20','allocated MPI ranks mismatch')
 jid=os.environ.get('SLURM_JOB_ID','');need(jid.isdigit(),'not a Slurm job')
 root=pathlib.Path(c['case_dir']);need(root.is_absolute() and root.is_dir() and not root.is_symlink(),'case root')
 claim=root/'submission.claim.json';need(claim.is_file(),'submission claim absent')
 need(json.loads(claim.read_text()).get('approval_sha256')==approval_sha,'claim differs')
 for k in ('author_go','qualification','mxd2r_disposition','live_preflight','source_manifest','policy','runner','binary','mpi_launcher'):
  a=c[k];p=pathlib.Path(a['path']);need(p.is_absolute() and p.is_file() and not p.is_symlink() and sha(p)==a['sha256'],'changed '+k)
 manifest=json.loads(pathlib.Path(c['source_manifest']['path']).read_text())
 own=pathlib.Path(__file__).resolve();need(any(x['path']==str(own) and x['sha256']==_SOURCE_SHA for x in manifest['files']),'job-start source not sealed')
 for a in manifest['files']:
  p=pathlib.Path(a['path']);need(p.is_file() and not p.is_symlink() and sha(p)==a['sha256'],'input/source drift')
 pol=json.loads(pathlib.Path(c['policy']['path']).read_text());need(pol.get('execution_authorized') is True and pol.get('allowed_to_submit')==[c['case']],'policy scope')
 need(c.get('wave_release')==[c['case']] and c.get('author_choices_confirmed') is True,'case release')
 for n in ('post','restart','log.liggghts','job_start.json'):need(not (root/n).exists(),'not fresh '+n)
 # Mandatory explicit bytes/inodes reserve, independently budgeted before approval.
 b=c.get('minimum_free_bytes');i=c.get('minimum_free_inodes')
 need(type(b) is int and b>0 and type(i) is int and i>0,'storage thresholds unknown')
 v=os.statvfs(root);need(v.f_bavail*v.f_frsize>=b and v.f_favail>=i,'storage headroom')
 # Quota and live allocation are preflight receipts; filesystem free space cannot prove quota.
 argv=[c['mpi_launcher']['path'],'--oversubscribe','--bind-to','none','-np','20',c['binary']['path'],'-in','in.mixer']
 fd=os.open(str(root/'job_start.json'),os.O_WRONLY|os.O_CREAT|os.O_EXCL,0o600)
 with os.fdopen(fd,'w') as f:json.dump({'job_id':jid,'approval_sha256':approval_sha,'case':c['case'],'argv':argv,'start_unix':time.time()},f)
 env=dict(os.environ,OMP_NUM_THREADS='1',OPENBLAS_NUM_THREADS='1',MKL_NUM_THREADS='1')
 with (root/'simulator.stdout').open('x') as out,(root/'simulator.stderr').open('x') as err:
  r=subprocess.run(argv,cwd=root,env=env,stdout=out,stderr=err)
 (root/'job_end.json').write_text(json.dumps({'job_id':jid,'returncode':r.returncode,'end_unix':time.time(),'state':'EXITED_REQUIRES_QUALIFICATION' if r.returncode==0 else 'TECH_HOLD'})+'\n')
 return r.returncode
if __name__=='__main__':
 try:sys.exit(main())
 except Exception as e:print('HOLD: '+str(e),file=sys.stderr);sys.exit(2)

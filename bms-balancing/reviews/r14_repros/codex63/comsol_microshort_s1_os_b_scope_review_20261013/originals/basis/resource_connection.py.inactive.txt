"""Inactive OS-evidence projection. No ctypes, WinDLL, processes, sleeps or I/O.
OS collector ABI/counter and stop capabilities remain OPEN. Pure code is untested.
"""
from math import isfinite
from dataclasses import dataclass


class ResourceConnectionError(ValueError):
    def __init__(self, reason):
        self.reason, self.stage = reason, 'resource_connection'
        super().__init__(reason)


def must(ok, reason):
    if not ok:
        raise ResourceConnectionError(reason)


def nonnegative_int(v):
    return type(v) is int and v >= 0


def finite_time(v):
    # Reject unrepresentable clock values explicitly, without inventing a limit.
    try:
        return type(v) in (int, float) and isfinite(v) and v >= 0
    except OverflowError:
        return False


@dataclass(frozen=True, slots=True)
class ExecutionClockContext:
    """Primitive-only start evidence, captured independently of resource samples.
    This binding does not establish that an OS collector measured the evidence.
    The collector and installed clock provenance remain OPEN.
    """
    run_id: str
    clock_id: str
    clock_origin: str
    clock_unit: str
    overall_origin_s: float
    native_origin_s: float

    def __post_init__(self):
        for name in ('run_id', 'clock_id', 'clock_origin'):
            value = getattr(self, name)
            must(type(value) is str and bool(value), 'CONTEXT_IDENTITY:'+name)
        must(type(self.clock_unit) is str and self.clock_unit == 'seconds',
             'CONTEXT_CLOCK_UNIT')
        for name in ('overall_origin_s', 'native_origin_s'):
            must(finite_time(getattr(self, name)), 'CONTEXT_ORIGIN_INVALID:'+name)
        must(self.overall_origin_s <= self.native_origin_s, 'CONTEXT_ORIGIN_ORDER')


@dataclass(frozen=True, slots=True, init=False)
class ResourceSession:
    """One immutable binding per execution; decide cannot accept a new origin.
    The future caller must construct this once from independent start evidence,
    before its first resource sample, and retain it for compile and batch.
    No OS collector, scheduler, stop dispatcher or native authorization is added.
    """
    clock: ExecutionClockContext
    anchor_items: tuple

    def __init__(self, start_record, anchors):
        must(type(start_record) is dict and type(anchors) is dict, 'CONTEXT_SCHEMA')
        names = ('run_id', 'clock_id', 'clock_origin', 'clock_unit',
                 'overall_origin_s', 'native_origin_s')
        for name in names:
            must(name in start_record, 'CONTEXT_MISSING:'+name)
        clock = ExecutionClockContext(**{k:start_record[k] for k in names})
        must(type(anchors.get('run_id')) is str and anchors['run_id'] == clock.run_id,
             'CONTEXT_RUN_BINDING')
        # Snapshot scalar anchors. Caller dictionary mutation cannot rebind us.
        must(all(type(k) is str and type(v) in (str, int, bool, type(None))
                 for k, v in anchors.items()), 'CONTEXT_ANCHOR_SCHEMA')
        object.__setattr__(self, 'clock', clock)
        object.__setattr__(self, 'anchor_items', tuple(anchors.items()))

    def decide(self, control, raw, previous_sample_end_s, contract):
        return resource_decision(control, raw, dict(self.anchor_items),
                                 previous_sample_end_s, contract, self.clock)


def bind_sample_clock(raw, anchors, context):
    must(type(context) is ExecutionClockContext, 'CLOCK_CONTEXT_MISSING')
    must(type(anchors.get('run_id')) is str and anchors['run_id'] == context.run_id,
         'CLOCK_ANCHOR_RUN_MISMATCH')
    for name in ('run_id', 'clock_id', 'clock_origin', 'clock_unit'):
        must(name in raw, 'CLOCK_BINDING_MISSING:'+name)
        must(type(raw[name]) is str and raw[name] == getattr(context, name),
             'CLOCK_BINDING_MISMATCH:'+name)
    for name in ('overall_origin_s', 'native_origin_s'):
        must(name in raw, 'CLOCK_ORIGIN_MISSING:'+name)
        must(finite_time(raw[name]), 'CLOCK_INVALID:'+name)
        must(raw[name] == getattr(context, name), 'CLOCK_ORIGIN_MISMATCH:'+name)
    return context


def project_owned_sample(raw, anchors, previous_sample_end_s, context):
    """Consume ONE bounded sampling transaction with two membership bookends.
    Collector must fail on churn; no zero for exited/inaccessible members.
    The submitted implementation only validates supplied observations.
    """
    must(type(raw) is dict and type(anchors) is dict, 'SAMPLE_SCHEMA')
    must(raw.get('errors') == [] and raw.get('query_success') is True, 'OS_QUERY_FAILURE')
    context = bind_sample_clock(raw, anchors, context)
    must(raw.get('job_identity') == anchors.get('job_identity') and bool(anchors.get('job_identity'))
         and anchors.get('assigned_before_resume') is True and
         anchors.get('held_job_and_root_handles') is True, 'OWNERSHIP_ANCHOR')
    before, after = raw.get('members_before'), raw.get('members_after')
    must(type(before) is list and before == after and len(before) > 0, 'MEMBERSHIP_CHURN_OR_EMPTY')
    must(all(type(x) is dict and set(x) == {'pid','creation_filetime'} and
             type(x['pid']) is int and x['pid'] > 0 and
             type(x['creation_filetime']) is int and x['creation_filetime'] > 0 for x in before), 'MEMBER_IDENTITY')
    pairs = [(x['pid'],x['creation_filetime']) for x in before]
    must(len(pairs) == len(set(pairs)) and len({x[0] for x in pairs}) == len(pairs), 'MEMBER_DUPLICATE')
    root = (anchors.get('root_pid'), anchors.get('root_creation_filetime'))
    must(type(root[0]) is int and root[0] > 0 and type(root[1]) is int and root[1] > 0,
         'ROOT_ANCHOR_SCHEMA')
    # Root may exit before its owned children: never add it again or invent a live root.
    must(root in pairs or raw.get('held_root_signaled') is True, 'ROOT_IDENTITY_MISSING')
    must(raw.get('membership_source') == 'QUERY_PRIVATE_HELD_JOB'
         and raw.get('member_times_source') == 'HELD_MEMBER_PROCESS_HANDLES', 'MEMBERSHIP_SOURCE')
    processes = raw.get('process_counters')
    must(type(processes) is list and len(processes) == len(pairs), 'RSS_MEMBERS_MISSING')
    counters = {}
    for p in processes:
        must(type(p) is dict, 'PROCESS_COUNTER_SCHEMA')
        must({'pid','creation_filetime','query_success','working_set_bytes'} <= set(p),
             'PROCESS_COUNTER_KEYS')
        must(type(p['pid']) is int and p['pid'] > 0 and
             type(p['creation_filetime']) is int and p['creation_filetime'] > 0,
             'PROCESS_COUNTER_IDENTITY_TYPE')
        key = (p['pid'],p['creation_filetime'])
        must(key in pairs and key not in counters and p.get('query_success') is True
             and nonnegative_int(p.get('working_set_bytes')), 'RSS_IDENTITY_OR_QUERY')
        counters[key] = p['working_set_bytes']
    must(raw.get('job_commit_source') == 'REVIEWED_JOB_ACCOUNTING_TOTAL'
         and raw.get('job_commit_counter_mapping_sha256') == anchors.get('accepted_commit_mapping_sha256')
         and bool(anchors.get('accepted_commit_mapping_sha256')),
         'JOB_COMMIT_MAPPING_OPEN')
    for k in ('job_private_commit_bytes','host_commit_limit_bytes','host_commit_total_bytes',
              'host_available_ram_bytes','owned_output_growth_bytes','volume_free_bytes'):
        must(nonnegative_int(raw.get(k)), 'COUNTER_MISSING:'+k)
    must(raw['host_commit_limit_bytes'] >= raw['host_commit_total_bytes'], 'COMMIT_UNDERFLOW')
    must(raw.get('output_walk_status') == 'COMPLETE_NO_REPARSE_NO_UNKNOWN_FILE'
         and raw.get('output_root') == anchors.get('owned_output_root')
         and bool(anchors.get('owned_output_root')), 'OUTPUT_SCOPE')
    for k in ('sample_start_s','sample_end_s'):
        must(finite_time(raw.get(k)), 'CLOCK_INVALID:'+k)
    start, end = raw['sample_start_s'], raw['sample_end_s']
    must(finite_time(previous_sample_end_s) and
         context.overall_origin_s <= context.native_origin_s <= start <= end
         and previous_sample_end_s <= start, 'CLOCK_ORDER')
    ownership = {'status':'VERIFIED_PRIVATE_JOB','assigned_before_resume':True,
                 'identity_fresh':True,'job_identity':anchors['job_identity'],
                 'root_pid':root[0],'root_creation_filetime':root[1],
                 'member_pid_creation_pairs':before}
    sample = {'status':'MEASURED','counter_errors':[],
              'scope':'EXACT_PRIVATE_JOB_MEMBERS_PLUS_OWNED_OUTPUT_TREE',
              'job_identity':anchors['job_identity'],
              'rss_aggregation':'SUM_UNIQUE_MEMBER_WORKING_SETS_NO_JOB_TOTAL_ADD',
              'commit_aggregation':'JOB_ACCOUNTING_TOTAL_NO_MEMBER_TOTAL_ADD',
              'monotonic_s':end,'native_elapsed_s':end-context.native_origin_s,
              'overall_elapsed_s':end-context.overall_origin_s,
              'sample_gap_s':end-previous_sample_end_s,'sampling_duration_s':end-start,
              'job_rss_bytes':sum(counters.values()),
              'job_private_commit_bytes':raw['job_private_commit_bytes'],
              'host_commit_available_bytes':raw['host_commit_limit_bytes']-raw['host_commit_total_bytes'],
              'host_available_ram_bytes':raw['host_available_ram_bytes'],
              'owned_output_growth_bytes':raw['owned_output_growth_bytes'],
              'volume_free_bytes':raw['volume_free_bytes']}
    return sample, ownership


def resource_decision(control, raw, anchors, previous_sample_end_s, contract, context):
    """Integration calls the unchanged actual assess_resources, never a duplicate."""
    try:
        sample, ownership = project_owned_sample(raw, anchors, previous_sample_end_s, context)
        return control.assess_resources(sample, ownership, contract)
    except ResourceConnectionError as exc:
        return {'status':'STOP_REQUIRED','reasons':[exc.reason],
                'sampled_max_is_os_peak':False,'counter_projection':'INCOMPLETE'}


def stop_action_plan(control, state, event, capabilities):
    """Pure result of accepted state machine; does NOT call any OS stop API."""
    result = control.advance_stop(state, event, capabilities)
    action = result['next_action']
    return {'transition':result, 'action':action, 'execute_now':False,
            'stop_classification':'FORCED' if action == 'FORCE_OWNED_JOB_ONCE' else
                                  'COOPERATIVE_REQUEST_UNOBSERVED' if action == 'REQUEST_COOPERATIVE_ONCE' else
                                  'NO_TERMINATION_CLAIM',
            'unknown_pid_termination_allowed':False,
            'provider_status':'OPEN_NO_OS_DISPATCH_IMPLEMENTED'}


class UnavailableOSProvider:
    """No implicit OS implementation / fallback. Native execute_once stays blocked."""
    def __getattr__(self, name):
        raise ResourceConnectionError('OS_PROVIDER_OPEN:'+name)


if __name__ == '__main__':
    raise SystemExit('OFFLINE_RESOURCE_CONNECTION_ONLY')

def kinetic(vs):
    return sum(sum(x*x for x in v) for v in vs) / 2


def momentum(vs):
    return [sum(v[k] for v in vs) for k in range(len(vs[0]))]


def bspline(x):
    x = abs(x)
    if x < Q(1, 2):
        return Q(3, 4) - x*x
    if x < Q(3, 2):
        return (Q(3, 2)-x)**2 / 2
    return Q(0)


def support(x):
    return [n for n in range(-5, 12) if bspline(x-n) > 0]


def result_for_jt(jt):
    jbc = (Q(21, 10) + jt) / 2
    vs = [[Q(-1, 2), -jt], [Q(-1, 2), Q(1, 10)+jt-jbc], [Q(0), -2+jbc]]
    slip = vs[1][1] - vs[0][1]
    return {'jt_on_B': jt, 'jbc_on_C': jbc, 'velocities': vs,
            'AB_final_slip': slip, 'AB_Jt_times_final_slip': jt*slip,
            'AB_Coulomb_bound_ok': abs(jt) <= Q(1, 4),
            'AB_final_Coulomb_law_ok': (slip == 0 and abs(jt) <= Q(1, 4))
                or (slip != 0 and abs(jt) == Q(1, 4) and jt*slip < 0),
            'kinetic_after': kinetic(vs), 'momentum_after': momentum(vs)}


def sequential(pair_order):
    vs = [[Q(0), Q(0)], [Q(-1), Q(1, 10)], [Q(0), Q(-2)]]
    abjt = Q(0)
    for pair in pair_order:
        if pair == 'AB':
            jn = -(vs[1][0]-vs[0][0])/2
            jt = max(-jn/2, min(jn/2, -(vs[1][1]-vs[0][1])/2))
            abjt = jt
            vs[0][0] -= jn
            vs[1][0] += jn
            vs[0][1] -= jt
            vs[1][1] += jt
        else:
            jn = -(vs[2][1]-vs[1][1])/2
            vs[1][1] -= jn
            vs[2][1] += jn
    slip = vs[1][1]-vs[0][1]
    return {'pair_order': pair_order, 'velocities': vs, 'AB_jt': abjt,
            'AB_final_slip': slip, 'AB_Jt_times_final_slip': abjt*slip,
            'BC_final_normal_separation_speed': vs[2][1]-vs[1][1],
            'kinetic_after': kinetic(vs), 'momentum_after': momentum(vs)}
